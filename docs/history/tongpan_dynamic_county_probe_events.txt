# Native engine feasibility test: a county office and a county governor are peers.
namespace = tca_tongpan_probe_test

tca_tongpan_probe_test.1 = {
    type = character_event
    hidden = yes
    immediate = {
        if = { limit = { has_character_flag = tca_test_delayed_case_active } debug_log = "TCA_TEST CASE_BUSY" }
        else = {
            debug_log = "TCA_TEST BEGIN tca_tongpan_probe_test.1 tca_test_tongpan_peer_diarchy"
            remove_character_flag = tca_test_case_failed
            if = {
                limit = {
                    debug_only = yes
                    title:h_china.holder = root
                    government_has_flag = government_is_celestial
                    exists = title:d_jiangxi.holder
                    exists = title:c_hongzhou.holder
                }
                # Independent fixture, not a pre-existing governor or judicial office.
                create_character = { age = 40 gender_female_chance = 0 random_traits = no dynasty = none employer = root culture = root.culture faith = root.faith rite = root.rite save_scope_as = tca_test_tp_duke }
                create_character = { age = 35 gender_female_chance = 0 random_traits = no dynasty = none employer = root culture = root.culture faith = root.faith rite = root.rite save_scope_as = tca_test_tp_governor }
                create_character = { age = 35 gender_female_chance = 0 random_traits = no dynasty = none employer = root culture = root.culture faith = root.faith rite = root.rite save_scope_as = tca_test_tp_officer }
                scope:tca_test_tp_duke = {
                    create_dynasty = { spread_to_descendants = yes }
                    give_noble_family_title = { name = noble_family_name tier = county article = DEFAULT_TITLE_NAME_ARTICLE government = celestial_government }
                    create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_test_tp_change }
                    title:d_jiangxi = { change_title_holder = { holder = scope:tca_test_tp_duke change = scope:tca_test_tp_change } }
                    resolve_title_and_vassal_change = scope:tca_test_tp_change
                    change_government = celestial_government
                    create_title_and_vassal_change = { type = swear_fealty add_claim_on_loss = no save_scope_as = tca_test_tp_fealty }
                    change_liege = { liege = root change = scope:tca_test_tp_fealty }
                    resolve_title_and_vassal_change = scope:tca_test_tp_fealty
                    set_primary_title_to = title:d_jiangxi
                    change_merit = 10000
                    tca_test_protect_character_effect = yes
                }
                scope:tca_test_tp_governor = {
                    create_dynasty = { spread_to_descendants = yes }
                    give_noble_family_title = { name = noble_family_name tier = county article = DEFAULT_TITLE_NAME_ARTICLE government = celestial_government }
                    create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_test_tp_change }
                    title:c_hongzhou = { change_title_holder = { holder = scope:tca_test_tp_governor change = scope:tca_test_tp_change } }
                    resolve_title_and_vassal_change = scope:tca_test_tp_change
                    change_government = celestial_government
                    create_title_and_vassal_change = { type = swear_fealty add_claim_on_loss = no save_scope_as = tca_test_tp_fealty }
                    change_liege = { liege = scope:tca_test_tp_duke change = scope:tca_test_tp_fealty }
                    resolve_title_and_vassal_change = scope:tca_test_tp_fealty
                    set_primary_title_to = title:c_hongzhou
                    change_merit = 10000
                    add_character_flag = tca_test_tongpan_governor
                    tca_test_protect_character_effect = yes
                }
                scope:tca_test_tp_duke = { create_dynamic_title = { tier = county name = tca_tongpan_probe_title } }
                scope:new_title = {
                    save_scope_as = tca_test_tp_office
                    set_capital_county = title:c_hongzhou
                    set_landless_title = yes
                    set_de_jure_liege_title = title:d_jiangxi
                    set_delete_on_destroy = no
                    set_destroy_on_succession = no
                    set_no_automatic_claims = yes
                    set_variable = { name = tca_test_tongpan_county value = title:c_hongzhou }
                }
                scope:tca_test_tp_officer = {
                    create_dynasty = { spread_to_descendants = yes }
                    give_noble_family_title = { name = noble_family_name tier = county article = DEFAULT_TITLE_NAME_ARTICLE government = celestial_government }
                    create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_test_tp_change }
                    scope:tca_test_tp_office = { change_title_holder = { holder = scope:tca_test_tp_officer change = scope:tca_test_tp_change } }
                    resolve_title_and_vassal_change = scope:tca_test_tp_change
                    change_government = celestial_government
                    create_title_and_vassal_change = { type = swear_fealty add_claim_on_loss = no save_scope_as = tca_test_tp_fealty }
                    change_liege = { liege = scope:tca_test_tp_duke change = scope:tca_test_tp_fealty }
                    resolve_title_and_vassal_change = scope:tca_test_tp_fealty
                    set_primary_title_to = scope:tca_test_tp_office
                    add_realm_law_skip_effects = celestial_appointment_succession_law
                    change_merit = 10000
                    add_character_flag = tca_test_tongpan_officer
                    tca_test_protect_character_effect = yes
                }
                if = { limit = {
                    scope:tca_test_tp_duke = { primary_title = title:d_jiangxi liege = root }
                    scope:tca_test_tp_governor = { primary_title = title:c_hongzhou liege = scope:tca_test_tp_duke }
                    scope:tca_test_tp_officer = { primary_title = scope:tca_test_tp_office liege = scope:tca_test_tp_duke is_landless_ruler = yes }
                    scope:tca_test_tp_office = { tier = tier_county is_landless_type_title = yes de_jure_liege = title:d_jiangxi var:tca_test_tongpan_county = title:c_hongzhou }
                } debug_log = "TCA_TEST CHECK_PASS tca_test_tongpan_peer_diarchy 1" }
                else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_tongpan_peer_diarchy 1" }
                # Official native sequence. Do not reparent the officer to make it pass.
                scope:tca_test_tp_governor = {
                    if = { limit = { has_active_diarchy = yes } end_diarchy = yes }
                    try_start_diarchy = tca_tongpan_probe
                    set_diarch = scope:tca_test_tp_officer
                    debug_log = "TCA_TONGPAN_PROBE after_set_diarch"
                    debug_log_scopes = yes
                }
                if = { limit = { scope:tca_test_tp_governor = { has_diarchy_type = tca_tongpan_probe diarch = scope:tca_test_tp_officer } }
                    debug_log = "TCA_TEST CHECK_PASS tca_test_tongpan_peer_diarchy 2" }
                else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_tongpan_peer_diarchy 2" }
                if = { limit = { scope:tca_test_tp_officer = { liege = scope:tca_test_tp_duke NOT = { is_vassal_of = scope:tca_test_tp_governor } has_realm_law = celestial_appointment_succession_law } }
                    debug_log = "TCA_TEST CHECK_PASS tca_test_tongpan_peer_diarchy 3" }
                else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_tongpan_peer_diarchy 3" }
                add_character_flag = tca_test_delayed_case_active
                debug_log = "TCA_TEST WAIT tca_test_tongpan_peer_diarchy 7 days"
                trigger_event = { id = tca_tongpan_probe_test.1001 days = 7 }
            }
            else = {
                debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_tongpan_peer_diarchy"
                if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
                tca_test_continue_delayed_effect = yes
            }
        }
    }
}

tca_tongpan_probe_test.1001 = {
    type = character_event
    hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = { limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            if = { limit = { scope:tca_test_tp_governor = { has_diarchy_type = tca_tongpan_probe diarch = scope:tca_test_tp_officer } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_tongpan_peer_diarchy 4" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_tongpan_peer_diarchy 4" }
            if = { limit = {
                scope:tca_test_tp_governor = { primary_title = title:c_hongzhou liege = scope:tca_test_tp_duke }
                scope:tca_test_tp_officer = { primary_title = scope:tca_test_tp_office liege = scope:tca_test_tp_duke is_landless_ruler = yes }
                scope:tca_test_tp_office = { de_jure_liege = title:d_jiangxi var:tca_test_tongpan_county = title:c_hongzhou }
            } debug_log = "TCA_TEST CHECK_PASS tca_test_tongpan_peer_diarchy 5" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_tongpan_peer_diarchy 5" }
            scope:tca_test_tp_governor = { debug_log = "TCA_TONGPAN_PROBE day_7" debug_log_scopes = yes }
            # Remove only state owned by this probe; no attempt to repair failure.
            scope:tca_test_tp_governor = {
                if = { limit = { has_diarchy_type = tca_tongpan_probe } end_diarchy = yes }
                remove_character_flag = tca_test_tongpan_governor
                tca_test_restore_character_effect = yes
            }
            scope:tca_test_tp_officer = { remove_character_flag = tca_test_tongpan_officer tca_test_restore_character_effect = yes }
            scope:tca_test_tp_duke = { tca_test_restore_character_effect = yes }
            if = { limit = { has_character_flag = tca_test_case_failed }
                debug_log = "TCA_TEST FAIL tca_test_tongpan_peer_diarchy"
                if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_failed add = 1 } }
            }
            else = {
                debug_log = "TCA_TEST PASS tca_test_tongpan_peer_diarchy"
                if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_passed add = 1 } }
            }
            remove_character_flag = tca_test_case_failed
            remove_character_flag = tca_test_delayed_case_active
            tca_test_continue_delayed_effect = yes
        }
    }
}
