# 只读查看：不授予或创建头衔，不收费，也不任命。
tca_view_judicial_titles_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
    desc = tca_view_judicial_titles_decision_desc
    selection_tooltip = tca_view_judicial_titles_decision_tooltip
    confirm_text = tca_view_judicial_titles_decision_confirm
    ai_check_interval = 0
    is_shown = {
        debug_only = yes
        is_ai = no
        OR = {
            any_held_title = { tca_has_judicial_title_trigger = yes }
            any_vassal_or_below = {
                any_held_title = { tca_has_judicial_title_trigger = yes }
            }
        }
    }
    is_valid = { always = yes }
    effect = {
        save_scope_as = tca_title_viewer
        every_held_title = {
            limit = { tca_has_judicial_title_trigger = yes }
            tca_show_judicial_title_effect = yes
        }
        every_vassal_or_below = {
            every_held_title = {
                limit = { tca_has_judicial_title_trigger = yes }
                tca_show_judicial_title_effect = yes
            }
        }
    }
    ai_will_do = { base = 0 }
}
