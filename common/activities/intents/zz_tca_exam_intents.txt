# 人物作用域；仅为原版乡试、会试应试意图增加共用任职门槛。
# 不限制其他活动意图；原版有地与及第条件继续保留。
imperial_exam_taker_intent = {
	icon = intent_study_hard

	auto_complete = yes

	on_invalidated = { trigger_event = activity_system.0090 }

	is_valid = {
		custom_tooltip = {
			text = tca_exam_not_serving_tt
			tca_exam_can_take_exam_trigger = yes
		}
		custom_tooltip = {
			text = already_taken_exam_tooltip
			NOT = { has_character_flag = passed_palace_exam }
		}
		custom_tooltip = {
			text = already_landed_tooltip
			is_landed = no
			NOT = { has_character_flag = passed_palace_exam }
		}
	}

	ai_will_do = {
		value = 100
		# 优先让尚未通过本级考试的AI选择应试。
		if = {
			limit = {
				NOT = {
					has_character_flag = passed_metropolitan_exam
				}
			}
			add = 5000
		}
	}

	scripted_animation = {
		animation = writing
	}
}
