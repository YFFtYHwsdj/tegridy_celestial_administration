# Native display rules use actual-liege loyalty; Tongpan peers use their bound governor.

is_diarch_visibly_loyal = {
    trigger_if = {
        limit = { tca_is_tongpan_for_loyalty_trigger = yes }
        tca_tongpan_loyalty_value >= diarch_loyalty_visibly_loyal_threshold
    }
    trigger_else = { diarch_loyalty >= diarch_loyalty_visibly_loyal_threshold }
}

is_diarch_visibly_disloyal = {
    trigger_if = {
        limit = { tca_is_tongpan_for_loyalty_trigger = yes }
        tca_tongpan_loyalty_value <= diarch_loyalty_visibly_disloyal_threshold
    }
    trigger_else = { diarch_loyalty <= diarch_loyalty_visibly_disloyal_threshold }
}
