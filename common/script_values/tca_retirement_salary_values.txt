# 原版00_basic_modifiers.txt的Merit Level基础权重；未入品及九品均为0。
# 仅供权重读数及回归预期使用；正式官品折半由官品契约承担，不能再添加全份M。
tca_retirement_merit_salary_base_value = {
    value = 0
    if = { limit = { merit_level = 2 } value = 0.05 }
    else_if = { limit = { merit_level = 3 } value = 0.1 }
    else_if = { limit = { merit_level = 4 } value = 0.15 }
    else_if = { limit = { merit_level = 5 } value = 0.2 }
    else_if = { limit = { merit_level = 6 } value = 0.25 }
    else_if = { limit = { merit_level = 7 } value = 0.3 }
    else_if = { limit = { merit_level = 8 } value = 0.5 }
    else_if = { limit = { merit_level = 9 } value = 0.75 }
}
# 基础分配权重只由官品和身份决定，不是固定月金币，也不复制预算分母或支付规则。
tca_retirement_salary_weight_value = {
    value = 0.05
    add = tca_retirement_merit_salary_base_value
    if = { limit = { has_trait = tca_retired } multiply = 0.5 }
}
