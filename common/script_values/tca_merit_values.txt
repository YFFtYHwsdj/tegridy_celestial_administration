# 人物作用域：激活后核定等级是同链真值；其余人物保留原版读取。
tca_merit_level_value = {
    value = merit_level
    if = { limit = { has_character_flag = tca_merit_active has_variable = tca_merit_approved_level } value = var:tca_merit_approved_level }
}

# 资格上限L按考试最高档取值，同档只取一档、不叠加；不得读取已锁定的实际cap。
# 默认1（九品）；乡试／跳级／荫补同属乡试档2（八品）；会试或殿试7（三品，普通铨叙最高）。
# 不再把各项修正相加，也不设max=9。中央授品的二品／一品不读取L，仍由授品mode分支处理。
# 只读修正存在性，不读原生上限；考试修正以 tca_merit_exam_scale_value=0 隔离。
tca_merit_theoretical_cap_value = {
    value = 1
    # 会试与殿试同档；两者同时持有仍只计7。
    if = {
        limit = {
            OR = {
                has_character_modifier = tgp_passed_metropolitan_exam_modifier
                has_character_modifier = tgp_passed_palace_exam_modifier
            }
        }
        value = 7
    }
    # 乡试、跳级、荫补同档；同时持有多项修正仍只计2，不能累加。
    else_if = {
        limit = {
            OR = {
                has_character_modifier = tgp_passed_provincial_exam_modifier
                has_character_modifier = tgp_bypassed_provincial_exam_modifier
                has_character_modifier = tgp_nepotism_bypass_exams_modifier
            }
        }
        value = 2
    }
}

tca_merit_next_threshold_value = {
    value = 100000000
    if = { limit = { tca_merit_level_value = 0 } value = merit_level_1_value }
    else_if = { limit = { tca_merit_level_value = 1 } value = merit_level_2_value }
    else_if = { limit = { tca_merit_level_value = 2 } value = merit_level_3_value }
    else_if = { limit = { tca_merit_level_value = 3 } value = merit_level_4_value }
    else_if = { limit = { tca_merit_level_value = 4 } value = merit_level_5_value }
    else_if = { limit = { tca_merit_level_value = 5 } value = merit_level_6_value }
    else_if = { limit = { tca_merit_level_value = 6 } value = merit_level_7_value }
    else_if = { limit = { tca_merit_level_value = 7 } value = merit_level_8_value }
    else_if = { limit = { tca_merit_level_value = 8 } value = merit_level_9_value }
}

# 原生scale只在添加时求值，初始化与离境必须刷新实际持有的考试修正。
# 未激活人物保留原版，允许0→1自然入品；贤能制始终保持原版贡献。
tca_merit_exam_scale_value = {
    value = 1
    if = { limit = { has_character_flag = tca_merit_active exists = top_liege top_liege ?= { government_has_flag = government_is_celestial } government_has_flag = government_is_celestial } value = 0 }
}
# 人物作用域；原版无自有统治头衔时使用DEFAULT_MERIT_LEVEL_CAP=1，天朝统治者政体基数为2。
# government_has_flag会沿廷臣雇主判断，不能据此误认本人拥有政体人物修正。
tca_merit_native_cap_base_value = {
    value = 1
    if = { limit = { is_ruler = yes } value = 2 }
}

# 原版倍率保持非负且与旧存档快照兼容；廷臣的一档差额由独立正修正补齐。
tca_merit_approved_reduction_value = { value = 9 subtract = var:tca_merit_approved_level }
tca_merit_courtier_cap_compensation_value = { value = 2 subtract = tca_merit_native_cap_base_value }
