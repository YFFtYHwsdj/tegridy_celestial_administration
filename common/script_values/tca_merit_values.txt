# 人物作用域：激活后核定等级是同链真值；其余人物保留原版读取。
tca_merit_level_value = {
    value = merit_level
    if = { limit = { has_character_flag = tca_merit_active has_variable = tca_merit_approved_level } value = var:tca_merit_approved_level }
}

    value = 2

    if = { limit = { has_character_modifier = tgp_bypassed_provincial_exam_modifier } add = 2 }
    if = { limit = { has_character_modifier = tgp_nepotism_bypass_exams_modifier } add = 2 }

    if = { limit = { has_character_modifier = tgp_passed_palace_exam_modifier } add = 3 }
    max = 9
}
