# 人物作用域：激活后核定等级是同链真值；其余人物保留原版读取。
tca_merit_level_value = {
    value = merit_level
    if = { limit = { has_character_flag = tca_merit_active has_variable = tca_merit_approved_level } value = var:tca_merit_approved_level }
}

# 资格上限L按考试最高档取值，同档只取一档、不叠加；不得读取已锁定的实际cap。
# 默认1（九品）；乡试／跳级／荫补同属乡试档2（八品）；会试或殿试7（三品，普通铨叙最高）。
# 不再把各项修正相加，也不设max=9。中央授品的二品／一品不读取L，仍由授品mode分支处理。
# 只读修正存在性，不读原生上限；考试修正以 tca_merit_exam_scale_value=0 隔离。
tca_merit_theoretical_cap_value = {
    value = 1
    # 会试与殿试同档；两者同时持有仍只计7。
    if = {
        limit = {
            OR = {
                has_character_modifier = tgp_passed_metropolitan_exam_modifier
                has_character_modifier = tgp_passed_palace_exam_modifier
            }
        }
        value = 7
    }
    # 乡试、跳级、荫补同档；同时持有多项修正仍只计2，不能累加。
    else_if = {
        limit = {
            OR = {
                has_character_modifier = tgp_passed_provincial_exam_modifier
                has_character_modifier = tgp_bypassed_provincial_exam_modifier
                has_character_modifier = tgp_nepotism_bypass_exams_modifier
            }
        }
        value = 2
    }
}

# 人物作用域：计算结余相对进度百分比 (0..160)
tca_merit_carryover_rate_percent_value = {
    value = 0
    if = {
        limit = {
            exists = scope:tca_merit_before
            var:tca_merit_grant_source >= 1
            var:tca_merit_grant_target < 9
            var:tca_merit_grant_mode = 1
            scope:tca_merit_before > tca_merit_target_threshold_value
        }
        # 分子: E * MaxRate (最大约 30000 * 160 = 4.8e6，安全不溢出)
        value = scope:tca_merit_before
        subtract = tca_merit_target_threshold_value
        multiply = tca_merit_carryover_max_rate_value
        # 分母: E + k * D_old
        divide = {
            value = scope:tca_merit_before
            subtract = tca_merit_target_threshold_value
            add = {
                value = tca_merit_source_span_value
                multiply = tca_merit_carryover_scale_value
            }
            min = 1
        }
    }
}

# 人物作用域：计算最终注入新阶的 Merit 点数
tca_merit_carryover_bonus_value = {
    value = tca_merit_carryover_rate_percent_value
    multiply = tca_merit_target_span_value
    divide = 100
}
