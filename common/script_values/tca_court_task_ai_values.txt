# 文武任务AI只读雇主与scope:employee；按需求、称职度、性格和实际付款资源统一评估。
# 无对象先返回-100；付不起返回-1000；正收益不足以负担工作机会成本时映射-100，越过原生停止线。
# 任务状态均由原生员工回调维护；同一员工只持一项差遣，不把其他员工任务费加回。
tca_court_task_ai_six_month_cost = { value = monthly_court_position_task_cost multiply = 6 }
tca_court_task_ai_projected_balance = {
    value = monthly_character_balance
    if = { limit = { has_treasury = yes } value = monthly_character_treasury_balance }
    if = { limit = { scope:employee ?= { has_variable = tca_court_paid_task_active } } add = monthly_court_position_task_cost }
    subtract = monthly_court_position_task_cost
}
# 费用压力：无费用不处罚；不能支付一个月或净亏且不足六个月储备时拒绝，净亏有储备则扣40。
tca_court_task_ai_budget_penalty = {
    value = 0
    if = { limit = { monthly_court_position_task_cost > 0 }
        if = { limit = { OR = {
            AND = { has_treasury = yes OR = { treasury < monthly_court_position_task_cost AND = { tca_court_task_ai_projected_balance < 0 treasury < tca_court_task_ai_six_month_cost } } }
            AND = { has_treasury = no OR = { gold < monthly_court_position_task_cost AND = { tca_court_task_ai_projected_balance < 0 gold < tca_court_task_ai_six_month_cost } } }
        } } value = -1000 }
        else_if = { limit = { tca_court_task_ai_projected_balance < 0 } value = -40 }
    }
}
# 原生aptitude档位0至4；无员工的纯需求测试使用1，真实任务始终有员工。
tca_court_task_ai_aptitude_factor = {
    value = 1
    if = {
        limit = { exists = scope:employee }
        scope:employee = {
            if = { limit = { has_court_position = tca_household_registrar_court_position } value = aptitude:tca_household_registrar_court_position }
            else_if = { limit = { has_court_position = tca_judicial_advisor_court_position } value = aptitude:tca_judicial_advisor_court_position }
            else_if = { limit = { has_court_position = tca_civil_adjutant_court_position } value = aptitude:tca_civil_adjutant_court_position }
            else_if = { limit = { has_court_position = tca_army_judge_court_position } value = aptitude:tca_army_judge_court_position }
            else_if = { limit = { has_court_position = tca_military_supervisor_court_position } value = aptitude:tca_military_supervisor_court_position }
            else_if = { limit = { has_court_position = tca_military_inspector_court_position } value = aptitude:tca_military_inspector_court_position }
            else_if = { limit = { has_court_position = tca_corps_inspector_court_position } value = aptitude:tca_corps_inspector_court_position }
        }
        multiply = 0.15
        add = 0.4
    }
}

tca_military_supervisor_drill_levies_demand = {
    value = 0
    if = { limit = {
        max_military_strength > 0
        any_held_county = { is_landless_type_title = no is_noble_family_title = no building_levies > 0 }
    }
        value = current_military_strength divide = max_military_strength
        save_temporary_value_as = tca_court_task_ai_strength_ratio
        value = tca_court_task_ai_replenishment_ratio_demand
    }
}
# 无副作用数值入口；输入scope:tca_court_task_ai_strength_ratio、可选员工的原生整补任务状态。
# 真实任务仅在上述实际受益前置满足后调用；比率等于60%／85%／95%分别进入下一档。
tca_court_task_ai_replenishment_ratio_demand = {
    value = 0
    if = { limit = { scope:tca_court_task_ai_strength_ratio < 0.6 } value = 130 }
    else_if = { limit = { scope:tca_court_task_ai_strength_ratio < 0.85 } value = 90 }
    else_if = { limit = { scope:employee ?= { has_variable = tca_court_replenishing_levies } scope:tca_court_task_ai_strength_ratio < 0.95 } value = 50 }
}
