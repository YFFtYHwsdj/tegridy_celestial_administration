# 十八项文武差遣任务的AI评分；与任务定义、原生回归共用，不写人物或领地状态。
# 当前作用域与scope:liege均为雇主；原生另提供scope:employee，测试无雇员时称职度倍率默认为1。
# 收益基础分在各任务入口维护，最后统一乘称职度倍率、加费用压力；性格只在有受益对象时加分。
# 无收益基础分-100；按最差称职度乘0.4后仍低于原版停任务阈值-10，不会被免费难度救活。

# 原版月费只读适配；必须提供scope:liege。保留原版头衔、时代与困难AI免费规则。
tca_court_task_ai_six_month_cost = { value = monthly_court_position_task_cost multiply = 6 }

# 费用压力：只查实际付款资源。收入不足两倍单项月费，或当月费也付不起时拒绝。
# 预计执行后的净亏且储备不足六个月同样拒绝；已执行者先加回本项费用，统一开始和保持条件。
# 国库净结余使用monthly_character_treasury_balance，不拿个人金钱补充国库支付能力。
# -1000足以压过所有收益与性格加分；原版费用为0时直接返回0。
tca_court_task_ai_budget_penalty = {
    value = 0
    if = {
        limit = { monthly_court_position_task_cost > 0 }
        if = {
            limit = {
                has_treasury = yes
                OR = {
                    monthly_character_treasury_income < monthly_court_position_task_cost_double
                    treasury < monthly_court_position_task_cost
                    AND = {
                        tca_court_task_ai_projected_balance < 0
                        treasury < tca_court_task_ai_six_month_cost
                    }
                }
            }
            value = -1000
        }
        else_if = {
            limit = {
                has_treasury = no
                OR = {
                    monthly_character_income < monthly_court_position_task_cost_double
                    gold < monthly_court_position_task_cost
                    AND = {
                        tca_court_task_ai_projected_balance < 0
                        gold < tca_court_task_ai_six_month_cost
                    }
                }
            }
            value = -1000
        }
    }
}

# 当前付款资源的预计月净结余；状态只由原生任务回调写在当前员工，读其他人的标记没有效力。
# 活跃任务的月费已含在真实支出中，先加回再扣候选费用；未开任务直接扣，避免启停改变判断基准。
tca_court_task_ai_projected_balance = {
    value = monthly_character_balance
    if = { limit = { has_treasury = yes } value = monthly_character_treasury_balance }
    if = {
        limit = { exists = scope:employee scope:employee = { has_variable = tca_court_paid_task_active } }
        add = monthly_court_position_task_cost
    }
    subtract = monthly_court_position_task_cost
}

# 原生aptitude是0至4的档位；倍率0.4／0.55／0.7／0.85／1，费用不随称职度打折。
# 本体系同人只能持一项差遣，因此只读实际持有职位，避免为每个任务复制倍率。
tca_court_task_ai_aptitude_factor = {
    value = 1
    if = {
        limit = { exists = scope:employee }
        scope:employee = {
            if = { limit = { has_court_position = tca_household_registrar_court_position } value = aptitude:tca_household_registrar_court_position }
            else_if = { limit = { has_court_position = tca_judicial_advisor_court_position } value = aptitude:tca_judicial_advisor_court_position }
            else_if = { limit = { has_court_position = tca_civil_adjutant_court_position } value = aptitude:tca_civil_adjutant_court_position }
            else_if = { limit = { has_court_position = tca_army_judge_court_position } value = aptitude:tca_army_judge_court_position }
            else_if = { limit = { has_court_position = tca_military_supervisor_court_position } value = aptitude:tca_military_supervisor_court_position }
            else_if = { limit = { has_court_position = tca_military_inspector_court_position } value = aptitude:tca_military_inspector_court_position }
            else_if = { limit = { has_court_position = tca_corps_inspector_court_position } value = aptitude:tca_corps_inspector_court_position }
        }
        multiply = 0.15
        add = 0.4
    }
}

    }

}
