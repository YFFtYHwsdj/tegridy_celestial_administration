tca_fw_personal_monthly_income_value = { value = monthly_character_income min = 0 }

# 交涉story作用域；皇帝兼顾其他防守战、国库和性格，不例行叫停每场地方战争。
tca_fw_order_weight_value = {
    value = 30
    if = { limit = { var:tca_fw_sovereign = { treasury < 0 } } add = 40 }
    if = { limit = { var:tca_fw_sovereign = { any_character_war = { is_defender = prev } } } add = 40 }
    if = { limit = { var:tca_fw_sovereign = { has_trait = content } } add = 25 }
    if = { limit = { var:tca_fw_sovereign = { has_trait = ambitious } } subtract = 20 }
    min = 5 max = 100
}
tca_fw_obey_weight_value = {
    value = 50
    add = { value = var:tca_fw_officer.ai_honor multiply = 0.25 }
    if = { limit = { var:tca_fw_officer = { opinion = { target = scope:tca_fw_record.var:tca_fw_sovereign value >= 50 } } } add = 10 }
    else_if = { limit = { var:tca_fw_officer = { opinion = { target = scope:tca_fw_record.var:tca_fw_sovereign value <= -50 } } } subtract = 10 }
    if = { limit = { var:tca_fw_officer = { has_trait = stubborn } } subtract = 30 }
    if = { limit = { var:tca_fw_officer = { has_trait = loyal } } add = 30 }
    min = 5 max = 100
}
