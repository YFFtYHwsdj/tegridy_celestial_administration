# 人物作用域；set_appointment_timeout的days字段只接收命名脚本值，不直接解析var:。
# 共享起复入口先计算并夹住剩余日数，写入原生超时后立即清除临时变量。
tca_retirement_candidate_remaining_timeout_days = {
    value = var:tca_retirement_candidate_remaining_timeout
}

# 借用真实继任者时，仅恢复尚未自然到期的原任命冷却。
tca_retirement_native_probe_remaining_timeout_days = {
    value = var:tca_re_native_original_timeout
    subtract = { value = current_date subtract = var:tca_re_native_borrowed_date }
    min = 0
}

# 真实官署借用夹具归还原任时，恢复原生任命冷却自然扣除借用日数后的余额。
tca_retirement_office_probe_remaining_timeout_days = {
    value = var:tca_ro_original_timeout
    subtract = { value = current_date subtract = var:tca_ro_borrowed_date }
    min = 0
}

# 临时封任原州主时，按实际借用日数恢复其原生冷却。
tca_retirement_supervisor_remaining_timeout_days = {
    value = var:tca_ro_supervisor_prior_timeout
    subtract = { value = current_date subtract = var:tca_ro_supervisor_borrowed_date }
    min = 0
}
