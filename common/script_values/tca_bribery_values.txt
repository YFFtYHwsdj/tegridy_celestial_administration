# 行贿只读数值；人物项在受贿者作用域；互动项要求scope:actor/recipient，均无状态输出。
tca_bribery_merit_price_value = { value = merit_level min = 0 max = 9 multiply = 25 }
tca_bribery_title_price_value = {
    value = 0
    if = { limit = { any_held_title = { tca_bribery_priced_title_trigger = yes tier >= tier_empire } } value = 100 }
    else_if = { limit = { any_held_title = { tca_bribery_priced_title_trigger = yes tier = tier_kingdom } } value = 75 }
    else_if = { limit = { any_held_title = { tca_bribery_priced_title_trigger = yes tier = tier_duchy } } value = 50 }
    else_if = { limit = { any_held_title = { tca_bribery_priced_title_trigger = yes tier = tier_county } } value = 25 }
}
tca_bribery_minister_price_value = {
    value = 0
    if = { limit = { tca_bribery_senior_minister_trigger = yes } value = 50 }
    else_if = { limit = { tca_bribery_six_minister_trigger = yes } value = 25 }
}
tca_bribery_influence_price_value = { value = influence_level min = 0 multiply = 10 }
tca_bribery_base_price_value = {
    value = tca_bribery_merit_price_value
    add = tca_bribery_title_price_value
    add = tca_bribery_minister_price_value
    add = tca_bribery_influence_price_value
}
tca_bribery_monthly_income_value = { value = monthly_character_income min = 0 }
tca_bribery_poor_threshold_value = { value = tca_bribery_monthly_income_value multiply = 3 min = 25 }
tca_bribery_rich_threshold_value = { value = tca_bribery_monthly_income_value multiply = 24 min = 300 }
# 0负债、1拮据、2正常、3富有；负债先判，四项互斥。
tca_bribery_finance_tier_value = {
    value = 2
    if = { limit = { gold < 0 } value = 0 }
    else_if = { limit = { gold < tca_bribery_poor_threshold_value } value = 1 }
    else_if = { limit = { gold >= tca_bribery_rich_threshold_value } value = 3 }
}
tca_bribery_finance_score_value = {
    value = 0
    if = { limit = { tca_bribery_finance_tier_value = 0 } value = 30 }
    else_if = { limit = { tca_bribery_finance_tier_value = 1 } value = 20 }
    else_if = { limit = { tca_bribery_finance_tier_value = 3 } value = -10 }
}
tca_bribery_greed_score_value = {
    value = ai_greed
    if = { limit = { ai_greed >= 0 } multiply = 0.5 }
    else = { multiply = 0.3 }
    min = -15 max = 25
}
tca_bribery_honor_score_value = {
    value = ai_honor
    if = { limit = { ai_honor >= 0 } multiply = -0.8 }
    else = { multiply = -0.4 }
    min = -40 max = 20
}
# 受贿者作用域，actor为付款者；高档关系有原生别名，必须先判避免重复。
tca_bribery_opinion_score_value = { value = "opinion(scope:actor)" multiply = 0.25 min = -25 max = 25 }
tca_bribery_relation_score_value = {
    value = 0
    if = { limit = { has_relation_best_friend = scope:actor } value = 20 }
    else_if = { limit = { has_relation_friend = scope:actor } value = 10 }
    else_if = { limit = { has_relation_nemesis = scope:actor } value = -50 }
    else_if = { limit = { has_relation_rival = scope:actor } value = -25 }
}
tca_bribery_acceptance_base_value = {
    value = -40
    add = tca_bribery_greed_score_value
    add = tca_bribery_honor_score_value
    add = tca_bribery_opinion_score_value
    add = tca_bribery_relation_score_value
    add = tca_bribery_finance_score_value
}
# 原生互斥选项；玩家必须显式选中，避免取消所有选项后免费或默认成交。
tca_bribery_selected_tier_value = {
    value = 0
    if = { limit = { scope:tca_bribery_thin ?= yes } value = 1 }
    else_if = { limit = { scope:tca_bribery_generous ?= yes } value = 2 }
    else_if = { limit = { scope:tca_bribery_lavish ?= yes } value = 3 }
    # 原生AI可能尚未建立GUI选项；此时按同一最低可接受档评估并在发送时固化。
    else_if = { limit = { scope:actor = { is_ai = yes } } value = tca_bribery_cheapest_tier_value }
}
tca_bribery_offer_score_value = {
    value = 0
    if = { limit = { tca_bribery_selected_tier_value = 2 } value = 25 }
    else_if = { limit = { tca_bribery_selected_tier_value = 3 } value = 50 }
}
tca_bribery_price_value = {
    value = scope:recipient.tca_bribery_base_price_value
    if = { limit = { tca_bribery_selected_tier_value = 2 } multiply = 1.5 }
    else_if = { limit = { tca_bribery_selected_tier_value = 3 } multiply = 2 }
    ceiling = yes
}
tca_bribery_cheapest_tier_value = {
    value = 0
    if = { limit = { scope:recipient.tca_bribery_acceptance_base_value >= 0 } value = 1 }
    else_if = { limit = { scope:recipient.tca_bribery_acceptance_base_value >= -25 } value = 2 }
    else_if = { limit = { scope:recipient.tca_bribery_acceptance_base_value >= -50 } value = 3 }
}
# 互动候选上下文需要actor/recipient；独立于尚未建立的发送选项flags计算预算。
tca_bribery_ai_price_value = {
    value = scope:recipient.tca_bribery_base_price_value
    if = { limit = { tca_bribery_cheapest_tier_value = 2 } multiply = 1.5 }
    else_if = { limit = { tca_bribery_cheapest_tier_value = 3 } multiply = 2 }
    ceiling = yes
}
# 付款者作用域；数值为设计者补齐的初版调参，资格和预算由ai_ready单独控制。
tca_bribery_ai_interest_value = {
    value = 10
    if = { limit = { has_trait = ambitious } add = 15 }
    add = { value = ai_honor multiply = -0.1 min = -20 max = 20 }
    subtract = { value = ai_greed min = 0 multiply = 0.05 max = 10 }
    min = 0 max = 50
}
# 互动上下文；目标层次与上级偏好影响原生随机发送兴趣，不建立公务请求链。
tca_bribery_ai_target_weight_value = {
    value = 1
    add = { value = scope:recipient.merit_level min = 0 max = 9 }
    add = { value = scope:recipient.tca_bribery_title_price_value divide = 25 }
    add = { value = scope:recipient.tca_bribery_minister_price_value divide = 25 }
    if = { limit = { scope:actor = { is_vassal_or_below_of = scope:recipient } } add = 10 }
}
tca_bribery_ai_will_do_value = {
    value = tca_bribery_ai_interest_value
    multiply = tca_bribery_ai_target_weight_value
    divide = 10
    max = 100
}
