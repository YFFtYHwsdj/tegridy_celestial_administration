# 共用分值；仅在铨选链读取目标倾向、原生档位与品级，不复制称职度原始公式。
tca_court_civil_supplement = {
    value = 0
    add = { value = stewardship multiply = 0.2 max = 5 }
    add = { value = learning multiply = 0.1 max = 3 }
    if = { limit = { has_trait = diligent } add = 2 }
    if = { limit = { has_trait = just } add = 2 }
    if = { limit = { has_trait = lazy } add = -2 }
    if = { limit = { has_trait = arbitrary } add = -2 }
}
tca_court_personal_selection_value = {
    value = merit_level multiply = 10
    # 永久进身经历仅影响铨选竞争；文武共用且同次评分不叠加。
    add = tca_origin_selection_penalty_value
    if = { limit = { scope:tca_court_bias = 1 }
        add = tca_court_civil_supplement
        if = { limit = { tca_court_military_route_trigger = yes } add = -10 }
    }
    else = {
        add = tca_court_military_supplement
        if = { limit = { tca_court_civil_route_trigger = yes } add = -10 }
    }
}
tca_court_military_supplement = {
    value = 0
    add = { value = martial multiply = 0.2 max = 5 }
    add = { value = prowess multiply = 0.1 max = 3 }
    if = { limit = { has_trait_with_flag = commander_trait_flag } add = 2 }
    if = { limit = { has_trait = diligent } add = 2 }
    if = { limit = { has_trait = brave } add = 2 }
    if = { limit = { has_trait = lazy } add = -2 }
    if = { limit = { has_trait = craven } add = -2 }
}

    value = monthly_character_treasury_income min = 0 divide = 2 max = 40
    if = { limit = { exists = capital_county } add = { value = capital_county.development_level divide = 2 max = 40 } }
}
