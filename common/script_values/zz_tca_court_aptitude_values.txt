# 原版无全体职位称职度追加接口；采用1.21主体并在根层增加本署署务加成。
general_aptitude_bonus = {
	add = general_aptitude_bonus_exclude_other_court_position
	# 由原生排除当前职位，避免对自身递归加成。
	if = {
		limit = { exists = liege }
		liege = {
			# 总管署务加成。
			if = {
				limit = {
					court_position:seneschal_court_position ?= {
						has_variable = performing_improve_court
					}
				}
				add = {
					value = court_position:seneschal_court_position.seneschal_improve_court_value
					desc = seneschal_improve_court_aptitude_desc
				}
			}
			# yeke_jarquchi职位的署务加成。
			if = {
				limit = {
					court_position:yeke_jarquchi_court_position ?= {
						has_variable = performing_improve_court
					}
				}
				add = {
					value = court_position:yeke_jarquchi_court_position.yeke_jarquchi_improve_court_value
					desc = yeke_jarquchi_improve_court_aptitude_desc
				}
			}
			if = {
				limit = {
					court_position:khlon_glan_court_position ?= {
						has_variable = performing_improve_court
					}
				}
				add = {
					value = court_position:khlon_glan_court_position.khlon_glan_improve_court_value
					desc = khlon_glan_improve_court_aptitude_desc
				}
			}
		}
	}
	# 家族团结提供的原生称职度加成。
	if = {
		limit = { court_position_liege_has_consulted_house_bonus = yes }
		add = {
			value = court_position_consulted_house_bonus_aptitude_value
			desc = unity_consulted_house_member_desc
		}
	}
    # 本署加成在候选人物作用域计算，独立于原版领主分支。
    add = { value = tca_court_organize_bonus desc = tca_court_organize_office_bonus }
}
