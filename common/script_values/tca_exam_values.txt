# 会试／殿试自动结算使用的数值。
# 分数逐项复刻原版 imperial_examination_entry_value，只把“活动侧重”替换为
# 知贡举在开科时自动选择的 tca_exam_focus：0 学识、1 军事、2 管理。
# 未设置侧重时按学识处理，保证测试和旧存档不会得到空分数。
tca_exam_entry_value = {
	value = 0
	if = {
		limit = { is_alive = yes }
		# 功绩
		if = {
			limit = { merit_level >= 7 }
			add = 20
		}
		else_if = {
			limit = { merit_level >= 5 }
			add = 15
		}
		else_if = {
			limit = { merit_level >= 3 }
			add = 9
		}
		else_if = {
			limit = { merit_level >= 1 }
			add = 6
		}
		# 儒学教育特质
		if = {
			limit = { has_trait = confucian_education }
			if = {
				limit = {
					has_trait_xp = {
						trait = confucian_education
						# 达到最高档：value >= 100
						value >= trait_max_value
					}
				}
				add = 20
			}
			else_if = {
				limit = {
					has_trait_xp = {
						trait = confucian_education
						# 达到第四档：value >= 80
						value >= trait_fourth_of_five_level_value
					}
				}
				add = 15
			}
			else_if = {
				limit = {
					has_trait_xp = {
						trait = confucian_education
						# 达到第三档：value >= 60
						value >= trait_third_of_five_level_value
					}
				}
				add = 12
			}
			else_if = {
				limit = {
					has_trait_xp = {
						trait = confucian_education
						# 达到第二档：value >= 40
						value >= trait_second_of_five_level_value
					}
				}
				add = 9
			}
			else_if = {
				limit = {
					has_trait_xp = {
						trait = confucian_education
						# 达到第一档：value >= 20
						value >= trait_first_of_five_level_value
					}
				}
				add = 6
			}
			else = { add = 3 } # 未达到第一档
		}
		
		if = {
			limit = { always = yes }
			# 学识侧重
			if = {
				limit = {
					OR = {
						scope:host ?= { NOT = { has_variable = tca_exam_focus } }
						scope:host ?= { var:tca_exam_focus = 0 }
					}
				}
				add = learning
				if = {
					limit = { has_trait = education_learning_1 }
					add = 3
				}
				if = {
					limit = { has_trait = education_learning_2 }
					add = 6
				}
				if = {
					limit = { has_trait = education_learning_3 }
					add = 9
				}
				if = {
					limit = { has_trait = education_learning_4 }
					add = 12
				}
				if = {
					limit = { has_trait = education_learning_5 }
					add = 15
				}
				if = {
					limit = { has_trait = shrewd }
					add = 3
				}
				if = {
					limit = { has_trait = lifestyle_mystic }
					add = 3
				}
				if = {
					limit = { 
						has_trait_xp = {
							trait = lifestyle_mystic 
							value >= trait_second_level
						}
					}
					add = 3
				}
				if = {
					limit = { 
						has_trait_xp = {
							trait = lifestyle_mystic 
							value >= trait_third_level
						}
					}
					add = 3
				}
				if = {
					limit = { has_trait = lifestyle_physician }
					add = 3
				}
				if = {
					limit = { 
						has_trait_xp = {
							trait = lifestyle_physician 
							value >= trait_second_level
						}
					}
					add = 3
				}
				if = {
					limit = { 
						has_trait_xp = {
							trait = lifestyle_physician 
							value >= trait_third_level
						}
					}
					add = 3
				}
				if = {
					limit = { has_trait = whole_of_body }
					add = 3
				}
				if = {
					limit = { has_trait = theologian }
					add = 3
				}
				if = {
					limit = { has_trait = erudite }
					add = 3
				}
				if = {
					limit = { has_trait = lifestyle_poet }
					add = 3
				}
				if = {
					limit = { has_trait = journaller }
					add = 3
				}
			}
			# 军事侧重
			if = {
				limit = {
					scope:host ?= { var:tca_exam_focus = 1 }
				}
				add = martial
				if = {
					limit = { has_trait = education_martial_1 }
					add = 3
				}
				if = {
					limit = { has_trait = education_martial_2 }
					add = 6
				}
				if = {
					limit = { has_trait = education_martial_3 }
					add = 9
				}
				if = {
					limit = { has_trait = education_martial_4 }
					add = 12
				}
				if = {
					limit = { has_trait = education_martial_5 }
					add = 15
				}
				if = {
					limit = { has_trait = strategist }
					add = 3
				}
				if = {
					limit = { has_trait = overseer }
					add = 3
				}
				if = {
					limit = { has_trait = gallant }
					add = 3
				}
				if = {
					limit = { has_trait = logistician }
					add = 3
				}
				if = {
					limit = { has_trait = military_engineer }
					add = 3
				}
				if = {
					limit = { has_trait = unyielding_defender }
					add = 3
				}
				if = {
					limit = { has_trait = forder }
					add = 3
				}
				if = {
					limit = { has_trait = flexible_leader }
					add = 3
				}
				if = {
					limit = { has_trait = desert_warrior }
					add = 3
				}
				if = {
					limit = { has_trait = reaver }
					add = 3
				}
				if = {
					limit = { has_trait = reckless }
					add = 3
				}
				if = {
					limit = { has_trait = holy_warrior }
					add = 3
				}
				if = {
					limit = { has_trait = open_terrain_expert }
					add = 3
				}
				if = {
					limit = { has_trait = rough_terrain_expert }
					add = 3
				}
				if = {
					limit = { has_trait = forest_fighter }
					add = 3
				}
				if = {
					limit = { has_trait = cautious_leader }
					add = 3
				}
				if = {
					limit = { has_trait = organizer }
					add = 3
				}
				if = {
					limit = { has_trait = winter_soldier }
					add = 3
				}
			}
			# 管理侧重
			if = {
				limit = {
					scope:host ?= { var:tca_exam_focus = 2 }
				}
				add = stewardship
				if = {
					limit = { has_trait = education_stewardship_1 }
					add = 3
				}
				if = {
					limit = { has_trait = education_stewardship_2 }
					add = 6
				}
				if = {
					limit = { has_trait = education_stewardship_3 }
					add = 9
				}
				if = {
					limit = { has_trait = education_stewardship_4 }
					add = 12
				}
				if = {
					limit = { has_trait = education_stewardship_5 }
					add = 15
				}
				if = {
					limit = { has_trait = architect }
					add = 3
				}
				if = {
					limit = { has_trait = administrator }
					add = 3
				}
				if = {
					limit = { has_trait = avaricious }
					add = 3
				}
				if = {
					limit = { has_trait = lifestyle_gardener }
					add = 3
				}
				if = {
					limit = { has_trait = governor }
					add = 3
				}
			}
		}
		
		# 家族抱负加成
		if = {
			limit = {
				exists = house
			}
			house = {
				if = {
					limit = { has_house_aspiration_parameter = examination_success_bonus }
					add = 10
				}
			}
		}
		
		# 住宅建筑加成
		if = {
			limit = { exists = house.house_head.domicile }
			house.house_head.domicile = {
				# 藏书楼倍率
				if = {
					limit = { has_domicile_parameter = estate_examination_start_score_bonus_4 }
					multiply = {
						value = 1
						add = estate_examination_start_score_bonus_4_value
					}
				}
				else_if = {
					limit = { has_domicile_parameter = estate_examination_start_score_bonus_3 }
					multiply = {
						value = 1
						add = estate_examination_start_score_bonus_3_value
					}
				}
				else_if = {
					limit = { has_domicile_parameter = estate_examination_start_score_bonus_2 }
					multiply = {
						value = 1
						add = estate_examination_start_score_bonus_2_value
					}
				}
				else_if = {
					limit = { has_domicile_parameter = estate_examination_start_score_bonus_1 }
					multiply = {
						value = 1
						add = estate_examination_start_score_bonus_1_value
					}
				}
			}
		}

		# 未过乡试的分数惩罚。
		if = {
			limit = {
				NOT = { has_character_flag = passed_provincial_exam }
			}
			divide = 3
		}

		# 不通晓主考文化语言的分数惩罚。
		if = {
			limit = {
				scope:host ?= { exists = culture }
				NOT = { knows_language_of_culture = scope:host.culture }
			}
			divide = 5
		}
	}
	min = 0
	ceiling = yes
}

# 玩家为皇帝时，按所选规模与当前侧重计算费用。
# 基础部分逐项复刻原版活动 cost；考试院与开封宫殿折扣使用皇帝首都。
tca_exam_cost_base_value = {
	value = 0
	add = {
		value = imperial_examination_actual_cost
		if = {
			limit = {
				scope:host ?= { capital_province = { has_building_or_higher = kaifeng_palace_01 } }
			}
			multiply = {
				value = {
					value = 1
					subtract = kaifeng_examination_cost_value
				}
			}
		}
	}
	if = {
		limit = {
			scope:host ?= { capital_province = { has_building_or_higher = examination_hall_01 } }
		}
		add = {
			value = 0
			add = {
				value = imperial_examination_base_cost
				add = {
					value = imperial_examination_base_cost
					multiply = activity_cost_scale_by_era
					subtract = imperial_examination_base_cost
				}
			}
			multiply = imperial_examination_cost_discount_max_value
		}
	}
	if = {
		limit = { factor_zero_if_entitled_to_freebie_activity_trigger = yes }
		multiply = 0
	}
	min = 0
}

tca_exam_cost_exclusive_value = {
	value = tca_exam_cost_base_value
	add = imperial_examination_breadth_exclusive_value

		limit = {

			scope:host ?= { var:tca_exam_focus > 0 }
		}
		add = imperial_examination_alternate_focus_value
	}
	min = 0
}

# 实际扣费按 tca_exam_breadth：0 小比、1 常科、2 大开。
tca_exam_cost_value = {
	value = tca_exam_cost_restricted_value
	if = {
		limit = {
			scope:host ?= { has_variable = tca_exam_breadth }
			scope:host ?= { var:tca_exam_breadth = 0 }
		}
		value = tca_exam_cost_exclusive_value
	}
	else_if = {
		limit = {
			scope:host ?= { has_variable = tca_exam_breadth }
			scope:host ?= { var:tca_exam_breadth = 2 }
		}
		value = tca_exam_cost_open_value
	}
}

	value = tca_exam_cost_value

		limit = {
			has_variable = tca_exam_breadth
			var:tca_exam_breadth = 0
			has_variable = tca_exam_cost_small
		}
		value = var:tca_exam_cost_small
	}
	else_if = {
		limit = {
			has_variable = tca_exam_breadth
			var:tca_exam_breadth = 1
			has_variable = tca_exam_cost_standard
		}
		value = var:tca_exam_cost_standard
	}
	else_if = {
		limit = {
			has_variable = tca_exam_breadth
			var:tca_exam_breadth = 2
			has_variable = tca_exam_cost_grand
		}
		value = var:tca_exam_cost_grand
	}

# 原版 imperial_examination_passing_grade 的 AI 权重，按知贡举自身取值。
# 原版权重是加权随机，不是取最高分。
tca_exam_grade_weight_low = {
	value = 100
	add = { value = ai_sociability multiply = 0.25 }
	if = { limit = { has_trait = lazy } add = 20 }
	if = { limit = { has_trait = impatient } add = 20 }
	if = {
		limit = {
			OR = {
				has_trait = education_diplomacy_1
				has_trait = education_martial_1
				has_trait = education_stewardship_1
				has_trait = education_intrigue_1
				has_trait = education_learning_1
			}
		}
		add = 50
	}
}

tca_exam_grade_weight_mediocre = {
	value = 100
	if = { limit = { has_trait = content } add = 20 }
	if = { limit = { has_trait = humble } add = 10 }
	if = { limit = { has_trait = craven } add = 10 }
	if = { limit = { has_trait = fickle } add = 10 }
	if = {
		limit = {
			OR = {
				has_trait = education_diplomacy_2
				has_trait = education_martial_2
				has_trait = education_stewardship_2
				has_trait = education_intrigue_2
				has_trait = education_learning_2
			}
		}
		add = 50
	}
}
