# 下列所有数值均在目标角色作用域求值。
# 相对政权数值使用目标的 top_liege；宫廷宾客使用实际 host 的 top_liege。
# 共享身份档位；原始评分比较只集中在此分类器中。
tca_judicial_identity_tier_2_threshold = 30
tca_judicial_identity_tier_3_threshold = 50
tca_judicial_identity_tier_4_threshold = 80
tca_judicial_identity_tier_5_threshold = 100
# 兼容调用的阈值别名：提刑权限使用第二档门槛，中央权限使用第四档门槛。
tca_judicial_tixing_threshold = { value = tca_judicial_identity_tier_2_threshold }
tca_judicial_central_threshold = { value = tca_judicial_identity_tier_4_threshold }
# 目标角色作用域；由高至低选唯一身份档，供审批、详覆费用和本地化共用。
tca_judicial_identity_tier = {
    value = 1
    if = { limit = { tca_judicial_authority_score >= tca_judicial_identity_tier_5_threshold } value = 5 }
    else_if = { limit = { tca_judicial_authority_score >= tca_judicial_identity_tier_4_threshold } value = 4 }
    else_if = { limit = { tca_judicial_authority_score >= tca_judicial_identity_tier_3_threshold } value = 3 }
    else_if = { limit = { tca_judicial_authority_score >= tca_judicial_identity_tier_2_threshold } value = 2 }
}

# 官职分只取最高命中类别；下方权重依次对应中央、路、州、县，不叠加兼任官职。
tca_judicial_office_score = {
    value = 0
    if = { limit = { tca_has_central_office_trigger = yes } add = tca_judicial_office_score_1_weight }
    else_if = { limit = { tca_has_regional_office_trigger = yes } add = tca_judicial_office_score_2_weight }
    else_if = { limit = { any_held_title = { tca_is_local_administrative_title_trigger = yes tier = tier_duchy } } add = tca_judicial_office_score_3_weight }
    else_if = { limit = { any_held_title = { tca_is_local_administrative_title_trigger = yes tier = tier_county } } add = tca_judicial_office_score_4_weight }
}
tca_judicial_office_score_1_weight = 80
tca_judicial_office_score_2_weight = 80
tca_judicial_office_score_3_weight = 35
tca_judicial_office_score_4_weight = 30

# 家族身份只取最高类别：显赫、强大家族、有世族家门头衔、一般非出身低微。
tca_judicial_family_score = {
    value = 0
    if = { limit = { house ?= { is_dominant_family = yes } } add = tca_judicial_family_score_1_weight }
    else_if = { limit = { house ?= { is_powerful_family = yes } } add = tca_judicial_family_score_2_weight }
    else_if = { limit = { tca_has_noble_family_trigger = yes } add = tca_judicial_family_score_3_weight }
    else_if = { limit = { is_lowborn = no } add = tca_judicial_family_score_4_weight }
}
tca_judicial_family_score_1_weight = 35
tca_judicial_family_score_2_weight = 25
tca_judicial_family_score_3_weight = 15
tca_judicial_family_score_4_weight = 5

# 宗族威望按最高达到档取一次加分，下方权重与分支自上而下对应。

# 按目标所属政权君主计算皇亲分：一等亲、同家族、同宗族互斥；宾客使用实际 host 的政权。
tca_judicial_royal_score = {
    value = 0
    if = { limit = { tca_is_emperor_first_degree_kin_trigger = yes } add = tca_judicial_royal_score_1_weight }
    else_if = { limit = { exists = house OR = {
        AND = { tca_is_hosted_court_guest_trigger = yes house = host.top_liege.house }
        AND = { tca_is_hosted_court_guest_trigger = no house = top_liege.house }
    } } add = tca_judicial_royal_score_2_weight }
    else_if = { limit = { exists = dynasty OR = {
        AND = { tca_is_hosted_court_guest_trigger = yes dynasty = host.top_liege.dynasty }
        AND = { tca_is_hosted_court_guest_trigger = no dynasty = top_liege.dynasty }
    } } add = tca_judicial_royal_score_3_weight }
}

# 发起者作用域的原生 AI 输入；自定义 redirect 作用域不可用时也可安全求值。
# 发起角色作用域，需互动 recipient；提刑按目标实际领主／路主官／最高领主解析军事支持。
# 供原生接受度计算，不改变实际发起者和最终收押者。
tca_judicial_backing_strength = {
    value = current_military_strength
    if = {
        limit = { tca_is_judicial_commissioner_trigger = yes }
        value = primary_title.var:song_judicial_circuit.holder.current_military_strength
        if = {
            limit = { scope:recipient = { is_ruler = yes exists = liege } }
            value = scope:recipient.liege.current_military_strength
        }
        else_if = {
            limit = { NOT = { primary_title.var:song_judicial_circuit.holder = { tca_basic_allowed_to_arrest_character_trigger = { CHARACTER = scope:recipient } } } }
            value = top_liege.current_military_strength
        }
    }
}
