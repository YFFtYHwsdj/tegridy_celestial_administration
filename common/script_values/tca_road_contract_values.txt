# 人物作用域；每个降格理由增加一次改约尝试的成功机会。
tca_road_contract_reversion_chance = {
    value = 0
    if = { limit = { ai_has_warlike_personality = no } add = 30 }
    if = { limit = { ai_has_cautious_personality = yes } add = 30 }
    if = { limit = { tca_road_contract_full_regiments_trigger = no } add = 30 }
}

# 人物作用域；资格具备后，开疆拓土提高升格意愿。
tca_road_contract_promotion_chance = {
    value = 60
    if = {
        limit = {
            situation:dynastic_cycle ?= { has_situation_top_phase_parameter = dynastic_cycle_hegemon_increased_protectorate_limit }
        }
        add = 20
    }
}

# 军团作用域；按大类基础招募价折价。新兵种类型采用最低常设兵种基价。
tca_road_contract_regiment_refund = {
    if = { limit = { is_unit_type = archers } value = bowmen_recruitment_cost }
    else_if = { limit = { is_unit_type = pikemen } value = pikemen_recruitment_cost }
    else_if = { limit = { is_unit_type = heavy_infantry } value = heavy_infantry_recruitment_cost }
    else_if = { limit = { is_unit_type = light_cavalry } value = light_cavalry_recruitment_cost }
    else_if = { limit = { is_unit_type = heavy_cavalry } value = heavy_cavalry_recruitment_cost }
    else_if = { limit = { is_unit_type = archer_cavalry } value = horse_archers_recruitment_cost }
    else_if = { limit = { is_unit_type = siege_weapon } value = onager_recruitment_cost }
    else_if = { limit = { is_unit_type = gunpowder } value = handgunner_recruitment_cost }
    else_if = { limit = { is_unit_type = camel_cavalry } value = camel_rider_recruitment_cost }
    else_if = { limit = { is_unit_type = elephant_cavalry } value = war_elephant_recruitment_cost }
    else = { value = skirmisher_recruitment_cost }
    multiply = maa_size
    multiply = 0.25
}

# 头衔作用域；只计本路实际拥有的军团，外调兵士仍归原所属头衔。
tca_road_contract_disband_refund = {
    value = 0
    every_owned_title_maa_regiment = { add = tca_road_contract_regiment_refund }
}
