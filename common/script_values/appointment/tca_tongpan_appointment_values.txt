# 临时作用域仅在本次数值求值内有效，供距离函数从省份上下文定位官职持有人。
tca_tongpan_appointment_value = {
    value = 0
    save_temporary_scope_as = tca_tongpan_valuation_office
    if = {
        limit = { exists = var:tca_tongpan_county.title_province }
        add = {
            value = var:tca_tongpan_county.title_province.monthly_income
            multiply = tca_tongpan_appointment_income_factor_value

            limit = { exists = holder.top_liege.capital_province }
            add = {
                value = 500
                subtract = {
                    value = "var:tca_tongpan_county.title_province.squared_distance(scope:tca_tongpan_valuation_office.holder.top_liege.capital_province)"
                    multiply = 0.0001
                }
                min = 0
                desc = appointment_title_value.distance_to_capital
            }
        }
    }
}
