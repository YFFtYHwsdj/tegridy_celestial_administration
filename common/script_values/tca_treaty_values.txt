# 提案作用域；基础接受意愿另由书信提供。给予的岁币更易接受，索取额按本人财力降低。
# 这是首版性格/支付方向判断，不带额外失信惩罚；固定年额绝不随收入自动变化。
tca_treaty_acceptance_value = {
    value = 0
    if = { limit = { var:tca_treaty_direction = 1 var:tca_treaty_amount > 0 } add = 50 }
    if = { limit = { var:tca_treaty_direction = 2 var:tca_treaty_amount > 0 }
        subtract = { value = var:tca_treaty_amount divide = 25 max = 100 }
    }
    if = { limit = { var:tca_treaty_to = { has_trait = content } } add = 20 }
    if = { limit = { var:tca_treaty_to = { has_trait = ambitious } } subtract = 20 }
    if = { limit = { var:tca_treaty_mode = 2 } subtract = 25 }
    min = -45 max = 150
}

# 日期保存在数值变量中；current_date触发器仅接受日期字面量，比较须经数值script_value。
tca_treaty_proposal_age_value = { value = current_date subtract = var:tca_treaty_created }
tca_treaty_grace_elapsed_value = { value = current_date subtract = var:tca_treaty_grace }
tca_treaty_due_elapsed_value = { value = current_date subtract = var:tca_treaty_due }
