# 初始平衡：官职基础值＋原生路经济价值的三分之一＋距离。
tca_judicial_appointment_base_value = 250
tca_judicial_appointment_income_factor_value = 8 # 原生 24 / 3。

# 头衔作用域：提刑官职，不依赖持有人的契约分类。
tca_judicial_appointment_value = {
    value = 0
    add = {
        value = tca_judicial_appointment_base_value
        desc = tca_judicial_prominence_base
    }
    if = {
        limit = { exists = var:song_judicial_circuit }
        add = {
            var:song_judicial_circuit = {
                every_de_jure_county = {
                    add = {
                        value = title_province.monthly_income
                        multiply = tca_judicial_appointment_income_factor_value
                    }
                }
            }
            desc = tca_judicial_prominence_circuit_income
        }
        if = {
            limit = {
                exists = holder.top_liege.capital_province
                exists = var:song_judicial_circuit.title_province
            }
            add = {
                value = 500
                subtract = {
                    var:song_judicial_circuit = {
                        add = {
                            value = "title_province.squared_distance(root.holder.top_liege.capital_province)"
                            multiply = 0.0001
                        }
                    }
                }
                min = 0
                desc = appointment_title_value.distance_to_capital
            }
        }
    }
}
