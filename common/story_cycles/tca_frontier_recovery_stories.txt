tca_frontier_recovery_loss = {
    on_owner_death = {
        if = { limit = { var:tca_fd_parent.var:tca_fs_attacker ?= { is_alive = yes } } make_story_owner = var:tca_fd_parent.var:tca_fs_attacker }
        else_if = { limit = { var:tca_fd_parent.var:tca_fd_attacker_country.holder ?= { is_alive = yes } }
            make_story_owner = var:tca_fd_parent.var:tca_fd_attacker_country.holder
        }
        else_if = { limit = { var:tca_fd_parent.var:tca_fd_country.holder ?= { is_alive = yes } }
            make_story_owner = var:tca_fd_parent.var:tca_fd_country.holder
        }
        else = { end_story = yes }
    }
}
