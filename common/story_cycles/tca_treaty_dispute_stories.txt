# 一份和约/一场战争/一名主动参战者一行；已服从行保留至战争结束防止同战重复追责。
tca_treaty_dispute = {
    on_owner_death = { end_story = yes }
    on_end = {
        save_scope_as = tca_treaty_ending_dispute
        var:tca_treaty_war_leader ?= { every_character_war = {
            limit = { this = scope:tca_treaty_ending_dispute.var:tca_treaty_war }
            remove_list_variable = { name = tca_treaty_disputes target = scope:tca_treaty_ending_dispute }
        } }
    }
    effect_group = { days = 1 triggered_effect = {
        trigger = { always = yes }
        effect = { tca_treaty_dispute_maintain_effect = yes }
    } }
}
