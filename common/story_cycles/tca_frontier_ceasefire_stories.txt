# 一场战争一份交涉；原生书信跨日与保存重载共用此记录，不用人物单一war指针。
tca_frontier_ceasefire = {
    on_owner_death = { end_story = yes }
    on_end = {
        save_scope_as = tca_fw_ending
        var:tca_fw_officer ?= { if = { limit = { is_alive = yes }
            remove_list_variable = { name = tca_fw_records target = scope:tca_fw_ending }
            every_character_war = {
                limit = { var:tca_fw_record ?= scope:tca_fw_ending }
                remove_variable = tca_fw_record
            }
        } }
    }
    effect_group = { days = 1 triggered_effect = {
        trigger = { OR = {
            NOT = { tca_fw_parties_valid_trigger = yes }
            AND = { var:tca_fw_state < 4 NOT = { tca_fw_live_trigger = yes } }
        } }
        effect = { end_story = yes }
    } }
}
