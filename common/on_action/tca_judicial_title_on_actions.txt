# Additive hooks; do not replace vanilla effect blocks.
on_game_start_after_lobby = { on_actions = { tca_judicial_setup } }
yearly_global_pulse = { on_actions = { tca_judicial_setup } }
on_title_gain = { on_actions = { tca_judicial_title_gain tca_judicial_position_gain } }
on_government_change = { on_actions = { tca_judicial_government_change tca_judicial_position_government_change } }
on_title_destroyed = { on_actions = { tca_judicial_title_destroyed } }

tca_judicial_setup = {
    effect = { tca_refresh_all_judicial_titles_effect = yes }
}

tca_judicial_title_gain = {
    trigger = { scope:title = { tier = tier_kingdom } }
    events = { tca_judicial_titles.1 }
}

tca_judicial_government_change = {
    trigger = { highest_held_title_tier >= tier_kingdom }
    events = { tca_judicial_titles.1 }
}

tca_judicial_title_destroyed = {
    effect = {
        save_scope_as = tca_lifecycle_actor
        scope:landed_title = {
            if = {
                limit = { has_variable = tca_judicial_title }
                tca_retire_judicial_title_effect = yes
            }
            else_if = {
                limit = { has_variable = song_judicial_title }
                tca_destroy_judicial_title_effect = yes
            }
        }
    }
}

# Fires on initial grant and later inheritance of an actual judicial title.
tca_judicial_position_gain = {
    trigger = {
        OR = {
            scope:title = { has_variable = tca_judicial_title }
            any_held_title = { has_variable = tca_judicial_title }
        }
    }
    effect = { trigger_event = { id = tca_judicial_titles.3 days = 1 } }
}

# No invented monthly global pulse: use the native quarterly character hook.
quarterly_playable_pulse = { on_actions = { tca_judicial_quarterly_maintenance } }

tca_judicial_quarterly_maintenance = {
    trigger = {
        OR = {
            any_held_title = { has_variable = tca_judicial_title }
            any_held_title = { tca_has_judicial_title_trigger = yes }
        }
    }
    effect = {
        tca_refresh_character_circuits_effect = yes
        tca_refresh_held_judicial_positions_effect = yes
    }
}

tca_judicial_position_government_change = {
    trigger = { any_held_title = { has_variable = tca_judicial_title } }
    effect = { trigger_event = { id = tca_judicial_titles.3 days = 1 } }
}

# A commissioner's new county/office or changed liege must also be checked.
on_vassal_change = { on_actions = { tca_judicial_position_government_change } }
