# Additive hooks; do not replace vanilla effect blocks.
on_game_start_after_lobby = { on_actions = { tca_judicial_setup } }
yearly_global_pulse = { on_actions = { tca_judicial_setup } }
on_title_gain = { on_actions = { tca_judicial_title_gain tca_judicial_position_gain } }
on_government_change = { on_actions = { tca_judicial_government_change tca_judicial_position_government_change } }
on_title_destroyed = { on_actions = { tca_judicial_title_destroyed } }

tca_judicial_setup = {
    effect = { tca_refresh_all_judicial_titles_effect = yes }
}

tca_judicial_title_gain = {
    trigger = { government_has_flag = government_is_celestial }
    effect = {
        tca_request_judicial_maintenance_effect = yes
        if = { limit = { OR = { has_council_position = minister_justice scope:title = title:e_minister_of_justice } } top_liege = { tca_request_judicial_maintenance_effect = yes } }
    }
}

tca_judicial_government_change = {
    trigger = { government_has_flag = government_is_celestial }
    effect = { tca_request_judicial_maintenance_effect = yes }
}

tca_judicial_title_destroyed = {
    effect = {
        save_scope_as = tca_lifecycle_actor
        scope:landed_title = {
            if = {
                limit = { has_variable = tca_judicial_title }
                tca_retire_judicial_title_effect = yes
            }
            else_if = {
                limit = { has_variable = song_judicial_title }
                tca_destroy_judicial_title_effect = yes
            }
        }
    }
}

# Fires on initial grant and later inheritance of an actual judicial title.
tca_judicial_position_gain = {
    trigger = {
        OR = {
            scope:title = { has_variable = tca_judicial_title }
            any_held_title = { has_variable = tca_judicial_title }
        }
    }
    effect = { tca_request_judicial_maintenance_effect = yes }
}

# No invented monthly global pulse: use the native quarterly character hook.
quarterly_playable_pulse = { on_actions = { tca_judicial_quarterly_maintenance tca_review_ai_quarterly } }

tca_judicial_quarterly_maintenance = {
    trigger = {
        OR = {
            government_has_flag = government_is_celestial
            has_character_flag = tca_song_local_names
            has_character_flag = tca_song_central_names
            any_held_title = { has_variable = tca_judicial_title }
            any_held_title = { tca_has_judicial_title_trigger = yes }
        }
    }
    effect = {
        tca_refresh_character_circuits_effect = yes
        tca_refresh_held_judicial_positions_effect = yes
    }
}

tca_judicial_position_government_change = {
    trigger = { OR = { government_has_flag = government_is_celestial has_character_flag = tca_song_local_names has_character_flag = tca_song_central_names any_held_title = { has_variable = tca_judicial_title } } }
    effect = { tca_request_judicial_maintenance_effect = yes }
}

# A commissioner's new county/office or changed liege must also be checked.
on_vassal_change = { on_actions = { tca_judicial_position_government_change tca_test_imperial_vassal_trace } }

# Native council callbacks: emperor-owned commissions must follow the current
# ministry, even when firing leaves a vacancy without a new title-gain callback.
on_fired_from_council = { on_actions = { tca_judicial_council_fired } }
on_councillor_left = { on_actions = { tca_judicial_council_left } }
on_councillors_swapped = { on_actions = { tca_judicial_council_swapped } }
tca_judicial_council_fired = { effect = { tca_request_judicial_maintenance_effect = yes } }
tca_judicial_council_left = { effect = { scope:old_employer ?= { tca_request_judicial_maintenance_effect = yes } } }
tca_judicial_council_swapped = { effect = { scope:councillor_liege ?= { tca_request_judicial_maintenance_effect = yes } } }

# Read-only trace for the original vassal-change path of this disposable fixture.
tca_test_imperial_vassal_trace = {
    trigger = { debug_only = yes has_character_flag = tca_test_em_trace_subject }
    effect = {
        if = { limit = { liege = title:h_china.holder } debug_log = "TCA_EM VASSAL_CALLBACK_DIRECT yes" }
        else = { debug_log = "TCA_EM VASSAL_CALLBACK_DIRECT no" }
        debug_log_scopes = yes
    }
}
