# 真实制度/归属/原生入品与中央授职入口；不覆写原版效果，不手动派发合法gain。
# 新局在原版开局所有初始化完成后通过二级 on_action 同步接管，核定中央现任下限与初始上限；不整文件替换原版 game_start.txt。
on_game_start_after_lobby = { on_actions = { tca_merit_startup } }
on_government_change = { on_actions = { tca_merit_government_changed } }
on_vassal_change = { on_actions = { tca_merit_liege_changed } }
on_join_court = { on_actions = { tca_merit_court_joined } }
on_leave_court = { on_actions = { tca_merit_court_left } }
on_merit_level_gain = { on_actions = { tca_merit_native_gain tca_retirement_rank_changed } }
on_merit_level_loss = { on_actions = { tca_retirement_rank_changed } }
on_title_gain = { on_actions = { tca_merit_central_title_gain tca_merit_cap_title_changed tca_retirement_title_guard } }
on_title_lost = { on_actions = { tca_merit_cap_title_changed } }

# 新开局全局入口；在原版开局 effect 全部跑完后接管，建立中央现任品级下限与初始核定状态。
tca_merit_startup = {
	effect = {
		# 二级回调之间不依赖文件合并顺序；显式先完成本模组预设，再接管原生人物。
		tca_initialize_startup_officials_effect = yes
		tca_merit_finish_native_startup_effect = yes
	}
}

tca_merit_government_changed = {
    trigger = { has_global_variable = tca_merit_startup_ready tca_merit_realm_change_relevant_trigger = yes }
    effect = { tca_merit_sync_realm_membership_effect = yes }
}
tca_merit_liege_changed = {
    trigger = { has_global_variable = tca_merit_startup_ready tca_merit_realm_change_relevant_trigger = yes }
    effect = {
        # 用本分支最后一次完成同步的国家判断；祖先移动也会改变旧领主的当前国家。
        # 首次无快照时安全访问一次；NOT内显式AND，避免把多个条件逐项否定。
        top_liege ?= { save_temporary_scope_as = tca_merit_current_realm }
        if = {
            limit = { NOT = { AND = { has_variable = tca_merit_branch_anchor exists = scope:tca_merit_current_realm var:tca_merit_branch_anchor ?= scope:tca_merit_current_realm } } }
            tca_merit_sync_realm_membership_effect = yes
        }
    }
}
# 原版on_join_court/on_leave_court的root是移动人物；新雇主已在scope:new_employer，
# 离场时旧雇主在scope:old_employer。这里只按移动后的真实top_liege/政体同步一次，
# 不补发考试奖励，也不把宫廷客居当成封臣迁国。
tca_merit_court_joined = {
    trigger = { has_global_variable = tca_merit_startup_ready tca_merit_court_change_relevant_trigger = yes }
    effect = {
        if = {
            limit = { NOT = { has_character_flag = tca_merit_court_sync_pending } }
            add_character_flag = tca_merit_court_sync_pending
            trigger_event = { id = tca_merit_entry.5 days = 1 }
        }
    }
}
tca_merit_court_left = {
    trigger = { has_global_variable = tca_merit_startup_ready tca_merit_court_change_relevant_trigger = yes }
    effect = {
        if = {
            limit = { NOT = { has_character_flag = tca_merit_court_sync_pending } }
            add_character_flag = tca_merit_court_sync_pending
            trigger_event = { id = tca_merit_entry.5 days = 1 }
        }
    }
}
tca_merit_native_gain = {
    trigger = { has_global_variable = tca_merit_startup_ready government_has_flag = government_is_celestial NOT = { has_character_flag = tca_merit_grant_suppress } NOT = { tca_merit_interface_probe_trigger = yes } }
    effect = {
        # 与根gain准入配合；在过渡保护中仍保留本事务首次真实目标gain，后续反弹不能重奖。
        if = { limit = { tca_merit_cap_grant_gain_trigger = yes } set_variable = { name = tca_merit_grant_gain_callback_token value = var:tca_merit_grant_token } }
        tca_merit_accept_native_gain_effect = yes
        tca_merit_maybe_queue_central_catchup_effect = yes
    }
}
tca_merit_central_title_gain = {
    trigger = {
        has_global_variable = tca_merit_startup_ready
        OR = { scope:title = title:e_minister_chancellor scope:title = title:e_minister_censor scope:title = title:e_minister_grand_marshal scope:title = title:e_minister_of_personnel scope:title = title:e_minister_of_revenue scope:title = title:e_minister_of_rites scope:title = title:e_minister_of_war scope:title = title:e_minister_of_justice scope:title = title:e_minister_of_works }
    }
    effect = { tca_merit_queue_central_effect = yes }
}

# root为原生受封者/失去头衔者；取得/失去最后一个头衔后，次日读取最终统治身份。
# 不重新初始化事务中的核定等级，仅刷新上限快照；同日多次转授只排一次。
tca_merit_cap_title_changed = {
    trigger = { has_global_variable = tca_merit_startup_ready has_character_flag = tca_merit_active NOT = { tca_merit_interface_probe_trigger = yes } }
    effect = { tca_merit_queue_cap_sync_effect = yes }
}

# 原生受封者人物作用域；scope:title是本次头衔。退休身份不影响家门继承。
tca_retirement_title_guard = {
    trigger = { tca_retirement_is_exited_trigger = yes scope:title = { is_noble_family_title = no NOT = { has_variable = tca_retirement_title } } }
    effect = {
        if = { limit = { NOT = { has_character_flag = tca_retirement_office_cleanup_pending } }
            add_character_flag = tca_retirement_office_cleanup_pending
            trigger_event = { id = tca_retirement.90 days = 1 }
        }
    }
}

# 原生升降品人物作用域；跨日读取最终等级，同链归零回填只合并一次刷新。
# 授品保护会屏蔽原生根回调，致仕加恩另由其审计成功入口同步。
tca_retirement_rank_changed = {
    trigger = { tca_retirement_is_exited_trigger = yes }
    effect = {
        if = { limit = { NOT = { has_character_flag = tca_retirement_income_sync_pending } }
            add_character_flag = tca_retirement_income_sync_pending
            trigger_event = { id = tca_retirement.93 days = 2 }
        }
    }
}
