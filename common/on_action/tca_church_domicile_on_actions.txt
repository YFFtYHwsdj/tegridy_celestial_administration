# 原生入口只跨文件追加二级回调；不覆盖父effect，也不在旅行每个省份全扫住所。
# 统一排队后枚举全部住所主人，包含previous_holder及一人多住所，不依赖单个角色的主住所。
on_game_start_after_lobby = { on_actions = { tca_church_domicile_changed } }
on_title_gain = { on_actions = { tca_church_domicile_changed } }
on_title_lost = { on_actions = { tca_church_domicile_changed } }
on_title_destroyed = { on_actions = { tca_church_domicile_changed } }
on_government_change = { on_actions = { tca_church_domicile_changed } }
on_domicile_moved = { on_actions = { tca_church_domicile_changed } }
on_county_faith_change = { on_actions = { tca_church_domicile_changed } }
on_vassal_change = { on_actions = { tca_church_domicile_changed } }

tca_church_domicile_changed = {
    effect = { tca_church_queue_domicile_sync_effect = yes }
}
# 局势作用域；原生排队对象即原Church，局势不存在后无需重建或补发。
tca_church_domicile_worker = {
    effect = {
        remove_variable = tca_church_domicile_pending
        tca_church_sync_domicile_candidates_effect = yes
    }
}
