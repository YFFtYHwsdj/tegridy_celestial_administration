# 军事差遣的原生回调追加；只挂载独立子on_action，不替换原版effect。
# 开局与季度由独立天朝最高领主集中补缺；改制只检查改制者自己的差遣。
# 主君死亡前释放模块保护并向实际员工排次日重绑，继任关系由原生结算。

on_game_start_after_lobby = {
	on_actions = {
		tca_military_court_positions_startup
	}
}

quarterly_playable_pulse = {
	on_actions = {
		tca_military_court_positions_quarterly_pulse
	}
}

# 开局全局回调：定位天朝独立王国及以上君主并触发开局首次补缺
tca_military_court_positions_startup = {
	effect = {
		every_independent_ruler = {
			limit = {
				is_alive = yes
				government_has_flag = government_is_celestial
				highest_held_title_tier >= tier_kingdom
			}
			tca_military_quarterly_fill_vacancies_effect = yes
		}
	}
}

# 季度脉冲回调：仅独立天朝王国及以上君主触发，杜绝各级封臣重复执行
tca_military_court_positions_quarterly_pulse = {
	trigger = {
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		government_has_flag = government_is_celestial
	}
	effect = {
		tca_military_quarterly_fill_vacancies_effect = yes
	}
}

# 原生根为改制人物；用独立子回调接入，不覆盖原版政府变更effect。
on_government_change = { on_actions = { tca_military_government_changed } }
tca_military_government_changed = {
    trigger = {
        OR = {
            employs_court_position = tca_military_supervisor_court_position
            employs_court_position = tca_military_inspector_court_position
            employs_court_position = tca_corps_inspector_court_position
            employs_court_position = tca_civil_adjutant_court_position
            employs_court_position = tca_household_registrar_court_position
            employs_court_position = tca_army_judge_court_position
        }
    }
    effect = { tca_military_validate_office_after_government_change_effect = yes tca_civil_validate_council_positions_effect = yes }
}

# root为死亡前的官署主君；此时沿实际宫廷职位员工关系释放本模块保护。
# 差遣可被原生撤销或随宫廷保留，故向每名实际在职文武官员排次日事件读取真实雇主。
# tca_military_council_rebind_pending由这里设置防重排，并在tca_military_council.1接收时清理。
on_death = { on_actions = { tca_military_governor_death } }
tca_military_governor_death = {
    trigger = {
        OR = {
            employs_court_position = tca_military_supervisor_court_position
            employs_court_position = tca_military_inspector_court_position
            employs_court_position = tca_corps_inspector_court_position
            employs_court_position = tca_civil_adjutant_court_position
            employs_court_position = tca_household_registrar_court_position
            employs_court_position = tca_army_judge_court_position
        }
    }
    effect = {
        every_court_position_holder = {
            limit = { tca_court_person_has_council_commission_trigger = yes }
            tca_military_release_council_lock_effect = { OWNER = root }
            if = {
                limit = { is_alive = yes NOT = { has_character_flag = tca_military_council_rebind_pending } }
                add_character_flag = tca_military_council_rebind_pending
                trigger_event = { id = tca_military_council.1 days = 1 }
            }
        }
    }
}
