# 追加原生王朝循环月脉冲；不替换原版局势效果。
dynastic_cycle_monthly_pulse = { on_actions = { tca_office_funding_monthly } }
on_title_gain = { on_actions = { tca_office_funding_title_gain } }

# 原生受封者人物作用域；scope:title 为本次授予头衔。
# 新任期清理旧收据，付款日期留在官署，不能借同人复任重复领款。
tca_office_funding_title_gain = {
    trigger = { scope:title = { OR = { has_variable = tca_judicial_title has_variable = tca_tongpan_title } } }
    effect = {
        scope:title = {
            remove_variable = tca_office_funding_last_merit
            remove_variable = tca_office_funding_last_merit_recipient
        }
    }
}

tca_office_funding_monthly = {
    effect = {
        tca_settle_all_office_funding_effect = yes
        if = { limit = { debug_only = yes } debug_log = "TCA_FUNDING MONTHLY_CALLBACK" }
    }
}
