# 新持有人为根人物。仅接入正式任命；延迟至次日读取到任后的正式首府与契约。
on_title_gain = { on_actions = { tca_province_contract_appointment } }

tca_province_contract_appointment = {
    trigger = {
        OR = {
            scope:transfer_type = flag:appointment
            scope:transfer_type = flag:appointment_succession
        }
    }
    effect = { trigger_event = { on_action = tca_province_contract_appointment_settled days = 1 } }
}

tca_province_contract_appointment_settled = {
    trigger = {
        is_alive = yes
        scope:title ?= {
            holder = root
            this = root.primary_title
        }
    }
    effect = { tca_province_contract_review_appointment_effect = yes }
}
