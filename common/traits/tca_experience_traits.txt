# 三种任职经验：基础属性在特质本体，轨道节点只写增量，保持累计总效果为 1/0、1/1、2/1、2/2、3/2。
# 名称在 0/50/100 变化；不影响文武职业资格、Merit 或职位任用。专用图标与特质标识同名。
tca_clerical_experience = {
    category = fame
    icon = tca_clerical_experience.dds
    shown_in_ruler_designer = no
    stewardship = 1
    track = {
        25 = { learning = 1 }
        50 = { stewardship = 1 }
        75 = { learning = 1 }
        100 = { stewardship = 1 }
    }
    name = { first_valid = {
        triggered_desc = { trigger = { NOT = { exists = this } } desc = trait_tca_clerical_experience_1 }
        triggered_desc = { trigger = { exists = this has_trait_xp = { trait = tca_clerical_experience value >= 100 } } desc = trait_tca_clerical_experience_3 }
        triggered_desc = { trigger = { exists = this has_trait_xp = { trait = tca_clerical_experience value >= 50 } } desc = trait_tca_clerical_experience_2 }
        desc = trait_tca_clerical_experience_1
    } }
    desc = trait_tca_clerical_experience_desc
}
