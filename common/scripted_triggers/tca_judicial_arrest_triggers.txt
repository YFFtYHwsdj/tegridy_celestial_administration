# Character scope. Ordinary civilian circuits and subordinate state/county offices.
tca_is_judicial_governor_trigger = {
    is_alive = yes
    is_imprisoned = no
    any_held_title = { tca_is_local_administrative_title_trigger = yes }
    tca_has_judicial_circuit_context_trigger = yes
    NOT = { tgp_is_any_minister = yes }
}

tca_is_judicial_commissioner_trigger = {
    exists = primary_title
    primary_title ?= { has_variable = tca_judicial_title }
    trigger_if = {
        limit = { primary_title ?= { has_variable = tca_judicial_title exists = var:song_judicial_circuit } }
        primary_title = {
            tca_is_valid_judicial_holder_trigger = yes
            var:song_judicial_circuit = {
                tca_is_judicial_circuit_trigger = yes
                tca_has_judicial_title_trigger = yes
                NOT = { has_variable = tca_judicial_disabled }
            }
        }
    }
    trigger_else = { always = no }
}

tca_uses_judicial_arrest_trigger = {
    tca_has_judicial_institutions_trigger = yes
    OR = {
        tca_is_judicial_governor_trigger = yes
        tca_is_judicial_commissioner_trigger = yes
        AND = {
            exists = primary_title
            primary_title ?= { has_variable = tca_judicial_title }
        }
    }
}

# Shared circuit membership; protected offices are classified by score, not excluded.
tca_is_judicial_target_in_circuit_trigger = {
    tca_is_target_in_administrative_title_trigger = { TITLE = $CIRCUIT$ }
}

# Use engine-saved participants; arbitrary redirect scopes are not assumed durable.
tca_judicial_arrest_context_valid_trigger = {
    scope:actor = scope:puppet_or_actor
    scope:actor = { is_alive = yes is_imprisoned = no }
    tca_judicial_actor_has_jurisdiction_trigger = yes
    trigger_if = {
        limit = { scope:actor = { tca_is_judicial_commissioner_trigger = yes } }
        scope:actor.primary_title.var:song_judicial_circuit = { save_temporary_scope_as = tca_checked_circuit }
    }
    trigger_else = {
        scope:actor = {
            OR = {
                any_held_title = {
                    tca_is_judicial_circuit_trigger = yes
                    NOT = { has_variable = tca_judicial_disabled }
                    scope:recipient = { tca_is_judicial_target_in_circuit_trigger = { CIRCUIT = prev } }
                    save_temporary_scope_as = tca_checked_circuit
                }
                any_liege_or_above = {
                    any_held_title = {
                        tca_is_judicial_circuit_trigger = yes
                        NOT = { has_variable = tca_judicial_disabled }
                        scope:recipient = { tca_is_judicial_target_in_circuit_trigger = { CIRCUIT = prev } }
                        save_temporary_scope_as = tca_checked_circuit
                    }
                }
            }
        }
    }
    exists = scope:tca_checked_circuit
    trigger_if = {
        limit = { exists = scope:tca_checked_circuit }
        trigger_if = {
            limit = { exists = scope:intermediary }
            tca_judicial_direct_authorized_trigger = no
            scope:intermediary = { this != scope:actor this != scope:recipient tca_is_available_judicial_approver_trigger = yes }
            trigger_if = {
                limit = { tca_judicial_requires_central_trigger = yes }
                scope:intermediary = { tca_is_current_central_approver_trigger = { COURT = scope:actor.top_liege } }
            }
            trigger_else = { scope:intermediary = scope:tca_checked_circuit.var:song_judicial_title.holder }
        }
        trigger_else = { tca_judicial_direct_authorized_trigger = yes }
    }
}

tca_judicial_arrest_still_valid_trigger = {
    tca_judicial_arrest_context_valid_trigger = yes
    trigger_if = {
        limit = { tca_judicial_arrest_context_valid_trigger = yes }
        scope:puppet_or_actor = {
            advanced_allowed_to_imprison_character_trigger = { CHARACTER = scope:recipient }
        }
        trigger_if = {
            limit = { scope:actor = { tca_is_judicial_commissioner_trigger = yes } }
            trigger_if = {
                limit = { scope:recipient = scope:tca_checked_circuit.holder }
                scope:actor.top_liege = {
                    is_alive = yes is_imprisoned = no
                    tca_basic_allowed_to_arrest_character_trigger = { CHARACTER = scope:recipient }
                    NOT = { is_at_war_with = scope:recipient }
                }
            }
            trigger_else = {
                scope:tca_checked_circuit.holder = {
                    is_alive = yes is_imprisoned = no
                    tca_basic_allowed_to_arrest_character_trigger = { CHARACTER = scope:recipient }
                    NOT = { is_at_war_with = scope:recipient }
                }
            }
        }
        trigger_else = { scope:actor = { tca_basic_allowed_to_arrest_character_trigger = { CHARACTER = scope:recipient } } }
        tca_judicial_arrest_native_common_valid_trigger = yes
        NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
        NOT = { scope:recipient ?= scope:tca_checked_circuit.holder.diarch }
        trigger_if = {
            limit = { scope:hook ?= yes }
            scope:puppet_or_actor = { has_usable_hook = scope:recipient }
        }
    }
}
