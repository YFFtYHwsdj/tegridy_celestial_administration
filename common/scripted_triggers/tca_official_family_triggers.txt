# 人物作用域；仅本模组明确选择无世族且仍专职任官者被排除。
# 原有世族、未记录选择、选择建立世族以及转任其它实职者均沿用原版。
# title标记由正式授职入口或固定官署维护设置，不推断所有无地头衔都是本模组官职。
tca_official_noble_family_creation_allowed_trigger = {
    save_temporary_scope_as = tca_family_policy_actor
    OR = {
        # 包括首次选择后继承世族者，继续允许原版补齐缺失庄园。
        any_held_title = { is_noble_family_title = yes }
        NOT = {
            # 必须显式合取后取反；NOT内多个条件不能当作NAND使用。
            AND = {
                has_variable = tca_official_family_decided
                NOT = { has_variable = tca_official_family_wants_noble }
                NOT = { has_variable = tca_official_family_preexisting_noble }
                primary_title = { has_variable = tca_official_family_policy_office }
                NOT = { any_held_title = { is_noble_family_title = no NOT = { this = scope:tca_family_policy_actor.primary_title } } }
            }
        }
    }
}
