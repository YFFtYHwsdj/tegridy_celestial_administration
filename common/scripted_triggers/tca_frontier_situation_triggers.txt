# 角色作用域；无参数；全部固定北疆伯爵领须有持有人且属于本人国家。
# 包括封臣土地，不读取高阶头衔是否创建，也不以宣称代替控制。
# 临时scope仅用于跨县比较，无持久状态写入。
tca_frontier_unification_land_trigger = {
    save_temporary_scope_as = tca_frontier_unification_ruler
    # 避免地理定义损坏为空时，否定式遍历意外判为统一。
    any_county_in_region = { region = tca_frontier_region }
    NOT = {
        any_county_in_region = {
            region = tca_frontier_region
            OR = {
                NOT = { exists = holder }
                holder = { NOT = { top_liege = scope:tca_frontier_unification_ruler } }
            }
        }
    }
}

# 角色作用域；统一结束决议与效果共用，不另加稳定期或按战争目标筛选。
tca_frontier_can_declare_unification_trigger = {
    exists = situation:tca_northern_frontier
    is_independent_ruler = yes
    is_at_war = no
    tca_frontier_unification_land_trigger = yes
}
