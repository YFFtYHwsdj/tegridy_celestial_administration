# 官制法律、外部官名兼容及改革资格；本文件只判断条件，不迁移法律。
# 只用于君主开局默认值，不持续以历史状态覆盖选择。
tca_should_start_with_song_system_trigger = {
    has_title = title:h_china
    OR = { game_start_date = 1066.9.15 game_start_date = 1178.10.1 }
}
# 可选模组在 on_game_start 设置这些标记，早于本模组离开大厅后的初始化。
# 绝不引用其法律标记：缺失定义会导致引擎加载错误。
tca_external_official_names_trigger = {
    OR = {
        has_global_variable = orient_empires_loaded
        has_global_variable = hanfan_loaded
    }
}
# 角色作用域；以实际最高领主的地方法律决定称号，外部官名模组启用时让出命名。
tca_uses_song_local_names_trigger = {
    tca_external_official_names_trigger = no
    top_liege = { has_realm_law_flag = tca_local_names_song }
}
# 角色作用域；中央官名独立读取中央法律，不随地方官名开关变化。
tca_uses_song_central_names_trigger = {
    tca_external_official_names_trigger = no
    top_liege = { has_realm_law_flag = tca_central_names_song }
}
# 角色作用域；实际最高领主须同时具备天朝政体及司法机构启用法。
tca_has_judicial_institutions_trigger = {
    top_liege = {
        government_has_flag = government_is_celestial
        has_realm_law_flag = tca_judicial_institutions_enabled
    }
}

# 正式法律和引擎回归测试共用。
tca_names_reform_can_pass_trigger = { is_imprisoned = no }
# 改革者作用域；司法改革受共用冷却限制，官名改革不使用此冷却。
tca_judicial_reform_can_pass_trigger = {
    is_imprisoned = no
    custom_tooltip = { text = tca_court_system_cooldown NOT = { has_variable = tca_court_system_cooldown } }
}
