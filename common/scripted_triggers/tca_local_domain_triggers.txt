# 角色作用域；国家政策只归实际独立天朝统治者所有，帝国级路长官不因此豁免。
tca_local_domain_sovereign_trigger = {
    government_has_flag = government_is_celestial
    is_independent_ruler = yes
}

# 地方官员作用域；读取实际最高领主政策和本人的原生文武契约。
# 无地官署、家族身份和未知契约不推定为文官；不判断公国数量或任职特质。
tca_local_domain_restricted_trigger = {
    government_has_flag = government_is_celestial
    is_independent_ruler = no
    is_landed = yes
    top_liege = { tca_local_domain_sovereign_trigger = yes }
    OR = {
        AND = {
            vassal_contract_has_flag = celestial_civil_appointment
            top_liege = { OR = { has_realm_law = tca_local_domain_civil_law has_realm_law = tca_local_domain_all_law } }
        }
        AND = {
            vassal_contract_has_flag = celestial_military_appointment
            top_liege = { has_realm_law = tca_local_domain_all_law }
        }
    }
}

# 改革者作用域；独立冷却，不与官名、司法或通判制度共享。
tca_local_domain_reform_can_pass_trigger = {
    is_imprisoned = no
    custom_tooltip = { text = tca_local_domain_reform_cooldown NOT = { has_variable = tca_local_domain_reform_cooldown } }
}
