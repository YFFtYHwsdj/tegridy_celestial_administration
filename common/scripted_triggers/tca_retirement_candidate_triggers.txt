# 闲退候选技术层；只对显式启用的新探针人物生效，旧付款／退出基线不变。
# 人物作用域；不依赖当前政体、国家或超时，迁国、改制、考试清理不消除身份。
tca_retirement_candidate_is_exited_trigger = {
    has_character_flag = tca_retirement_candidate_enabled
    OR = { has_trait = tca_retirement_probe_fengci has_trait = tca_retirement_probe_retired }
}

# 人物作用域；仅验证候选起复接口，尚非正式健康／高品皇帝审批规则。
# 双身份异常一律拒绝，致仕没有恢复入口；失败不得扣款或清原生冷却。
tca_retirement_candidate_can_resume_trigger = {
    tca_retirement_candidate_is_exited_trigger = yes
    has_trait = tca_retirement_probe_fengci
    NOT = { has_trait = tca_retirement_probe_retired }
    is_alive = yes
    is_adult = yes
    is_incapable = no
    # 本入口必有奉祠；扣除其固定+0.2后，基础健康必须至少为1。
    health >= 1.2
    influence >= 60
    has_variable = tca_retirement_candidate_prior_timeout
    has_variable = tca_retirement_candidate_entered_date
}
