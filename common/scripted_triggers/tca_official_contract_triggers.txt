# 角色作用域；仅显式标记的非家门官职纳入。
# 当前支持提刑及通判官职。
# 未来官职类型必须加入实际登记和回归覆盖。
tca_is_landless_official_trigger = {
    government_has_flag = government_is_celestial
    is_independent_ruler = no
    is_landless_ruler = yes
    primary_title = {
        is_noble_family_title = no
        OR = { has_variable = tca_judicial_title has_variable = tca_tongpan_title }
    }
}
# 人物作用域；按已登记主头衔区分实际官职，供身份维护和契约资格共用。
tca_is_judicial_official_trigger = {
    tca_is_landless_official_trigger = yes
    primary_title = { has_variable = tca_judicial_title }
}
tca_is_tongpan_official_trigger = {
    tca_is_landless_official_trigger = yes
    primary_title = { has_variable = tca_tongpan_title }
}

# 人物作用域；两类经费契约互斥显示，沿用既有正式官署识别条件。
tca_uses_circuit_budget_trigger = {
    tca_is_judicial_official_trigger = yes
}
tca_uses_tongpan_budget_trigger = {
    tca_is_tongpan_official_trigger = yes
}

# 契约所属人物作用域；EDITOR 为改约者角色。
# 通判只接受对应州主官或州主官的直属领主，其他人物沿用原生直属领主权限。
# 同时检查当前专职持有人、有效州资格和双向绑定，避免旧辖区或错绑授予权限。
tca_can_edit_admin_contract_trigger = {
    trigger_if = {
        limit = { tca_is_tongpan_official_trigger = yes }
        top_liege = $EDITOR$.top_liege
        primary_title = {
            tca_is_valid_tongpan_holder_trigger = yes
            exists = var:tca_tongpan_county
            var:tca_tongpan_county = {
                tca_is_tongpan_county_trigger = yes
                tca_has_tongpan_office_trigger = yes
                var:tca_tongpan_office = prev
                OR = {
                    holder = $EDITOR$
                    holder.liege = $EDITOR$
                }
            }
        }
    }
    trigger_else = { liege = $EDITOR$ }
}
