# Character scope. Only explicitly marked non-family offices opt in.
# Current supported offices are judicial civil offices.
# Future office types must add their actual registration and regression coverage.
tca_is_landless_official_trigger = {
    government_has_flag = government_is_celestial
    is_independent_ruler = no
    is_landless_ruler = yes
    primary_title = {
        is_noble_family_title = no
        has_variable = tca_judicial_title
    }
}
tca_is_military_official_trigger = {
    # No military office is implemented yet. Keep the public trigger for callers.
    always = no
}
tca_is_civil_official_trigger = {
    tca_is_landless_official_trigger = yes
    NOT = { tca_is_military_official_trigger = yes }
}
