# AI入口均使用正式资格；本文件不保存评分或下一次检查日期。
# 人物作用域。请求、授品及加恩尚未结案时，避免同一人物再发起新的进退事务。
tca_retirement_ai_request_available_trigger = {
    NOT = { has_variable = tca_retirement_request_kind }
    NOT = { has_character_flag = tca_retirement_request_denied }
    NOT = { has_variable = tca_merit_grant_phase }
    NOT = { has_character_flag = tca_merit_grant_suppress }
    NOT = { has_variable = tca_retirement_honor_target }
    NOT = { has_character_flag = tca_retirement_honor_failed }
}
# 90日检查只面向至少有一种可行进退选择的AI；不是独立人口登记或全球脉冲。
tca_retirement_ai_can_evaluate_trigger = {
    is_ai = yes is_alive = yes is_adult = yes
    exists = top_liege this != top_liege
    top_liege = { government_has_flag = government_is_celestial }
    NOT = { has_trait = tca_retired }
    tca_retirement_ai_request_available_trigger = yes
    OR = { tca_retirement_can_fengci_trigger = yes tca_retirement_can_retire_trigger = yes tca_retirement_can_resume_trigger = yes }
}
# 年度传家与玩家手动传家共用继承资格，但AI自然评估限于本朝体系。
tca_retirement_ai_can_consider_transfer_trigger = {
    is_ai = yes
    tca_retirement_income_valid_trigger = yes
    tca_retirement_ai_request_available_trigger = yes
    tca_retirement_can_transfer_trigger = yes
}
