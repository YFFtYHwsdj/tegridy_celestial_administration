# 天朝地方主官身份；均为人物作用域，只读，无参数或状态输出。
# 地方有地主官是地方主官的子集，实土功能必须继续使用带 landed 的入口。
# 生存、成年、健康、品级和具体行动权限由调用者按业务另行检查。

# 地方有地主官：非独立天朝实土领主；保留原有军事与宫廷差遣边界。
tca_is_landed_celestial_governor_trigger = {
    is_ruler = yes
    is_landed = yes
    is_independent_ruler = no
    government_has_flag = government_is_celestial
}

# 地方主官：地方有地主官，加上实际登记的无地地方官署主官（提刑、通判）。
# 无地官署复用正式身份入口；不纳入中央部院、世族家门或奉祠致仕头衔。
# 新增无地地方官署须先在正式身份入口登记，不以无地身份或头衔等级推断。
tca_is_celestial_governor_trigger = {
    OR = {
        tca_is_landed_celestial_governor_trigger = yes
        tca_is_landless_official_trigger = yes
    }
}
