# 人物作用域；原生季度决议的硬资格，不把发送概率混入资格。
# 本批仅显式调试授权，两个开关分别授权战争与自然AI；复用参与者资格，仍不授予无地官署进攻辖权。
tca_fe_actor_trigger = {
    debug_only = yes
    is_ai = yes
    has_character_flag = tca_frontier_border_war_enabled
    has_character_flag = tca_frontier_ai_enabled
    exists = situation:tca_northern_frontier
    is_alive = yes is_landed = yes is_imprisoned = no is_in_debt = no
    tca_fs_finance_value >= -30
    military_power > 0
    tca_frontier_is_participant_trigger = yes
    NOT = { any_character_war = { primary_attacker = prev tca_fs_war_trigger = yes } }
}
# 预测story作用域；我方须达到预计防御战力的1.25倍，零兵力不得利用零分母通过。
tca_fe_power_sufficient_trigger = {
    scope:actor = { military_power > 0 military_power >= { value = scope:tca_fs_record.var:tca_fe_defence multiply = 1.25 } }
}
