# D记录story作用域；原受攻击者即使失去全部土地，仍按战前国家代表头衔识别国别。
# 自带记录上下文；只核对实际失地，不向第三方索地，不追讨已经归还的土地。
tca_fd_land_valid_trigger = {
    save_temporary_scope_as = tca_fd_checked_land
    exists = var:tca_fs_defender
    exists = var:tca_fd_enemy
    exists = var:tca_fd_country.holder
    var:tca_fs_defender = { is_alive = yes }
    # 以代表头衔当前持有人承接正常继位；不沿其新top_liege追随改隶或国家被兼并。
    # 旧调试记录缺少快照时取消，不能用当前归属补写并掩盖已经发生的改隶。
    exists = var:tca_fd_attacker_country.holder
    var:tca_fd_attacker_country ?= { holder ?= { is_alive = yes is_independent_ruler = yes primary_title = scope:tca_fd_checked_land.var:tca_fd_attacker_country } }
    var:tca_fd_enemy = {
        is_alive = yes
        top_liege = scope:tca_fd_checked_land.var:tca_fd_attacker_country.holder
    }
    var:tca_fd_country ?= { holder ?= { is_alive = yes is_landed = yes is_independent_ruler = yes primary_title = scope:tca_fd_checked_land.var:tca_fd_country NOT = { top_liege = scope:tca_fd_checked_land.var:tca_fd_enemy.top_liege } } }
    trigger_if = {
        limit = { var:tca_fd_state ?= 3 }
        var:tca_fd_leader ?= { tca_fd_person_valid_trigger = { RECORD = scope:tca_fd_checked_land } }
    }
    any_in_list = { variable = tca_fd_losses always = yes }
    any_in_list = { variable = tca_fd_losses tca_fd_loss_valid_trigger = yes }
}
# 失地行作用域；kind=1仅由已经核实的带地改隶适配写入，不从当前外国身份倒推战果。
# 输入tca_fd_checked_land记录，由土地核验／剔除事务建立；直接夺地保留旧主与国别边界，改隶型须原主持县且仍在本次征服者辖下。
tca_fd_loss_valid_trigger = {
    exists = var:tca_fd_title.holder
    exists = var:tca_fd_holder
    var:tca_fd_holder ?= { is_alive = yes }
    trigger_if = {
        limit = { var:tca_fd_transfer_kind ?= 1 }
        var:tca_fd_title ?= { holder = prev.var:tca_fd_holder }
        var:tca_fd_holder ?= { is_vassal_or_below_of = scope:tca_fd_checked_land.var:tca_fd_enemy }
    }
    trigger_else_if = {
        limit = { var:tca_fd_transfer_kind ?= 2 }
        var:tca_fd_title ?= { holder ?= { OR = { this = scope:tca_fd_checked_land.var:tca_fd_enemy is_vassal_or_below_of = scope:tca_fd_checked_land.var:tca_fd_enemy } } }
        var:tca_fd_holder ?= { OR = {
            top_liege = scope:tca_fd_checked_land.var:tca_fd_country.holder.top_liege
            top_liege = scope:tca_fd_checked_land.var:tca_fd_attacker_country.holder
            AND = { is_landed = no exists = liege_or_court_owner liege_or_court_owner ?= { top_liege = scope:tca_fd_checked_land.var:tca_fd_attacker_country.holder } }
        } }
    }
    trigger_else = {
        var:tca_fd_title ?= { holder ?= { OR = { this = scope:tca_fd_checked_land.var:tca_fd_enemy is_vassal_or_below_of = scope:tca_fd_checked_land.var:tca_fd_enemy } } }
        var:tca_fd_holder ?= {
            OR = {
                AND = {
                    is_landed = no
                    NOT = { tca_is_celestial_governor_trigger = yes }
                    OR = {
                        NOT = { exists = liege_or_court_owner }
                        liege_or_court_owner ?= { top_liege = scope:tca_fd_checked_land.var:tca_fd_country.holder.top_liege }
                    }
                }
                top_liege = scope:tca_fd_checked_land.var:tca_fd_country.holder.top_liege
            }
        }
    }
}
# 人物作用域；参数RECORD为所属战争记录，共享国别和敌对检查。无地地方主官仍由正式身份入口纳入。
tca_fd_person_valid_trigger = {
    exists = $RECORD$.var:tca_fd_country.holder
    is_alive = yes
    is_imprisoned = no
    OR = { is_landed = yes tca_is_celestial_governor_trigger = yes }
    top_liege = $RECORD$.var:tca_fd_country.holder.top_liege
    this != $RECORD$.var:tca_fd_enemy
    NOT = { is_allied_in_war = $RECORD$.var:tca_fd_enemy }
    NOT = { is_at_war_with = $RECORD$.var:tca_fs_defender }
}
# C承诺行作用域；只有真实上级且明确承诺者可牵头。仍有地的受攻击者须保留实际上级关系。
tca_fd_leader_valid_trigger = {
    var:tca_fs_kind = 2
    has_variable = tca_fs_accepted var:tca_fs_accepted ?= yes
    var:tca_fd_eligible ?= yes
    # 原战关闭后的未加入行是state=4；仅排除已参战行，兼容旧组件夹具没有state的记录。
    NOT = { var:tca_fs_state ?= 3 }
    exists = var:tca_fs_person
    var:tca_fs_person = {
        tca_fd_person_valid_trigger = { RECORD = scope:tca_fs_record }
        OR = {
            scope:tca_fs_record.var:tca_fs_defender = { is_landed = no }
            is_liege_or_above_of = scope:tca_fs_record.var:tca_fs_defender
            # 战果改隶破坏的是旧领主关系，不否定战果前已经冻结的上级承诺。
            AND = {
                exists = scope:tca_fs_record.var:tca_fd_attacker_country.holder
                scope:tca_fs_record.var:tca_fd_attacker_country.holder ?= { scope:tca_fs_record.var:tca_fs_defender = { top_liege = prev } }
                scope:tca_fs_record = { any_in_list = { variable = tca_fd_losses
                    var:tca_fd_transfer_kind ?= 1
                    var:tca_fd_holder ?= { OR = { this = scope:tca_fs_record.var:tca_fs_defender is_vassal_or_below_of = scope:tca_fs_record.var:tca_fs_defender } }
                } }
            }
        }
        NOT = { is_allied_to = scope:tca_fs_record.var:tca_fd_enemy }
        NOT = { is_at_war_with = scope:tca_fs_record.var:tca_fd_enemy }
    }
}
# C行作用域；只有指定四政体的拒绝才能约束同政体下级。
tca_fd_vetoed_trigger = {
    OR = {
        AND = { scope:tca_fs_record = { has_variable = tca_fd_veto_celestial } var:tca_fs_person = { has_government = celestial_government } }
        AND = { scope:tca_fs_record = { has_variable = tca_fd_veto_meritocratic } var:tca_fs_person = { has_government = meritocratic_government } }
        AND = { scope:tca_fs_record = { has_variable = tca_fd_veto_steppe } var:tca_fs_person = { has_government = steppe_admin_government } }
        AND = { scope:tca_fs_record = { has_variable = tca_fd_veto_administrative } var:tca_fs_person = { has_government = administrative_government } }
    }
}
# D记录作用域；先确认新战争仍在当前主战双方集合内，才解引用war。
tca_fd_live_trigger = {
    save_temporary_scope_as = tca_fd_checked_live
    var:tca_fd_state = 3
    exists = var:tca_fd_leader
    exists = var:tca_fd_war
    var:tca_fd_leader = { is_alive = yes any_character_war = {
        this = scope:tca_fd_checked_live.var:tca_fd_war
        primary_attacker = scope:tca_fd_checked_live.var:tca_fd_leader
        primary_defender = scope:tca_fd_checked_live.var:tca_fd_enemy
        using_cb = tca_frontier_recovery_cb
    } }
}

# 战争作用域；原生死亡继承先于异步on_death／每日维护时，不能因旧人物已死误判失效。
# 只保护原生正在处理死亡的同一场战争；活着的新主仍须属于原代表国家。
# 无可继承者仍由原生inherit规则结束战争，不另建战争或转交无关国家。
tca_ws_recovery_transition_trigger = {
    exists = var:tca_fd_record
    var:tca_fd_record = { has_variable = tca_fd_active }
    OR = {
        primary_attacker = { is_alive = no }
        primary_defender = { is_alive = no }
        AND = {
            var:tca_fd_record.var:tca_fd_leader ?= { is_alive = no }
            primary_attacker = { is_alive = yes top_liege.primary_title = prev.var:tca_fd_record.var:tca_fd_country }
        }
        AND = {
            var:tca_fd_record.var:tca_fd_enemy ?= { is_alive = no }
            primary_defender = { is_alive = yes top_liege.primary_title = prev.var:tca_fd_record.var:tca_fd_attacker_country }
        }
    }
}
