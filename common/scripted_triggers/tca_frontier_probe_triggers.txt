# 北疆P0互动共用资格；调用方必须提供actor、recipient人物作用域，无副作用。
# 只处理显式准备的县级地方目标；边境、阶段、费用等正式资格留P3，不对普通游戏开放。
tca_frontier_probe_ready_trigger = {
    debug_only = yes
    scope:actor = {
        is_alive = yes
        is_landed = yes
        is_at_war = no
        has_variable = tca_frontier_probe_target
        var:tca_frontier_probe_target = scope:recipient
        NOT = { has_variable = tca_frontier_probe_pending }
    }
    scope:recipient = {
        is_alive = yes
        is_landed = yes
        is_at_war = no
        highest_held_title_tier = tier_county
        is_independent_ruler = no
        top_liege != scope:actor.top_liege
        capital_county = { holder = scope:recipient }
        liege = { is_alive = yes }
    }
}
