# 任意作用域；正式游戏启用自然生命周期。CLI适配仅覆盖本模组自有入口，默认关闭。
# 专用生命周期WORLD由准备器在加载前授权，不能用运行后的测试flag代替开局隔离。
tca_frontier_automatic_lifecycle_allowed_trigger = { always = yes }

# 任意作用域；正式游戏不屏蔽通知。年度补漏WORLD仅在加载前关闭此入口，模拟错过变动通知。
# 不影响开局和年度回调，不由测试事件临时切换，也不改变正式重启资格。
tca_frontier_change_notifications_allowed_trigger = { always = yes }

# 人物作用域；必须是拥有北疆土地的独立王国级以上统治者，并非原统一国家。
# 国家沿休止时代表头衔当前持有人所属国家续接，普通继位不算挑战者。
tca_frontier_restart_challenger_trigger = {
    is_alive = yes
    is_independent_ruler = yes
    highest_held_title_tier >= tier_kingdom
    tca_frontier_realm_has_land_trigger = yes
    trigger_if = {
        limit = { exists = global_var:tca_frontier_unified_title.holder }
        NOT = { this = global_var:tca_frontier_unified_title.holder.top_liege }
    }
}
