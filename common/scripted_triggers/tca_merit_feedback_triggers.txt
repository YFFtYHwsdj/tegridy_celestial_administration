# 人物作用域；年度冻结候选在结算后是否仍保持同一国家锚点/路线/来源且未触犯硬条件。
# 参数 LEVEL/ROUTE 来自该人物本轮 tca_ma_frozen_level/route。
# 用于区分“正常落选”和迁国、转轨、退休、中央待授等资格失效；不放宽正式候选规则。
tca_merit_feedback_candidate_intact_trigger = {
    is_alive = yes
    exists = top_liege
    top_liege.primary_title ?= var:tca_ma_frozen_anchor
    tca_merit_ordinary_hard_candidate_trigger = yes
    var:tca_merit_route ?= $ROUTE$
    var:tca_merit_approved_level ?= $LEVEL$
}
