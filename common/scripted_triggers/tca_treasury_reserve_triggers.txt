# 官库留储法触发条件。角色作用域；只判资格，不改任何状态。
# 结算对象为全部天朝官员，排除皇帝、奉祠与致仕；历史变量随主头衔。

# 角色作用域；可作为本法结算对象。
tca_treasury_reserve_scope_trigger = {
    is_alive = yes
    has_treasury = yes
    is_independent_ruler = no
    exists = top_liege
    top_liege = {
        is_alive = yes
        is_independent_ruler = yes
        government_has_flag = government_is_celestial
    }
    # 免缴档停收羡余：本领域官库不进入年度结算。
    NOT = { top_liege = { has_realm_law = tca_treasury_reserve_exempt_law } }
    tca_retirement_is_exited_trigger = no
    exists = primary_title
}

# 角色作用域；应缴额达到门槛。低于门槛不转移、不通知。
tca_treasury_reserve_payable_trigger = {
    tca_treasury_reserve_sweep_value >= tca_treasury_reserve_threshold_value
}

# 角色作用域；天朝独立君主。年度结算清账与开局法律兜底共用。
tca_treasury_reserve_sovereign_trigger = {
    is_alive = yes
    is_independent_ruler = yes
    government_has_flag = government_is_celestial
    exists = primary_title
}
