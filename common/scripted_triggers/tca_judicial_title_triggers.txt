# 头衔作用域；初始范围仅限普通天朝王国级路。
tca_is_judicial_circuit_trigger = {
    tier = tier_kingdom
    is_landless_type_title = no
    is_noble_family_title = no
    exists = holder
    # 动态头衔设为无地前，引擎要求先设置偏好首都。
    OR = {
        exists = title_capital_county
        exists = holder.capital_county
    }
    holder = {
        is_alive = yes
        tca_has_judicial_institutions_trigger = yes
        government_has_flag = government_is_celestial
        OR = {
            AND = { is_independent_ruler = no highest_held_title_tier = tier_kingdom }
            AND = { is_independent_ruler = yes highest_held_title_tier >= tier_empire }
        }
        NOT = { vassal_contract_has_flag = celestial_province_military }
        NOT = { vassal_contract_has_flag = celestial_province_protectorate }
    }
}

# 头衔作用域；持久指针必须反向指回本路。
tca_has_judicial_title_trigger = {
    save_temporary_scope_as = tca_binding_check_circuit
    exists = var:song_judicial_title
    trigger_if = {
        limit = { exists = var:song_judicial_title }
        var:song_judicial_title = {
            has_variable = tca_judicial_title
            exists = var:song_judicial_circuit
            trigger_if = {
                limit = { exists = var:song_judicial_circuit }
                var:song_judicial_circuit = scope:tca_binding_check_circuit
            }
        }
    }
}

# 角色作用域；scope:tca_appointment_liege 为对应路持有人。
# 既有角色功绩保持不变，生成兜底角色另行处理。
tca_is_initial_judicial_candidate_trigger = {
    NOT = { has_character_flag = tca_history_defined_character }
    is_alive = yes
    is_landed = no
    highest_held_title_tier < tier_duchy
    # 修复候选授予后也必须满足专职官员规则。
    NOT = { any_held_title = { is_noble_family_title = no } }
    merit_level >= merit_level_expected_duchy_value
    tgp_can_be_a_minister_trigger = { COURT_OWNER = scope:tca_appointment_liege }
    liege_or_court_owner ?= {
        top_liege = scope:tca_appointment_liege.top_liege
    }
    NOT = { tgp_is_any_minister = yes }
}

# 头衔作用域；在任提刑可同时持有本官职和家门
# 头衔，但不能有土地或第二个非家门官职。已在任的历史官员
# 予以保留；历史人物过滤仅用于新授予。
tca_is_valid_judicial_holder_trigger = {
    save_temporary_scope_as = tca_validated_judicial_title
    exists = holder
    holder = {
        is_alive = yes
        government_has_flag = government_is_celestial
        is_landed = no
        highest_held_title_tier = tier_duchy
        primary_title = scope:tca_validated_judicial_title
        NOT = {
            any_held_title = {
                is_noble_family_title = no
                # 迭代器可能改变 PREV，使用显式保存的官职。
                NOT = { this = scope:tca_validated_judicial_title }
            }
        }
    }
}
