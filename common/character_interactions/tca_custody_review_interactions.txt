# 详覆互动：actor 为提刑，recipient 为关押者，secondary_recipient 为囚犯。
# 原生 cost 扣发送费；真实操作走 on_send，on_accept 只提供安全的确认页预览；AI 从季度入口发起。
tca_custody_review_interaction = {
    category = interaction_category_prison
    icon = imprison
    common_interaction = yes
    use_diplomatic_range = no
    desc = tca_custody_review_interaction_desc
    send_name = tca_review_send_name
    cost = { influence = tca_review_interaction_cost }
    auto_accept = yes
    cooldown = { months = 3 }
    redirect = { tca_review_interaction_pick_prisoner_effect = yes }
    is_shown = { tca_review_interaction_shown_trigger = yes }
    populate_recipient_list = {
        scope:recipient = {
            every_prisoner = {
                limit = { tca_review_prisoner_trigger = { ACTOR = scope:actor KEEPER = scope:recipient } NOT = { has_character_flag = tca_review_pending } NOT = { has_character_flag = tca_review_prisoner_cooldown } }
                add_to_list = characters
            }
        }
    }
    is_valid_showing_failures_only = {
        scope:actor = { tca_review_actor_can_start_trigger = yes }
        scope:recipient = { tca_review_keeper_trigger = { ACTOR = scope:actor } }
        trigger_if = {
            limit = { exists = scope:secondary_recipient }
            scope:secondary_recipient = { tca_review_prisoner_trigger = { ACTOR = scope:actor KEEPER = scope:recipient } NOT = { has_character_flag = tca_review_pending } NOT = { has_character_flag = tca_review_prisoner_cooldown } }
        }
        trigger_if = { limit = { scope:hook ?= yes } scope:actor = { has_usable_hook = scope:recipient } }
    }
    can_send = {
        tca_review_interaction_send_trigger = yes
        trigger_if = {
            limit = { scope:hook ?= yes }
            custom_description = { text = tca_review_hook_preview always = yes }
        }
        trigger_else = {
            custom_description = { text = tca_review_debate_details always = yes }
        }
    }
    send_options_exclusive = no
    send_option = {
        is_valid = { scope:actor = { has_usable_hook = scope:recipient } }
        flag = hook
        localization = tca_review_use_hook
    }
    # on_accept 也会在确认窗口预览中求值。
    # 只有 on_send 可以创建真实请求或安排事件。
    on_send = { tca_review_interaction_send_effect = yes }
    on_accept = { tca_review_interaction_preview_effect = yes }
    ai_will_do = { base = 0 }
}
