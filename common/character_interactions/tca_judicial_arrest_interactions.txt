# 司法拘捕双入口；actor 是实际发起者，recipient 是目标，intermediary 仅承担审批角色。
# 辖区、身份档和审批资格由共享 trigger 决定；回复时重建内部 saved scopes，避免依赖菜单 redirect 残留。
# 需审批的司法拘捕；由原生 intermediary 队列先处理许可，再由目标接受或拒捕。
tca_request_imprison_interaction = {
    desc = tca_request_imprison_interaction_desc
    redirect = {
        tca_setup_judicial_arrest_roles_effect = yes
        tca_setup_judicial_approval_effect = yes
    }
    is_shown = {
        exists = scope:intermediary
        tca_judicial_arrest_context_valid_trigger = yes
    }
    intermediary_notification_text = tca_judicial_approval_notification
    intermediary_breakdown_yes = tca_judicial_approval_yes
    intermediary_breakdown_no = tca_judicial_approval_no
    intermediary_breakdown_maybe = tca_judicial_approval_maybe
    intermediary_answer_accept_key = REPLY_ANSWER_ALLOW
    intermediary_answer_reject_key = REPLY_ANSWER_VETO
    ai_intermediary_maybe = yes
    ai_intermediary_accept = {
        base = 20
        # 原生队列回归控制；只有可丢弃的调试角色携带这些标记。
        modifier = { add = -1000 scope:actor = { debug_only = yes has_character_flag = tca_test_arrest_force_veto } }
        modifier = { add = 1000 scope:actor = { debug_only = yes has_character_flag = tca_test_arrest_force_approval } }
        modifier = { add = 80 desc = tca_judicial_approval_reason scope:actor = { has_imprisonment_reason = scope:recipient } }
        opinion_modifier = { who = scope:intermediary opinion_target = scope:actor multiplier = 0.25 desc = AI_OPINION_REASON }
        opinion_modifier = { who = scope:intermediary opinion_target = scope:recipient multiplier = -0.25 desc = tca_judicial_approval_target_opinion }
        ai_value_modifier = {
            who = scope:intermediary
            ai_honor = {
                if = {
                    limit = { scope:actor = { NOT = { has_imprisonment_reason = scope:recipient } } }
                    value = -0.25
                }
            }
        }
    }
    on_intermediary_accept = { tca_handle_judicial_approval_accept_effect = yes }
    on_intermediary_decline = { tca_handle_judicial_approval_decline_effect = yes }

    interface_priority = 121
    common_interaction = yes
    icon = prison
    category = interaction_category_hostile
    cooldown = { months = 3 }
    cooldown_against_recipient = { years = 1 }
    ai_maybe = yes
    ai_min_reply_days = 4
    ai_max_reply_days = 9
    can_send_despite_rejection = yes
    popup_on_receive = yes
    pause_on_receive = yes
    greeting = negative
    notification_text = tca_judicial_arrest_notification
    pre_answer_maybe_key = ANSWER_MIGHT_SUCCEED
    pre_answer_no_key = ANSWER_CANT_SUCCEED
    pre_answer_yes_key = ANSWER_WILL_SUCCEED
    pre_answer_maybe_breakdown_key = ANSWER_SUM_CHANCE
    on_decline_summary = tca_judicial_arrest_decline_summary
    secondary_scopes_optional = yes
    is_valid = { tca_judicial_arrest_context_valid_trigger = yes }
    is_valid_showing_failures_only = {
        custom_tooltip = { text = tca_judicial_arrest_eligible tca_judicial_arrest_still_valid_trigger = yes }
        custom_tooltip = { text = tca_judicial_arrest_not_puppet NOT = { scope:is_puppet_action ?= yes } }
    }
    can_send = {
        custom_tooltip = { text = tca_judicial_arrest_eligible tca_judicial_arrest_still_valid_trigger = yes }
        scope:puppet_or_actor = {
            custom_description = {
                text = character_interactions_hostile_actions_disabled_delay
                NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
            }
            trigger_if = {
                limit = { is_ai = yes }
                has_imprisonment_reason = scope:recipient
            }
        }
    }
    is_highlighted = { scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient } }
    send_option = {
		is_valid = {
			exists = scope:recipient
			scope:puppet_or_actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
    should_use_extra_icon = { scope:puppet_or_actor = { has_usable_hook = scope:recipient } }
    extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
    send_options_exclusive = no
    cost = { prestige = liege_diarchy_surcharge_interaction_imprison_interaction_value influence = scope:puppet_or_actor.tca_tongpan_influence_surcharge_value }
    on_send = {
        tca_restore_judicial_arrest_context_effect = yes
        tca_test_record_arrest_send_effect = yes
        tca_trace_judicial_arrest_effect = { STAGE = REQUEST_SEND }
        scope:puppet_or_actor = { add_character_flag = { flag = flag_hostile_actions_disabled_delay days = 10 } }
    }
    on_accept = { tca_handle_judicial_arrest_accept_effect = yes }
    on_decline = { tca_handle_judicial_arrest_decline_effect = yes }
    on_auto_accept = { tca_trace_judicial_arrest_effect = { STAGE = REQUEST_AUTO_ACCEPT } }
    # CK3 的互动级 auto_accept 也会自动通过中间审批。
    auto_accept = no
    ai_accept = {
        base = 0
        tca_judicial_target_acceptance = yes
        # MTTH 修正按顺序执行：仅重置豁免的 AI 目标，再加 100。
        # 此项仅影响审批后的目标接受度，不影响中间审批者接受度。
        modifier = {
            factor = 0
            scope:recipient = { tca_judicial_arrest_ai_cannot_refuse_trigger = yes }
        }
        modifier = {
            add = 100
            desc = tca_judicial_arrest_cannot_refuse_reason
            scope:recipient = { tca_judicial_arrest_ai_cannot_refuse_trigger = yes }
        }
    }
    ai_will_do = { base = -100 tca_judicial_arrest_ai_will_do = yes }
    ai_frequency_by_tier = { barony = 0 county = 36 duchy = 10 kingdom = 10 empire = 10 hegemony = 10 }
    ai_targets = { ai_recipients = vassals chance = 0.5 }
    ai_targets = { ai_recipients = courtiers chance = 0.25 }
}

    is_valid_showing_failures_only = {
        custom_tooltip = { text = tca_judicial_arrest_eligible tca_judicial_arrest_still_valid_trigger = yes }

							OR = {
