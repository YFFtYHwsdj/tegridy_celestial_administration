# 家门人物作用域；离任回调可能早于职位移除，也可能处在换职中，次日按最终状态结算。
tca_official_family_schedule_return_check_effect = {
    if = { limit = {
        is_alive = yes
        tca_official_landless_family_trigger = yes
        NOT = { has_character_flag = tca_official_family_return_check_pending }
    }
        add_character_flag = tca_official_family_return_check_pending
        trigger_event = { on_action = tca_official_family_return_check days = 1 }
    }
}

# 家门人物作用域；最高领主已是直属时不另行迁隶，原版回调也会跳过归顶分支。
tca_official_family_return_if_unassigned_effect = {
    if = { limit = { tca_official_family_needs_return_trigger = yes }
        create_title_and_vassal_change = {
            type = appointment
            save_scope_as = tca_official_family_return_change
        }
        change_liege = {
            liege = top_liege
            change = scope:tca_official_family_return_change
        }
        resolve_title_and_vassal_change = scope:tca_official_family_return_change
    }
}
