# 路头衔作用域；查看者由调试决议提供。
tca_show_judicial_title_effect = {
    save_scope_as = tca_viewed_circuit
    tca_resolve_judicial_office_liege_effect = yes
    scope:tca_expected_judicial_liege = { save_scope_as = tca_viewed_expected_liege }
    var:song_judicial_title = {
        save_scope_as = tca_viewed_title
        # 在本地化中求值与 window_title.gui 相同的 Title 数据函数。
        # 无需猜测脚本条件，也不修改状态即可检查。
        tca_judicial_status_message_effect = {
            TITLE = tca_judicial_destroyed_on_succession_title
            DESC = tca_judicial_destroyed_on_succession_desc
        }
        if = {
            limit = { exists = holder }
            holder = { save_scope_as = tca_viewed_holder }
            if = {
                limit = { exists = holder.liege }
                holder.liege = { save_scope_as = tca_viewed_actual_liege }
                if = {
                    limit = { holder.liege = scope:tca_viewed_expected_liege }
                    tca_judicial_status_message_effect = { DESC = tca_judicial_status_held TITLE = tca_judicial_status_held_title }
                }
                else = {
                    tca_judicial_status_message_effect = { DESC = tca_judicial_status_wrong_liege TITLE = tca_judicial_status_wrong_liege_title }
                }
            }
            else = {
                tca_judicial_status_message_effect = { DESC = tca_judicial_status_independent TITLE = tca_judicial_status_wrong_liege_title }
            }
        }
        else = {
            tca_judicial_status_message_effect = { DESC = tca_judicial_status_vacant TITLE = tca_judicial_status_vacant_title }
        }
        if = {
            limit = { exists = holder }
            tca_judicial_status_message_effect = {
                TITLE = tca_judicial_succession_law_title
                DESC = tca_judicial_succession_law_desc
            }
            if = {
                limit = { tca_is_valid_judicial_holder_trigger = yes }
                tca_judicial_status_message_effect = {
                    TITLE = tca_judicial_eligibility_title DESC = tca_judicial_holder_eligible
                }
            }
            else = {
                tca_judicial_status_message_effect = {
                    TITLE = tca_judicial_eligibility_title DESC = tca_judicial_holder_ineligible
                }
            }
            tca_show_judicial_succession_effect = yes
        }
    }
}

# 提刑头衔作用域；查看操作本身不改法律或授予头衔。
tca_judicial_status_message_effect = {
    scope:tca_title_viewer = {
        send_interface_message = {
            type = tca_judicial_diagnostic_message
            title = $TITLE$
            desc = $DESC$
            left_icon = scope:tca_viewed_title
        }
    }
}

# 检查本官职的继任，而非持有人主头衔、玩家或家门继承人。
tca_show_judicial_succession_effect = {
    if = {
        limit = { NOT = { has_order_of_succession = appointment } }
        tca_judicial_status_message_effect = {
            TITLE = tca_judicial_succession_message_title
            DESC = tca_judicial_succession_not_appointment
        }
    }
    else_if = {
        limit = { exists = current_heir }
        current_heir = { save_scope_as = tca_viewed_office_heir }
        if = {
            limit = { current_heir = scope:tca_viewed_circuit.holder }
            tca_judicial_status_message_effect = {
                TITLE = tca_judicial_succession_message_title DESC = tca_judicial_succession_liege_heir
            }
        }
        tca_judicial_status_message_effect = {
            TITLE = tca_judicial_succession_message_title
            DESC = tca_judicial_succession_has_heir
        }
    }
    else = {
        tca_judicial_status_message_effect = {
            TITLE = tca_judicial_succession_message_title
            DESC = tca_judicial_succession_no_heir
        }
    }
}
