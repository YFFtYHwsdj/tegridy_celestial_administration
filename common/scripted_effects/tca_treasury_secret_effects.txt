# 侵吞国库留证入口。作用域：仍存在的计谋；生产由celestial_siphon_treasury调用。
# 原生测试采用固定隐秘性边界的专用计谋，提供相同的赃款变量与受害者参数。
# 参数TARGET：原版结算时的受害人物；读取本计谋累计embezzlement_stake。
# 保存计谋为scope:tca_treasury_secret_scheme供秘密金额写入；无其它saved scopes依赖。
# 将本轮结算标记保存在计谋，防止收尾与失效重复掷骰。
# 无赃款或实施者死亡不新建秘密；未掷中不删除旧秘密。金额写入沿用原版语义。
tca_treasury_resolve_secret_effect = {
    if = {
        limit = { NOT = { has_variable = tca_treasury_secret_resolved } }
        set_variable = { name = tca_treasury_secret_resolved value = yes }
        save_scope_as = tca_treasury_secret_scheme
        if = {
            limit = {
                var:embezzlement_stake > 0
                scheme_owner = { is_alive = yes secret_siphoned_treasury_is_valid_trigger = { OWNER = this TARGET = $TARGET$ } }
            }
            random = {
                chance = tca_treasury_secret_chance_value
                scheme_owner = {
                    add_secret = { type = secret_siphoned_treasury target = $TARGET$ }
                    random_secret = {
                        limit = { secret_type = secret_siphoned_treasury secret_target = $TARGET$ }
                        set_variable = { name = embezzlement_stake value = scope:tca_treasury_secret_scheme.var:embezzlement_stake }
                    }
                }
            }
        }
    }
}
