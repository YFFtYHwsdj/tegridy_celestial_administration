# 共享战争作用域入口；通用开战回调及有限CB声明均可进入，同场只建立一份story。
# 输出tca_fs_record。目标县、上级链和响应行均归该记录；不写人物单一待援战争指针。
tca_fs_start_effect = {
    if = { limit = { tca_fs_war_trigger = yes NOT = { has_variable = tca_fs_record } }
        save_scope_as = tca_fs_war
        primary_attacker = { save_scope_as = tca_fs_attacker }
        primary_defender = { save_scope_as = tca_fs_defender create_story = { type = tca_frontier_support_record save_scope_as = tca_fs_record } }
        set_variable = { name = tca_fs_record value = scope:tca_fs_record }
        scope:tca_fs_record = {
            set_variable = { name = tca_fs_war value = scope:tca_fs_war }
            set_variable = { name = tca_fs_attacker value = scope:tca_fs_attacker }
            set_variable = { name = tca_fs_defender value = scope:tca_fs_defender }
            set_variable = { name = tca_fs_attacker_title value = scope:tca_fs_attacker.primary_title }
            set_variable = { name = tca_fs_defender_title value = scope:tca_fs_defender.primary_title }
            set_variable = { name = tca_ws_country value = scope:tca_fs_defender.top_liege.primary_title }
            set_variable = { name = tca_ws_attacker_country value = scope:tca_fs_attacker.top_liege.primary_title }
            set_variable = { name = tca_ws_schema value = 2 }
            set_variable = { name = tca_fs_started_on value = current_date }
        }
        scope:tca_fs_defender = { add_to_variable_list = { name = tca_fs_records target = scope:tca_fs_record } }
        scope:tca_fs_defender.primary_title = { add_to_variable_list = { name = tca_ws_defender_records target = scope:tca_fs_record } }
        tca_ws_collect_goals_effect = { LIST = tca_fs_goals }
        tca_fs_collect_candidates_effect = yes
        # 等名单完全建立再发信，避免AI即时选择修改迭代中的候选表。
        scope:tca_fs_record = {
            every_in_list = {
                variable = tca_fs_rows
                save_scope_as = tca_fs_row
                scope:tca_fs_record.var:tca_fs_defender = { save_scope_as = tca_fs_letter_sender }
                scope:tca_fs_record.var:tca_fs_attacker = { save_scope_as = tca_fs_letter_attacker }
                save_scope_value_as = { name = tca_fs_letter_delay value = var:tca_fs_delay }
                var:tca_fs_person = { trigger_event = { id = tca_frontier_support.1 } }
            }
        }
    }
}

# 共享候选收集：输入record、defender、attacker和record内已建立的goals；输出rows/people/uppers/neighbors。
# C实际战争与E瞬时预测共用；只收集、不发信、不承诺。原战争存在时排除其已参战者。
tca_fs_collect_candidates_effect = {
        # 先收集真实上级，响应类别不会被后面的邻近或目标县命中改写。
        scope:tca_fs_defender = { save_scope_as = tca_fs_chain }
        save_scope_value_as = { name = tca_fs_depth value = 0 }
        while = {
            limit = { scope:tca_fs_chain = { is_independent_ruler = no exists = liege } }
            scope:tca_fs_chain.liege = { save_scope_as = tca_fs_chain }
            save_scope_value_as = { name = tca_fs_depth value = { value = scope:tca_fs_depth add = 1 } }
            scope:tca_fs_record = { add_to_variable_list = { name = tca_fs_uppers target = scope:tca_fs_chain } }
            scope:tca_fs_chain = { tca_fs_add_candidate_effect = { KIND = 2 DEPTH = scope:tca_fs_depth } }
        }
        scope:tca_fs_record = {
            every_in_list = {
                variable = tca_fs_goals
                holder ?= {
                    tca_fs_add_candidate_effect = { KIND = 1 DEPTH = 0 }
                    every_liege_or_above = {
                        limit = { is_vassal_or_below_of = scope:tca_fs_defender }
                        tca_fs_add_candidate_effect = { KIND = 1 DEPTH = 0 }
                    }
                }
                every_neighboring_county = {
                    limit = { NOT = { scope:tca_fs_record = { is_target_in_variable_list = { name = tca_fs_goals target = prev } } } }
                    if = { limit = { NOT = { scope:tca_fs_record = { is_target_in_variable_list = { name = tca_fs_neighbors target = prev } } } }
                        scope:tca_fs_record = { add_to_variable_list = { name = tca_fs_neighbors target = prev } }
                    }
                }
            }
            every_in_list = {
                variable = tca_fs_neighbors
                holder ?= {
                    tca_fs_add_neighbor_effect = yes
                    every_liege_or_above = { tca_fs_add_neighbor_effect = yes }
                }
            }
        }
        # 仅从正式地方官署注册表补候选，按绑定辖区识别，不扫描全部人物。
        if = { limit = { has_global_variable_list = tca_tongpan_counties }
            every_in_global_list = { variable = tca_tongpan_counties var:tca_tongpan_office.holder ?= { tca_fs_add_office_effect = yes } }
        }
        if = { limit = { has_global_variable_list = tca_judicial_circuits }
            every_in_global_list = { variable = tca_judicial_circuits var:song_judicial_title.holder ?= { tca_fs_add_office_effect = yes } }
        }
}

# 人物作用域，输入tca_fs_record；KIND=1受威胁、2上级、3邻近，DEPTH仅上级有效。
# 行号无全局计数；同一人物在该记录的people列表中只能出现一次。
tca_fs_add_candidate_effect = {
    save_scope_as = tca_fs_candidate
    if = { limit = {
        tca_fs_candidate_trigger = yes
        NOT = { scope:tca_fs_record = { is_target_in_variable_list = { name = tca_fs_people target = prev } } }
        NOT = { scope:tca_fs_record = { has_variable = tca_fs_war var:tca_fs_war ?= { any_war_participant = { this = scope:tca_fs_candidate } } } }
    }
        save_scope_as = tca_fs_candidate
        scope:tca_fs_defender = { create_story = { type = tca_frontier_support_row save_scope_as = tca_fs_row } }
        scope:tca_fs_row = {
            set_variable = { name = tca_fs_parent value = scope:tca_fs_record }
            set_variable = { name = tca_fs_person value = scope:tca_fs_candidate }
            set_variable = { name = tca_fs_kind value = $KIND$ }
            set_variable = { name = tca_fs_depth value = $DEPTH$ }
            set_variable = { name = tca_fs_state value = 0 }
        }
        # E瞬时预测复用候选，但不持久登记继承索引，防止预测结束后留下无效story引用。
        if = { limit = { scope:tca_fs_record = { has_variable = tca_fs_war } }
            scope:tca_fs_candidate = { tca_ws_index_support_row_effect = yes }
        }
        scope:tca_fs_candidate = { save_scope_value_as = { name = tca_fs_delay value = tca_fs_delay_value } }
        scope:tca_fs_row = { set_variable = { name = tca_fs_delay value = scope:tca_fs_delay } }
        scope:tca_fs_record = {
            add_to_variable_list = { name = tca_fs_people target = scope:tca_fs_candidate }
            add_to_variable_list = { name = tca_fs_rows target = scope:tca_fs_row }
        }
    }
}
tca_fs_add_neighbor_effect = {
    save_temporary_scope_as = tca_fs_checked_person
    save_scope_as = tca_fs_candidate
    if = { limit = { tca_fs_neighbor_trigger = yes } tca_fs_add_candidate_effect = { KIND = 3 DEPTH = 0 } }
}
tca_fs_add_office_effect = {
    save_scope_as = tca_fs_candidate
    if = { limit = { tca_is_celestial_governor_trigger = yes }
        if = { limit = { tca_ws_office_in_counties_trigger = { COUNTIES = tca_fs_goals } }
            tca_fs_add_candidate_effect = { KIND = 1 DEPTH = 0 }
        }
        else_if = { limit = {
            NOT = { is_liege_or_above_of = scope:tca_fs_defender }
            tca_ws_office_in_counties_trigger = { COUNTIES = tca_fs_neighbors }
        } tca_fs_add_candidate_effect = { KIND = 3 DEPTH = 0 } }
    }
}

# 承诺行作用域；输入record／row。唯一接受事务：立即记承诺，原生限时变量保存到期点。
# committed_on与delay共同定义预计到达日，不用current_day（月内日）冒充绝对日期。
tca_fs_accept_effect = {
    if = { limit = { var:tca_fs_state = 0 tca_fs_row_valid_trigger = yes }
        set_variable = { name = tca_fs_state value = 1 }
        set_variable = { name = tca_fs_accepted value = yes }
        set_variable = { name = tca_fs_committed_on value = current_date }
        set_variable = { name = tca_fs_wait value = yes days = tca_fs_wait_days_value }
        scope:tca_fs_record = {
            if = { limit = { NOT = { is_target_in_variable_list = { name = tca_fs_committed target = scope:tca_fs_row.var:tca_fs_person } } }
                add_to_variable_list = { name = tca_fs_committed target = scope:tca_fs_row.var:tca_fs_person }
            }
        }
        if = { limit = { var:tca_fs_kind = 3 }
            var:tca_fs_person = { save_scope_as = tca_fs_notice_person }
            save_scope_value_as = { name = tca_fs_notice_delay value = var:tca_fs_delay }
            if = { limit = { var:tca_fs_person = { highest_held_title_tier >= tier_duchy } }
                scope:tca_fs_record.var:tca_fs_defender = {
                    send_interface_toast = { type = event_toast_text_good title = tca_frontier_support_pledged_title desc = tca_frontier_support_pledged_desc left_icon = scope:tca_fs_notice_person }
                }
            }
            else = {
                scope:tca_fs_record.var:tca_fs_defender = {
                    send_interface_message = { type = event_generic_good_text title = tca_frontier_support_pledged_title desc = tca_frontier_support_pledged_desc left_icon = scope:tca_fs_notice_person }
                }
            }
        }
    }
}
tca_fs_refuse_effect = {
    if = { limit = { var:tca_fs_state = 0 }
        set_variable = { name = tca_fs_state value = 2 }
        if = { limit = { var:tca_fs_kind = 2 }
            var:tca_fs_person = { save_scope_as = tca_fs_reply_sender }
            scope:tca_fs_record.var:tca_fs_defender ?= { trigger_event = { id = tca_frontier_support.2 } }
        }
    }
}

# 战争记录作用域；局部story每日维护，仅遍历本战候选；不是全局寻敌或人物扫描。
tca_fs_tick_effect = {
    save_scope_as = tca_fs_record
    # 旧调试记录缺少继承身份与版本时安全关闭，不为旧战争补建新承诺。
    if = { limit = { NOT = { var:tca_ws_schema ?= 2 } }
        if = { limit = { has_variable = tca_fd_active } tca_fd_end_effect = { RESULT = 0 } }
        tca_fs_close_record_effect = { RESULT = 4 }
    }
    if = { limit = { NOT = { has_variable = tca_fs_closed } }
        tca_ws_refresh_parties_effect = yes
    }
    if = { limit = { has_variable = tca_fs_closed }
        if = { limit = { has_variable = tca_fd_active } tca_fd_tick_effect = yes }
        else_if = { limit = { NOT = { has_variable = tca_fs_retention } } end_story = yes }
    }
    else_if = { limit = { tca_fs_live_trigger = yes }
        # 先在继承完整落定后接续及合并行，再遍历正式响应表，避免删除重复行改变在途迭代。
        every_in_list = { variable = tca_fs_rows
            limit = { has_variable_list = tca_ws_successors }
            save_scope_as = tca_fs_row
            tca_ws_resolve_support_successor_effect = yes
        }
        every_in_list = {
            variable = tca_fs_rows
            limit = { OR = { var:tca_fs_state = 0 var:tca_fs_state = 1 } }
            save_scope_as = tca_fs_row
            if = { limit = { NOT = { tca_fs_row_valid_trigger = yes } }
                set_variable = { name = tca_fs_state value = 4 }
                remove_variable = tca_fs_wait
            }
            else_if = { limit = { scope:tca_fs_record.var:tca_fs_war = { is_defender = scope:tca_fs_row.var:tca_fs_person } }
                set_variable = { name = tca_fs_state value = 3 }
                remove_variable = tca_fs_wait
            }
            else_if = { limit = { var:tca_fs_state = 1 OR = {
                NOT = { has_variable = tca_fs_wait }
                any_in_list = { variable = tca_ws_wait_sources NOT = { has_variable = tca_fs_wait } }
            } }
                scope:tca_fs_record.var:tca_fs_war = { add_defender = scope:tca_fs_row.var:tca_fs_person }
                set_variable = { name = tca_fs_state value = 3 }
            }
        }
    }
    else_if = { limit = { NOT = { has_variable = tca_ws_inheritance_grace } } tca_fs_close_record_effect = { RESULT = 4 } }
}

# CB结束前的战争作用域；先冻结参战名单再停止旧任务，结果1胜利、2白和平、3失败、4失效。
tca_fs_finish_effect = {
    if = { limit = { exists = var:tca_fs_record var:tca_fs_record = { NOT = { has_variable = tca_fs_closed } } }
        var:tca_fs_record = { save_scope_as = tca_fs_record }
        every_war_defender = { scope:tca_fs_record = { add_to_variable_list = { name = tca_fs_participants target = prev } } }
        scope:tca_fs_record = {
            # 旧存档可能先结算、后到每日维护；缺版本记录不能进入依赖新身份快照的冻结事务。
            if = { limit = { has_variable = tca_ws_schema var:tca_ws_schema ?= 2 }
                if = { limit = { NOT = { has_variable = tca_ws_prepared } } tca_ws_freeze_commitments_effect = yes }
                tca_fs_close_record_effect = { RESULT = $RESULT$ }
            }
            else = { tca_fs_close_record_effect = { RESULT = 4 } }
        }
    }
}

# 记录作用域；不再解引用原war。结束快照保留30日，供D领用；未领用自动回收。
# D在征服转地后领用并延长记录生命；旧承诺永不自动转入新战争。
tca_fs_close_record_effect = {
    if = { limit = { NOT = { has_variable = tca_fs_closed } }
        set_variable = { name = tca_fs_closed value = $RESULT$ }
        set_variable = { name = tca_fs_closed_on value = current_date }
        set_variable = { name = tca_fs_retention value = yes days = 30 }
        every_in_list = {
            variable = tca_fs_rows
            remove_variable = tca_fs_wait
            if = { limit = { OR = { var:tca_fs_state = 0 var:tca_fs_state = 1 } } set_variable = { name = tca_fs_state value = 4 } }
        }
        remove_variable = tca_fs_war
    }
}
