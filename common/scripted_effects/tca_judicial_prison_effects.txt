# 从引擎标准保存的互动角色重建命名效果作用域。
tca_restore_judicial_arrest_context_effect = {
    clear_saved_scope = tca_position
    clear_saved_scope = tca_commissioner
    tca_setup_judicial_arrest_roles_effect = yes
    scope:actor = { save_scope_as = tca_authority }
    scope:actor = {
        if = {
            limit = { tca_is_judicial_commissioner_trigger = yes exists = scope:tca_circuit }
            save_scope_as = tca_commissioner
            primary_title = { save_scope_as = tca_position }
            # 保留共享角色解析器选定的军事支持统治者。
        }
    }
}

# 直接使用引擎保存的角色提供审批反馈，包括人工回复。
# 批准使用消息流，否决使用带可见正文的屏幕上方弹出通知。
tca_judicial_arrest_approved_effect = {
    scope:actor = {
        send_interface_message = {
            type = tca_judicial_arrest_approval_message
            title = tca_judicial_arrest_approved
            desc = tca_judicial_arrest_approved_message
            left_icon = scope:intermediary
            right_icon = scope:recipient
        }
    }
}

# 否决绝不视为目标拒捕。
tca_judicial_arrest_veto_effect = {
    # 请求者承担政治代价，提刑仅履行职责。
    scope:actor = {
        custom_tooltip = tca_judicial_arrest_veto_preview
        add_prestige = -100
        change_influence = -50
        add_opinion = {
            target = scope:intermediary
            modifier = tca_judicial_arrest_veto_disappointed_opinion
        }
        remove_character_flag = flag_hostile_actions_disabled_delay
        send_interface_message = {
            type = tca_judicial_arrest_veto_message
            title = tca_judicial_arrest_vetoed
            desc = tca_judicial_arrest_veto_message
            left_icon = scope:intermediary
            right_icon = scope:recipient
        }
    }
    scope:recipient = {
        add_opinion = {
            target = scope:actor
            modifier = tca_judicial_arrest_veto_requested_opinion
        }
        add_opinion = {
            target = scope:intermediary
            modifier = tca_judicial_arrest_veto_protected_opinion
        }
    }
}

# 已恢复互动上下文；清除发起者临时敌对行动锁并提示失效，不执行拘捕或否决处罚。
tca_judicial_arrest_invalidated_effect = {
    scope:tca_authority ?= {
        remove_character_flag = flag_hostile_actions_disabled_delay
        send_interface_toast = {
            type = event_toast_effect_neutral
            title = tca_judicial_arrest_invalidated
            left_icon = scope:recipient
        }
    }
}

# 囚犯角色作用域；维护按官职索引的列表，不扫描整个世界。
tca_register_judicial_prisoner_effect = {
    set_variable = { name = tca_judicial_prison_title value = scope:tca_position }
    save_scope_as = tca_registered_prisoner
    scope:tca_position = {
        if = {
            limit = { NOT = { is_target_in_variable_list = { name = tca_judicial_prisoners target = scope:tca_registered_prisoner } } }
            add_to_variable_list = { name = tca_judicial_prisoners target = scope:tca_registered_prisoner }
        }
    }
}

# 囚犯角色作用域；从原登记官职列表移除自己并清除反向指针，官职缺失时仍清除本地登记。
tca_unregister_judicial_prisoner_effect = {
    save_scope_as = tca_released_prisoner
    var:tca_judicial_prison_title ?= {
        remove_list_variable = { name = tca_judicial_prisoners target = scope:tca_released_prisoner }
    }
    remove_variable = tca_judicial_prison_title
}

# 囚犯作用域；scope:tca_custodian 是下一任羁押者。
# 仅使用引擎效果，不重新施加拘捕好感、暴政或理由检查。
# 引擎拒绝对已被囚者使用 imprison，因此同步释放再关押；
# 转移标记在释放回调期间保护本模组登记。
# 原生释放／关押 on_actions 仍会执行，需要完整玩法验收。
tca_transfer_judicial_prisoner_effect = {
    if = {
        limit = {
            is_alive = yes
            is_imprisoned = yes
            exists = scope:tca_custodian
            scope:tca_custodian = { is_alive = yes is_ruler = yes }
            NOT = { this = scope:tca_custodian }
            NOT = { is_imprisoned_by = scope:tca_custodian }
        }
        save_scope_as = tca_transferred_prisoner
        add_character_flag = tca_judicial_transfer_in_progress
        if = {
            limit = { is_in_prison_type = house_arrest }
            release_from_prison = yes
            scope:tca_custodian = { imprison = { target = scope:tca_transferred_prisoner type = house_arrest } }
        }
        else = {
            release_from_prison = yes
            scope:tca_custodian = { imprison = { target = scope:tca_transferred_prisoner type = dungeon } }
        }
        remove_character_flag = tca_judicial_transfer_in_progress
    }
}

# 提刑头衔作用域；在现有任命／领主维护后执行。
tca_reconcile_judicial_prisoners_effect = {
    if = {
        limit = { exists = var:song_judicial_circuit.holder has_variable_list = tca_judicial_prisoners }
        var:song_judicial_circuit.holder = { save_scope_as = tca_custodian }
        if = {
            limit = { tca_is_valid_judicial_holder_trigger = yes }
            holder = { save_scope_as = tca_custodian }
        }
        every_in_list = {
            variable = tca_judicial_prisoners
            if = {
                limit = { is_alive = yes is_imprisoned = yes }
                tca_transfer_judicial_prisoner_effect = yes
            }
            else = { tca_unregister_judicial_prisoner_effect = yes }
        }
    }
}

# 提刑头衔作用域；在撤下持有人之前退休羁押登记。
tca_retire_judicial_prisoners_effect = {
    clear_saved_scope = tca_custodian
    if = {
        limit = { has_variable_list = tca_judicial_prisoners }
        var:song_judicial_circuit.holder ?= { save_scope_as = tca_custodian }
        every_in_list = {
            variable = tca_judicial_prisoners
            tca_transfer_judicial_prisoner_effect = yes
            # 交接失败时保留登记，避免静默丢失羁押数据。
            if = {
                limit = {
                    OR = {
                        is_alive = no
                        is_imprisoned = no
                        AND = { exists = scope:tca_custodian is_imprisoned_by = scope:tca_custodian }
                    }
                }
                tca_unregister_judicial_prisoner_effect = yes
            }
        }
    }
}

# 囚犯作用域；原生拒捕战争或战斗已实际捕获目标后处理。
tca_resolve_pending_judicial_capture_effect = {
    if = {
        limit = {
            exists = var:tca_pending_judicial_title
            exists = var:tca_pending_judicial_governor
            is_imprisoned_by = var:tca_pending_judicial_governor
        }
        var:tca_pending_judicial_title = { save_scope_as = tca_position }
        if = {
            limit = {
                scope:tca_position = {
                    tca_is_valid_judicial_holder_trigger = yes
                    var:song_judicial_circuit = {
                        tca_is_judicial_circuit_trigger = yes
                        tca_has_judicial_title_trigger = yes
                        NOT = { has_variable = tca_judicial_disabled }
                    }
                }
                tca_is_judicial_target_in_circuit_trigger = { CIRCUIT = scope:tca_position.var:song_judicial_circuit }
            }
            scope:tca_position.holder = { save_scope_as = tca_custodian }
            # 转移前先清除，因为 imprison 可能再次触发 on_imprison。
            remove_variable = tca_pending_judicial_title
            remove_variable = tca_pending_judicial_governor
            tca_transfer_judicial_prisoner_effect = yes
            if = {
                limit = { is_imprisoned_by = scope:tca_custodian }
                tca_register_judicial_prisoner_effect = yes
            }
        }
        else = {
            remove_variable = tca_pending_judicial_title
            remove_variable = tca_pending_judicial_governor
        }
    }
}

# 共享回复调度：正式互动和回归测试使用相同门槛。
tca_handle_judicial_approval_accept_effect = {
    tca_restore_judicial_arrest_context_effect = yes
    tca_test_record_arrest_approval_effect = yes
    tca_trace_judicial_arrest_effect = { STAGE = APPROVER_ACCEPT }
    if = {
        limit = { tca_judicial_arrest_still_valid_trigger = yes }
        tca_judicial_arrest_approved_effect = yes
    }
    else = { tca_judicial_arrest_invalidated_effect = yes }
}

# 审批者否决回调；先恢复参与者并重查，只有仍有效的请求才执行否决代价。
tca_handle_judicial_approval_decline_effect = {
    tca_restore_judicial_arrest_context_effect = yes
    tca_test_record_arrest_veto_effect = yes
    tca_trace_judicial_arrest_effect = { STAGE = APPROVER_DECLINE }
    if = {
        limit = { tca_judicial_arrest_still_valid_trigger = yes }
        tca_judicial_arrest_veto_effect = yes
    }
    else = { tca_judicial_arrest_invalidated_effect = yes }
}

# 目标接受回调；恢复并重查后调用原生改编拘捕效果，审批同意本身不执行此收押步骤。
tca_handle_judicial_arrest_accept_effect = {
    tca_restore_judicial_arrest_context_effect = yes
    tca_test_record_arrest_target_accept_effect = yes
    tca_trace_judicial_arrest_effect = { STAGE = TARGET_ACCEPT }
    if = {
        limit = { tca_judicial_arrest_still_valid_trigger = yes }
        tca_judicial_arrest_success_native_effect = yes
        tca_test_record_arrest_custody_effect = yes
        tca_trace_judicial_arrest_effect = { STAGE = TARGET_ACCEPT_EXECUTED }
    }
    else = {
        tca_judicial_arrest_invalidated_effect = yes
        tca_trace_judicial_arrest_effect = { STAGE = TARGET_ACCEPT_CANCELLED }
    }
}

# 目标拒捕回调；恢复并重查后进入原生失败／战争分支，不能与审批者否决混用。
tca_handle_judicial_arrest_decline_effect = {
    tca_restore_judicial_arrest_context_effect = yes
    tca_trace_judicial_arrest_effect = { STAGE = TARGET_DECLINE }
    if = {
        limit = { tca_judicial_arrest_still_valid_trigger = yes }
        tca_judicial_arrest_decline_native_effect = yes
    }
    else = { tca_judicial_arrest_invalidated_effect = yes }
}
