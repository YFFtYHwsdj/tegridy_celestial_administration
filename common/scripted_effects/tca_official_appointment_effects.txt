# 官署头衔作用域；需已有持有人，首次补位由挂靠上级临时持有后调用。
# 幂等安装本职头衔法，保留持有人的领地法及家门法；不创建候选、不转移头衔。
# 头衔法只允许天朝政体；非天朝暂持者不调用add_title_law，避免安装失败后继续评分。
# 提刑先清理仅供旧存档解析的旧法；新法使用独立标识，不被旧法清理误删。
tca_ensure_official_title_succession_effect = {
    if = {
        limit = {
            exists = holder
            holder = { government_has_flag = government_is_celestial }
            has_variable = tca_tongpan_title
            NOT = { has_title_law = tca_tongpan_office_succession_law }
        }
        add_title_law = tca_tongpan_office_succession_law
    }
    else_if = {
        limit = { exists = holder holder = { government_has_flag = government_is_celestial } has_variable = tca_judicial_title }
        if = { limit = { has_title_law = tca_judicial_appointment_succession_law } remove_title_law = tca_judicial_appointment_succession_law }
        if = { limit = { NOT = { has_title_law = tca_judicial_office_succession_law } } add_title_law = tca_judicial_office_succession_law }
    }
}

# 实际获任角色作用域；只在首次／纠错精确授职后调用，日常维护不续期。
# granted事务不会自动设置原生近期受任超时；补齐原版期限，既有更长的拒任／退休超时不缩短。
# 不对暂持官署的上级调用，避免给上级附带冷却。
# 写入冷却的同时写持久标记 tca_recent_appointment（与冷却同期限自动到期）：
# 路主任命继承补授的季度预清只清带本标记（或原生近期受任区间）的冷却，不无条件清 appointment_timeout_days。
tca_set_official_appointment_cooldown_effect = {
    if = {
        limit = { appointment_timeout_days < tca_official_appointment_cooldown_days }
        set_appointment_timeout = { days = tca_official_appointment_cooldown_days desc = tca_appointment_recent_office_desc }
        set_variable = { name = tca_recent_appointment value = yes days = tca_official_appointment_cooldown_days }
    }
}

# 角色作用域；为初次补位生成的专职文官授予正途科举身份。
# 必过童试、解试（乡试）与礼部省试（会试）；殿试独立低概率获得（10%）。
tca_assign_official_scholar_exams_effect = {
    add_character_flag = passed_child_exam
    add_character_flag = passed_provincial_exam
    add_character_modifier = {
        modifier = tgp_passed_provincial_exam_modifier
    }
    add_character_flag = passed_metropolitan_exam
    add_character_modifier = {
        modifier = tgp_passed_metropolitan_exam_modifier
    }
    random = {
        chance = 10
        add_character_flag = passed_palace_exam
        add_character_modifier = {
            modifier = tgp_passed_palace_exam_modifier
        }
    }
}

# 角色作用域；为通判补位新生成的学者赋予科举与初始功绩。
# 官品与殿试解耦独立随机：80% 为九品（100 功绩），20% 为八品（1000 功绩）。
tca_initialize_new_tongpan_scholar_effect = {
    tca_assign_official_scholar_exams_effect = yes
    random_list = {
        80 = {
            set_variable = { name = tca_merit_approved_level value = 1 }
            set_merit_level = 1
            change_merit = county_starting_merit_value
        }
        20 = {
            set_variable = { name = tca_merit_approved_level value = 2 }
            set_merit_level = 2
            change_merit = merit_level_2_value
        }
    }
    add_character_flag = tca_merit_active
    tca_merit_refresh_cap_effect = yes
}

# 角色作用域；为提刑补位新生成的学者赋予科举与初始功绩。
# 官品与殿试解耦独立随机：80% 为七品（2000 功绩），20% 为六品（3500 功绩）。

        }

            set_merit_level = 4
            change_merit = merit_level_4_value
        }
    }
    add_character_flag = tca_merit_active
