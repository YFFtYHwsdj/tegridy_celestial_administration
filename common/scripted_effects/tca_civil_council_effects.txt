# 主君作用域；POSITION静态ID、SKILL为能力底线。共用sync已提供主君scope。
# 原生回调和选席调用同一正式人物资格，显式传入雇主，不遮蔽原生scope；输出由sync消费。
tca_court_select_linked_officer_effect = {
    if = { limit = { NOT = { exists = scope:tca_military_selected_marshal } employs_court_position = $POSITION$ }
        court_position:$POSITION$ = {
            if = { limit = {
                tca_court_person_assignment_validity_trigger = { OWNER = scope:tca_military_council_liege POSITION = $POSITION$ SKILL = $SKILL$ ROUTE = tca_court_civil_route_trigger }
                trigger_if = { limit = { exists = scope:tca_military_departing_officer } this != scope:tca_military_departing_officer }
            } save_scope_as = tca_military_selected_marshal }
        }
    }
}

}
