# Test-only fixtures. Never called from gameplay on_actions or decisions.
# Character scope: disposable test emperor. Rebuilds state for each case.
tca_test_prepare_judicial_law_realm_effect = {
    save_scope_as = tca_test_law_sovereign
    clear_saved_scope = tca_test_law_circuit
    clear_saved_scope = tca_test_law_other_circuit
    clear_saved_scope = tca_test_law_position
    clear_saved_scope = tca_test_law_commissioner
    clear_saved_scope = tca_test_law_target
    clear_saved_scope = actor
    clear_saved_scope = recipient
    clear_saved_scope = intermediary
    clear_saved_scope = puppet_or_actor
    clear_saved_scope = is_puppet_action
    clear_saved_scope = hook
    if = { limit = { has_realm_law = tca_court_system_song_law } remove_realm_law = tca_court_system_song_law }
    if = { limit = { has_realm_law = tca_court_system_tang_law } remove_realm_law = tca_court_system_tang_law }
    set_variable = { name = tca_court_system_migrated_v2 value = yes }
    remove_variable = tca_court_system_cooldown
    clear_variable_list = tca_test_law_circuits
    if = { limit = { NOT = { has_realm_law = tca_local_names_default_law } } add_realm_law_skip_effects = tca_local_names_default_law }
    if = { limit = { NOT = { has_realm_law = tca_central_names_default_law } } add_realm_law_skip_effects = tca_central_names_default_law }
    if = { limit = { NOT = { has_realm_law = tca_judicial_institutions_enabled_law } } add_realm_law_skip_effects = tca_judicial_institutions_enabled_law }
    every_vassal_or_below = {
        every_held_title = {
            limit = { tca_is_judicial_circuit_trigger = yes }
            remove_variable = tca_judicial_disabled
        }
    }
    tca_refresh_court_system_realm_effect = yes
    # Appointment/title callbacks can change the political graph during repair.
    # Take the assertion snapshot AFTER repair, not from its earlier input set.
    every_vassal_or_below = {
        every_held_title = {
            limit = { tca_is_judicial_circuit_trigger = yes }
            save_scope_as = tca_test_law_collect_circuit
            scope:tca_test_law_sovereign = { add_to_variable_list = { name = tca_test_law_circuits target = scope:tca_test_law_collect_circuit } }
        }
    }
    every_in_list = {
        variable = tca_test_law_circuits
        set_variable = { name = tca_test_law_original_governor value = holder }
        remove_variable = tca_test_law_original_position
        remove_variable = tca_test_law_original_commissioner
        if = {
            limit = { exists = var:song_judicial_title }
            set_variable = { name = tca_test_law_original_position value = var:song_judicial_title }
            if = {
                limit = { exists = var:song_judicial_title.holder }
                set_variable = { name = tca_test_law_original_commissioner value = var:song_judicial_title.holder }
            }
        }
        if = { limit = { NOT = { exists = scope:tca_test_law_circuit } } save_scope_as = tca_test_law_circuit }
        else_if = { limit = { NOT = { exists = scope:tca_test_law_other_circuit } } save_scope_as = tca_test_law_other_circuit }
    }
    scope:tca_test_law_circuit.var:song_judicial_title ?= { save_scope_as = tca_test_law_position }
    scope:tca_test_law_position.holder ?= { save_scope_as = tca_test_law_commissioner }
    # Isolate prisoner cases from previously registered disposable characters.
    scope:tca_test_law_position ?= { tca_retire_judicial_prisoners_effect = yes }
}

# Test-only native replacement transaction, with a fresh dedicated-office scholar.

    scope:actor = { save_scope_as = tca_test_replacement_requester }
    scope:recipient = { save_scope_as = tca_test_replacement_target }
    scope:intermediary = { save_scope_as = tca_test_replacement_old_approver }
    scope:tca_test_law_circuit.holder = {
        create_character = {
            template = tgp_minister_scholar_template
            employer = this
            faith = this.faith
            rite = this.rite
            culture = this.culture
            save_scope_as = tca_test_replacement
        }
    }
    scope:tca_test_replacement = {
        if = { limit = { is_lowborn = yes } create_dynasty = { spread_to_descendants = yes } }
        create_noble_family_effect = { GOVERNMENT_GIVER = scope:tca_test_law_circuit.holder }
        create_title_and_vassal_change = {
            type = appointment
            save_scope_as = tca_test_replacement_change
            add_claim_on_loss = no
        }
        scope:tca_test_law_position = {
            change_title_holder = { holder = scope:tca_test_replacement change = scope:tca_test_replacement_change }
        }
        resolve_title_and_vassal_change = scope:tca_test_replacement_change
        change_government = celestial_government
    }
    scope:tca_test_law_circuit = { save_scope_as = tca_circuit }
    scope:tca_test_law_position = {
        tca_sync_judicial_holder_liege_effect = yes
        holder ?= { tca_ensure_native_judicial_realm_succession_effect = yes }
        holder ?= { tca_ensure_official_contract_effect = yes }
    }
    # Keep the original pending interaction participants through preparation callbacks.
    scope:tca_test_replacement_requester = { save_scope_as = actor save_scope_as = puppet_or_actor }
    scope:tca_test_replacement_target = { save_scope_as = recipient }
    scope:tca_test_replacement_old_approver = { save_scope_as = intermediary }
}
