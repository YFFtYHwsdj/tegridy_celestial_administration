# Judicial title scope. Declare a fixed institution's circuit from title history.
# Idempotent: a valid existing office (including a legacy dynamic one) wins.
tca_bind_judicial_circuit_effect = {
    if = {
        limit = { game_start_date = 1066.9.15 exists = $CIRCUIT$ }
        save_scope_as = tca_binding_position
        set_variable = { name = tca_judicial_static_title value = yes }
        # Fixed object persistence is declared in landed_titles, not the dynamic-only effect.
        set_destroy_on_succession = no
        $CIRCUIT$ = {
            # A separate fallback pointer does not replace the active old-save binding.
            set_variable = { name = tca_static_judicial_title value = scope:tca_binding_position }
            if = {
                limit = { NOT = { tca_has_judicial_title_trigger = yes } }
                save_scope_as = tca_circuit
                set_variable = { name = song_judicial_title value = scope:tca_binding_position }
                scope:tca_binding_position = {
                    set_variable = { name = tca_judicial_title value = yes }
                    set_variable = { name = song_judicial_circuit value = scope:tca_circuit }
                    set_coa = scope:tca_circuit
                }
                if = {
                    limit = { NOT = { has_variable = tca_judicial_registered } }
                    add_to_global_variable_list = { name = tca_judicial_circuits target = this }
                    set_variable = { name = tca_judicial_registered value = yes }
                }
            }
        }
    }
}
