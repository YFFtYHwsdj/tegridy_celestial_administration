# Character scope: the relevant emperor, not a global minister title.
# Resolve only the first available office. A target cannot approve their own arrest.
tca_resolve_central_judicial_authority_effect = {
    if = {
        limit = { tca_has_available_justice_trigger = yes }
        random_councillor = {
            limit = { has_council_position = minister_justice tca_is_available_judicial_approver_trigger = yes }
            save_scope_as = tca_approver
        }
    }
    else_if = {
        limit = { tca_has_available_censor_trigger = yes }
        random_councillor = {
            limit = { has_council_position = councillor_spymaster tca_is_available_judicial_approver_trigger = yes }
            save_scope_as = tca_approver
        }
    }
    else_if = {
        limit = { tca_has_available_chancellor_trigger = yes }
        random_councillor = {
            limit = { has_council_position = councillor_chancellor tca_is_available_judicial_approver_trigger = yes }
            save_scope_as = tca_approver
        }
    }
    else = { save_scope_as = tca_approver }
}

# Character scope. Resolve the target's single circuit from the bounded registry.
# The recipient has already passed the actor's narrower jurisdiction check.
tca_find_arrest_circuit_effect = {
    every_in_global_list = {
        variable = tca_judicial_circuits
        limit = {
            tca_is_judicial_circuit_trigger = yes
            NOT = { has_variable = tca_judicial_disabled }
            scope:recipient = { tca_is_judicial_target_in_circuit_trigger = { CIRCUIT = prev } }
        }
        save_scope_as = tca_circuit
    }
}

# Internal context is reconstructed; do not put backing officials in UI participant slots.
tca_setup_judicial_arrest_roles_effect = {
    clear_saved_scope = tca_circuit
    if = {
        limit = {
            scope:actor = { OR = { tca_is_judicial_governor_trigger = yes tca_is_judicial_commissioner_trigger = yes } }
            tca_judicial_actor_has_jurisdiction_trigger = yes
        }
        scope:actor = {
            save_scope_as = tca_governor
            if = {
                limit = { tca_is_judicial_commissioner_trigger = yes }
                primary_title.var:song_judicial_circuit = { save_scope_as = tca_circuit }
                scope:tca_circuit.holder = { save_scope_as = tca_governor }
                # A native refusal war must target the rebel's actual liege.
                if = { limit = { scope:recipient = { is_ruler = yes exists = liege } } scope:recipient.liege = { save_scope_as = tca_governor } }
                else_if = { limit = { NOT = { scope:tca_governor = { tca_basic_allowed_to_arrest_character_trigger = { CHARACTER = scope:recipient } } } } top_liege = { save_scope_as = tca_governor } }
            }
            else = { tca_find_arrest_circuit_effect = yes }
        }
    }
}

tca_setup_judicial_approval_effect = {
    clear_saved_scope = tca_approver
    if = {
        limit = { exists = scope:tca_circuit }
        scope:tca_circuit = { save_temporary_scope_as = tca_checked_circuit }
        if = {
            limit = { tca_judicial_requires_central_trigger = yes }
            scope:actor.top_liege = { tca_resolve_central_judicial_authority_effect = yes }
        }
        else = { scope:tca_circuit.var:song_judicial_title.holder = { save_scope_as = tca_approver } }
        scope:tca_approver ?= { save_scope_as = intermediary }
    }
}
