# LEVEL 是目标档位；LAW 是原版对应任命法。确认落档才执行原版锁约与补法。
# scope:province_change_recipient 为受益人，scope:petitioner 为请愿人。
tca_province_contract_apply_petition_effect = {
    if = {
        limit = {
            NOT = { scope:province_change_recipient ?= { tca_province_contract_petition_recipient_trigger = yes } }
        }
        custom_tooltip = $TOOLTIP$
    }
    else = {
        scope:province_change_recipient = {
            tca_road_contract_apply_effect = { LEVEL = $LEVEL$ }
            if = {
                limit = { scope:tca_road_contract_changed = yes }
                hidden_effect = {
                    set_vassal_contract_modification_blocked = yes
                    if = {
                        limit = { NOT = { has_realm_law = $LAW$ } }
                        add_realm_law_skip_effects = $LAW$
                    }
                }
            }
        }
    }
}
