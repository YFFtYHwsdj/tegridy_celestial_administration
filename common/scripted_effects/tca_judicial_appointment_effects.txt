# 角色作用域；粗筛统治者及宫廷，不使用某个官署的拟任直属上级条件。
tca_collect_judicial_candidates_effect = {
    if = { limit = { is_alive = yes } add_to_list = tca_initial_judicial_candidates }
    every_courtier = { limit = { is_alive = yes } add_to_list = tca_initial_judicial_candidates }
}
# 任意作用域；清空本事件链临时候选池及政权归属，不写跨日持久状态。
tca_clear_judicial_candidate_pool_effect = {
    every_in_list = { list = tca_initial_judicial_candidates remove_from_list = tca_initial_judicial_candidates }
    clear_saved_scope = tca_judicial_pool_sovereign
    # 显式零值遮住继承事件的旧ready，不能clear后再以同君主引用判断池有效。
    save_scope_value_as = { name = tca_judicial_pool_ready value = 0 }
}
# 调用者提供当前 tca_appointment_liege；同政权同批只收集一次。
# 换政权必须重建；每职抽选仍使用实时硬资格和本职原生评分。
tca_prepare_judicial_candidate_pool_effect = {
    if = {
        limit = {
            NOT = {
                # NOT中的并列条件逐个否定；必须把“已就绪且同政权”显式合取后再取反。
                AND = {
                    exists = scope:tca_judicial_pool_ready
                    scope:tca_judicial_pool_ready ?= 1
                    scope:tca_judicial_pool_sovereign ?= scope:tca_appointment_liege.top_liege
                }
            }
        }
        tca_clear_judicial_candidate_pool_effect = yes
        scope:tca_appointment_liege.top_liege = {
            save_scope_as = tca_judicial_pool_sovereign
            tca_collect_judicial_candidates_effect = yes
            every_vassal_or_below = { tca_collect_judicial_candidates_effect = yes }
        }
        save_scope_value_as = { name = tca_judicial_pool_ready value = 1 }
    }
}

# 路头衔作用域；输入appointment_liege/title，输出candidate显式为本次命中者，
# 零合格候选时输出不可能通过无官资格的最高君主。不得用clear/exists读继承旧人。
tca_select_judicial_candidate_effect = {
    scope:tca_appointment_liege.top_liege = { save_scope_as = tca_appointment_candidate }
    tca_prepare_judicial_candidate_pool_effect = yes
    random_in_list = {
        list = tca_initial_judicial_candidates
        limit = { tca_is_initial_judicial_candidate_trigger = yes }
        weight = { base = 0 modifier = { add = tca_judicial_candidate_weight_value } }
        save_scope_as = tca_appointment_candidate
    }
}

# 路头衔作用域；保留合格在任者，修复空缺和无效持有人。
tca_fill_initial_judicial_holder_in_batch_effect = {
    if = {
        limit = {
            tca_is_judicial_circuit_trigger = yes
            NOT = { has_variable = tca_judicial_disabled }
            tca_has_judicial_title_trigger = yes
            var:song_judicial_title = { NOT = { tca_is_valid_judicial_holder_trigger = yes } }
        }
        # 仅转移官职，离任者保留其土地与家门。
        var:song_judicial_title = {
            if = {
                limit = { exists = holder }
                debug_log = "TCA: judicial holder failed eligibility; appointing replacement"
            }
        }
        tca_resolve_judicial_office_liege_effect = yes
        scope:tca_expected_judicial_liege = { save_scope_as = tca_appointment_liege }
        var:song_judicial_title = { save_scope_as = tca_appointment_title }
        # 引擎不允许为空缺官职读取候选分；临时挂靠上级持有，安装本职法后再抽取。
        create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_judicial_caretaker_change }
        scope:tca_appointment_title = { change_title_holder = { holder = scope:tca_appointment_liege change = scope:tca_judicial_caretaker_change } }
        resolve_title_and_vassal_change = scope:tca_judicial_caretaker_change
        scope:tca_appointment_title = { tca_ensure_official_title_succession_effect = yes }
        tca_select_judicial_candidate_effect = yes
        # 仅当现有候选均不合格时，生成原生学者。
        if = {
            limit = { scope:tca_appointment_candidate = scope:tca_appointment_liege.top_liege }
            scope:tca_appointment_liege = {
                create_character = {
                    template = tgp_minister_scholar_template
                    employer = scope:tca_appointment_liege
                    faith = scope:tca_appointment_liege.faith
                    rite = scope:tca_appointment_liege.rite
                    culture = scope:tca_appointment_liege.culture
                    gender_female_chance = {
                        value = 0
                        if = {
                            limit = {
                                scope:tca_appointment_liege = {
                                    OR = {
                                        has_realm_law = female_only_law
                                        has_realm_law = female_preference_law
                                    }
                                }
                            }
                            add = 100
                        }
                    }
                    save_scope_as = tca_appointment_candidate
                }
            }
            if = { limit = { NOT = { scope:tca_appointment_candidate = scope:tca_appointment_liege.top_liege } }
                scope:tca_appointment_candidate = { tca_initialize_new_judicial_scholar_effect = yes }
            }
        }
        if = {
            limit = { NOT = { scope:tca_appointment_candidate = scope:tca_appointment_liege.top_liege } }
            scope:tca_appointment_candidate = {
                # 非统治者补齐家族，并仅在首次选择命中10%时建立世族。
                if = {
                    limit = { is_ruler = no }
                    if = {
                        limit = { is_lowborn = yes }
                        create_dynasty = { spread_to_descendants = yes }
                    }
                    tca_prepare_official_family_effect = { GOVERNMENT_GIVER = scope:tca_appointment_liege }
                }
                tca_grant_official_title_effect = { TITLE = scope:tca_appointment_title LIEGE = scope:tca_appointment_liege }
                # 仅成功授职后消费候选；失败且仍无官者留待后续官署重新校验。
                if = { limit = { any_held_title = { is_noble_family_title = no } } remove_from_list = tca_initial_judicial_candidates }
            }
            scope:tca_appointment_title = {
                tca_sync_judicial_holder_liege_effect = yes
                holder ?= { tca_ensure_native_judicial_realm_succession_effect = yes }
                holder ?= { tca_ensure_official_contract_effect = yes }
                # 固定官职已在 landed_titles 中声明 delete_on_destroy = no。
                if = {
                    limit = { NOT = { has_variable = tca_judicial_static_title } }
                    set_delete_on_destroy = no
                }
                set_destroy_on_succession = no
                tca_restore_judicial_title_identity_effect = yes
            }
        }
        scope:tca_appointment_liege.top_liege = { save_scope_as = tca_appointment_candidate }
    }
}

# 旧存档调试入口，与正常补缺使用完全相同的算法。
tca_test_judicial_initial_appointment_effect = {
    tca_ensure_judicial_title_effect = yes
    tca_show_judicial_title_effect = yes
}

# 角色作用域；tca_circuit 由头衔维护建立。
# 兼容未记录家族策略的旧官员；已有选择不得在维护时重抽或补建世族。
tca_ensure_judicial_holder_family_effect = {
    if = {
        limit = {
            government_has_flag = government_is_celestial
            NOT = { any_held_title = { is_noble_family_title = yes } }
            exists = scope:tca_circuit.holder
        }
        if = {
            limit = { is_lowborn = yes }
            create_dynasty = { spread_to_descendants = yes }
        }
        tca_prepare_official_family_effect = { GOVERNMENT_GIVER = scope:tca_circuit.holder }
    }
}

# 提刑头衔作用域；绑定当前路持有人，绝不沿用保存的皇帝引用。
# 在授予头衔、政体和世族家门初始化完成后运行。
tca_sync_judicial_holder_liege_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = holder
            exists = var:song_judicial_circuit
            var:song_judicial_circuit = {
                tca_is_judicial_circuit_trigger = yes
                NOT = { has_variable = tca_judicial_disabled }
            }
        }
        var:song_judicial_circuit = { tca_resolve_judicial_office_liege_effect = yes }
        holder = {
            if = {
                limit = {
                    is_alive = yes
                    government_has_flag = government_is_celestial
                    highest_held_title_tier < scope:tca_expected_judicial_liege.highest_held_title_tier
                    OR = {
                        NOT = { exists = liege }
                        NOT = { liege ?= scope:tca_expected_judicial_liege }
                    }
                }
                create_title_and_vassal_change = {
                    type = swear_fealty
                    save_scope_as = tca_judicial_liege_change
                    add_claim_on_loss = no
                }
                change_liege = {
                    liege = scope:tca_expected_judicial_liege
                    change = scope:tca_judicial_liege_change
                }
                resolve_title_and_vassal_change = scope:tca_judicial_liege_change
            }
        }
    }
}

# 路头衔作用域；帝直属路归属该皇帝内阁中现任刑部官员，
# 刑部空缺时以皇帝兜底。
tca_resolve_judicial_office_liege_effect = {
    holder = { save_scope_as = tca_expected_judicial_liege }
    if = {
        limit = { holder = { is_independent_ruler = yes } }
        holder = {
            random_councillor = {
                limit = {
                    has_council_position = minister_justice
                    is_alive = yes
                    is_imprisoned = no
                    government_has_flag = government_is_celestial
                    highest_held_title_tier > tier_duchy
                }
                save_scope_as = tca_expected_judicial_liege
            }
        }
    }
}

# 皇帝角色作用域；监禁回调中部职状态已稳定，
# 此时重新挂接既有官员，不在监禁事务内创建头衔或
# 任命候选；完整维护仍延迟执行。
tca_refresh_imperial_judicial_lieges_effect = {
    every_held_title = {
        limit = {
            tca_is_judicial_circuit_trigger = yes
            NOT = { has_variable = tca_judicial_disabled }
            tca_has_judicial_title_trigger = yes
        }
        var:song_judicial_title = { tca_sync_judicial_holder_liege_effect = yes }
    }
}

# 头衔作用域；在引擎完成头衔／封臣事务后执行。
tca_refresh_judicial_position_in_batch_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = var:song_judicial_circuit
            var:song_judicial_circuit = {
                tca_is_judicial_circuit_trigger = yes
                NOT = { has_variable = tca_judicial_disabled }
            }
        }
        var:song_judicial_circuit = {
            save_scope_as = tca_circuit
            tca_fill_initial_judicial_holder_in_batch_effect = yes
        }
        holder ?= {
            primary_title = { set_variable = { name = tca_official_family_policy_office value = yes } }
            tca_ensure_judicial_holder_family_effect = yes
        }
        tca_sync_judicial_holder_liege_effect = yes
        holder ?= { tca_ensure_native_judicial_realm_succession_effect = yes }
        holder ?= { tca_ensure_official_contract_effect = yes }
        tca_restore_judicial_title_identity_effect = yes
        tca_reconcile_judicial_prisoners_effect = yes
    }
    else_if = {
        limit = { has_variable = tca_judicial_title exists = var:song_judicial_circuit }
        var:song_judicial_circuit = { tca_destroy_judicial_title_effect = yes }
    }
}

# 角色作用域；用于延迟受封／改制回调及季度维护。
tca_refresh_held_judicial_positions_in_batch_effect = {
    every_held_title = {
        limit = { has_variable = tca_judicial_title }
        tca_refresh_judicial_position_in_batch_effect = yes
    }
}

# 角色作用域；保留原生文官领地法律作为持有人的默认继承法。
# 官职另装本职头衔法以选择评分类型，家门头衔保留自身法律。
tca_ensure_native_judicial_realm_succession_effect = {
    if = {
        limit = {
            primary_title = { has_variable = tca_judicial_title }
            government_has_flag = government_is_celestial
            is_independent_ruler = no
            NOT = { has_realm_law = celestial_appointment_succession_law }
        }
        add_realm_law_skip_effects = celestial_appointment_succession_law
    }
}
