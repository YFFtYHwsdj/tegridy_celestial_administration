# Character scope: collect this ruler and their court into a local pool.
tca_collect_judicial_candidates_effect = {
    if = {
        limit = { tca_is_initial_judicial_candidate_trigger = yes }
        add_to_list = tca_initial_judicial_candidates
    }
    every_courtier = {
        limit = { tca_is_initial_judicial_candidate_trigger = yes }
        add_to_list = tca_initial_judicial_candidates
    }
}

# Circuit title scope. Keep eligible officers; repair vacancies and invalid holders.
tca_fill_initial_judicial_holder_effect = {
    if = {
        limit = {
            tca_is_judicial_circuit_trigger = yes
            tca_has_judicial_title_trigger = yes
            var:song_judicial_title = { NOT = { tca_is_valid_judicial_holder_trigger = yes } }
        }
        # Transfer only the office; the outgoing holder keeps land and family.
        var:song_judicial_title = {
            if = {
                limit = { exists = holder }
                debug_log = "TCA: judicial holder failed eligibility; appointing replacement"
            }
        }
        holder = { save_scope_as = tca_appointment_liege }
        var:song_judicial_title = { save_scope_as = tca_appointment_title }
        clear_saved_scope = tca_appointment_candidate
        every_in_list = {
            list = tca_initial_judicial_candidates
            remove_from_list = tca_initial_judicial_candidates
        }
        scope:tca_appointment_liege.top_liege = {
            tca_collect_judicial_candidates_effect = yes
            every_vassal_or_below = { tca_collect_judicial_candidates_effect = yes }
        }
        random_in_list = {
            list = tca_initial_judicial_candidates
            limit = { tca_is_initial_judicial_candidate_trigger = yes }
            weight = {
                base = 1
                modifier = { add = learning }
            }
            save_scope_as = tca_appointment_candidate
        }
        # Generate a native scholar only when no existing candidate qualifies.
        if = {
            limit = { NOT = { exists = scope:tca_appointment_candidate } }
            scope:tca_appointment_liege = {
                create_character = {
                    template = tgp_minister_scholar_template
                    employer = scope:tca_appointment_liege
                    faith = scope:tca_appointment_liege.faith
                    rite = scope:tca_appointment_liege.rite
                    culture = scope:tca_appointment_liege.culture
                    gender_female_chance = {
                        value = 0
                        if = {
                            limit = {
                                scope:tca_appointment_liege = {
                                    OR = {
                                        has_realm_law = female_only_law
                                        has_realm_law = female_preference_law
                                    }
                                }
                            }
                            add = 100
                        }
                    }
                    save_scope_as = tca_appointment_candidate
                }
            }
            scope:tca_appointment_candidate = {
                set_variable = { name = tca_generated_initial_judicial_holder value = yes }
                change_merit = duchy_starting_merit_value
                assign_completed_exams_based_on_merit_effect = yes
            }
        }
        if = {
            limit = { exists = scope:tca_appointment_candidate }
            scope:tca_appointment_candidate = {
                # Non-rulers need the native noble-family setup for celestial careers.
                if = {
                    limit = { is_ruler = no }
                    if = {
                        limit = { is_lowborn = yes }
                        create_dynasty = { spread_to_descendants = yes }
                    }
                    create_noble_family_effect = { GOVERNMENT_GIVER = scope:tca_appointment_liege }
                }
                create_title_and_vassal_change = {
                    type = appointment
                    save_scope_as = tca_appointment_change
                    add_claim_on_loss = no
                }
                scope:tca_appointment_title = {
                    change_title_holder = {
                        holder = scope:tca_appointment_candidate
                        change = scope:tca_appointment_change
                    }
                }
                resolve_title_and_vassal_change = scope:tca_appointment_change
                change_government = celestial_government
            }
            scope:tca_appointment_title = {
                tca_sync_judicial_holder_liege_effect = yes
                holder ?= { tca_ensure_native_judicial_realm_succession_effect = yes }
                holder ?= { tca_ensure_official_contract_effect = yes }
                set_delete_on_destroy = no
                set_destroy_on_succession = no
                tca_restore_judicial_title_identity_effect = yes
            }
        }
        every_in_list = {
            list = tca_initial_judicial_candidates
            remove_from_list = tca_initial_judicial_candidates
        }
        clear_saved_scope = tca_appointment_candidate
    }
}

# Debug entry for existing saves; uses exactly the normal fill algorithm.
tca_test_judicial_initial_appointment_effect = {
    tca_ensure_judicial_title_effect = yes
    tca_show_judicial_title_effect = yes
}

# Character scope; tca_circuit is established by title maintenance.
# Repair previously appointed lowborn holders whose family setup failed.
tca_ensure_judicial_holder_family_effect = {
    if = {
        limit = {
            government_has_flag = government_is_celestial
            NOT = { any_held_title = { is_noble_family_title = yes } }
            exists = scope:tca_circuit.holder
        }
        if = {
            limit = { is_lowborn = yes }
            create_dynasty = { spread_to_descendants = yes }
        }
        create_noble_family_effect = { GOVERNMENT_GIVER = scope:tca_circuit.holder }
    }
}

# Judicial title scope. Bind to the CURRENT circuit holder, never a saved emperor.
# Run after title grant, government and noble-family setup have resolved.
tca_sync_judicial_holder_liege_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = holder
            exists = var:song_judicial_circuit
            var:song_judicial_circuit = { tca_is_judicial_circuit_trigger = yes }
        }
        var:song_judicial_circuit.holder = { save_scope_as = tca_expected_judicial_liege }
        holder = {
            if = {
                limit = {
                    is_alive = yes
                    government_has_flag = government_is_celestial
                    highest_held_title_tier < scope:tca_expected_judicial_liege.highest_held_title_tier
                    OR = {
                        NOT = { exists = liege }
                        NOT = { liege ?= scope:tca_expected_judicial_liege }
                    }
                }
                create_title_and_vassal_change = {
                    type = swear_fealty
                    save_scope_as = tca_judicial_liege_change
                    add_claim_on_loss = no
                }
                change_liege = {
                    liege = scope:tca_expected_judicial_liege
                    change = scope:tca_judicial_liege_change
                }
                resolve_title_and_vassal_change = scope:tca_judicial_liege_change
            }
        }
    }
}

# Title scope, after the engine has finished title/vassal transactions.
tca_refresh_judicial_position_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = var:song_judicial_circuit
            var:song_judicial_circuit = { tca_is_judicial_circuit_trigger = yes }
        }
        var:song_judicial_circuit = {
            save_scope_as = tca_circuit
            tca_fill_initial_judicial_holder_effect = yes
        }
        holder ?= { tca_ensure_judicial_holder_family_effect = yes }
        tca_sync_judicial_holder_liege_effect = yes
        holder ?= { tca_ensure_native_judicial_realm_succession_effect = yes }
        holder ?= { tca_ensure_official_contract_effect = yes }
        tca_restore_judicial_title_identity_effect = yes
        tca_reconcile_judicial_prisoners_effect = yes
    }
}

# Character scope; delayed gain/government hooks and quarterly maintenance.
tca_refresh_held_judicial_positions_effect = {
    every_held_title = {
        limit = { has_variable = tca_judicial_title }
        tca_refresh_judicial_position_effect = yes
    }
}

# Character scope: keep the native civilian realm law as the holder default.
# The office inherits this native realm law; family titles keep their own laws.
tca_ensure_native_judicial_realm_succession_effect = {
    if = {
        limit = {
            primary_title = { has_variable = tca_judicial_title }
            government_has_flag = government_is_celestial
            is_independent_ruler = no
            NOT = { has_realm_law = celestial_appointment_succession_law }
        }
        add_realm_law_skip_effects = celestial_appointment_succession_law
    }
}
