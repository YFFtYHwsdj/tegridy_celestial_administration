# 司法拘捕原生改编效果；正式回复入口先恢复上下文并通过 still_valid 检查，再调用接受／拒绝分支。
# 原版来源入口为 imprison_interaction 的 on_accept／on_decline 与 war_for_imprison_effect；差异与验收见 docs/judicial/arrest.md。
# 改编自原版 on_accept／on_decline 与 war_for_imprison_effect。
# 这些主体仅在通过 tca_judicial_arrest_still_valid_trigger 后执行。
# tca_authority 是实际发起者，tca_governor 提供军事支持。
tca_judicial_arrest_success_native_effect = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				if = {
					limit = { liege_diarchy_surcharge_interaction_imprison_interaction_value > 0 }
					add_prestige = { value = { add = liege_diarchy_surcharge_interaction_imprison_interaction_value max = prestige multiply = -1 } }
				}
				if = {
					limit = {
						has_title = title:e_minister_of_justice
						scope:recipient = {
							NOT = { is_courtier_of = scope:puppet_or_actor }
						}
					}
					remove_treasury = scope:puppet_or_actor.major_treasury_value
				}
			}
		}
		scope:puppet_or_actor = {
			if = {
				limit = {
					has_title = title:e_minister_of_justice
				}
				if = {
					limit = {
						NOR = {
							has_imprisonment_reason = scope:recipient
							scope:recipient.liege = {
								has_imprisonment_reason = scope:recipient
							}
							scope:hook ?= yes
						}
						any_character_situation = {
							situation_type = dynastic_cycle
							situation_top_has_catalyst = catalyst_minister_imprison
						}
					}
					save_scope_value_as = {
						name = use_strife_not_tyranny
						value = yes
					}
					situation:dynastic_cycle = {
						trigger_situation_catalyst = {
							catalyst = catalyst_minister_imprison
							character = scope:recipient
						}
					}
				}
				if = {
					limit = {
						government_allows = merit
					}
					change_merit = {
						value = minor_merit_gain
						multiply = scope:recipient.highest_held_title_tier
					}
				}
				if = {
					limit = {
						government_has_flag = government_has_influence
					}
					change_influence = {
						value = minor_influence_gain
						multiply = scope:recipient.highest_held_title_tier
					}
				}
				scope:recipient.liege = {
					if = {
						limit = { has_imprisonment_reason = scope:recipient }
						consume_imprisonment_reasons = scope:recipient
					}
				}
			}
		}
		if = {
			limit = { scope:hook ?= yes }
			scope:puppet_or_actor = { use_hook = scope:recipient }
		}
		illegal_imprisonment_dread_effect = yes
		imprison_court_chaplain_consequences_effect = yes
		imprison_HoF_consequences_effect = yes
		scope:recipient = {
			if = {
				limit = {
					is_a_faction_member = yes
					liege = scope:puppet_or_actor
				}
				joined_faction = { save_scope_as = retaliating_faction }
			}
		}
		# 暴政由发起者承担。
        # 批准不改变羁押归属，实际发起者负责关押。
        scope:puppet_or_actor = { imprison_tyranny_effect = yes save_scope_as = tca_custodian }
        save_scope_value_as = { name = disable_display_rightful_imprisonment_tooltip value = yes }
        rightfully_imprison_character_effect = {
            TARGET = scope:recipient
            IMPRISONER = scope:tca_custodian
        }
        if = {
            limit = { scope:puppet_or_actor = { tca_is_judicial_commissioner_trigger = yes } }
            scope:recipient = { tca_register_judicial_prisoner_effect = yes }
        }
		# 若对方近期撤销过你的头衔或请求绝罚你，则升级为仇敌。
		imprison_attempt_escalates_rivalry_effect = yes
		scope:puppet_or_actor = {
			# 原生事件预览发起者收押，因此下方使用反映实际羁押者的弹出通知。
			if = {
				limit = {
					has_imprisonment_reason = scope:recipient
				}
				stress_and_fulfillment_impact = {
					compassionate = minor_stress_impact_gain
					forgiving = minor_stress_impact_gain
				}
			}
			else = {
				stress_and_fulfillment_impact = {
					just = medium_stress_impact_gain
					compassionate = medium_stress_impact_gain
				}
				if = {
					limit = { exists = scope:retaliating_faction}
					scope:retaliating_faction = {
						imprisonment_retribution_start_faction_war_if_valid = yes
					}
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
			scope:puppet_or_actor = { remove_character_flag = flag_hostile_actions_disabled_delay }
		}

		# 氏族政体下，此互动影响团结度。
		if = { # 无拘捕理由且目标有地时，团结度损失较大。
			limit = {
				NOT = { scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient } }
				scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
			}
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = major_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}
		else_if = { # 无拘捕理由且目标无地时，团结度损失稍小。
			limit = {
				NOT = { scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient } }
				scope:recipient = { is_landed_or_landless_administrative_or_theocratic = no }
			}
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = medium_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}
		else = { # 有正当理由时仅损失少量团结度：无论是否犯罪，对方仍是家人。
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = minor_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}

		# 劫掠处理逻辑。
		scope:puppet_or_actor = {
			if = {
				limit = {
					character_is_valid_for_harrying_of_the_north_trigger = yes
					scope:recipient = {
						character_is_valid_for_harrying_of_the_north_trigger = yes
						is_important_or_vip_struggle_character = yes
					}
					culture != scope:recipient.culture
				}
				global_var:harrying_of_the_north = { save_scope_as = story }
				if = {
					limit = {
						character_is_valid_norman_for_harrying_of_the_north_trigger = yes
						scope:recipient = { character_is_valid_anglo_saxon_for_harrying_of_the_north_trigger = yes }
					}
					ep3_increase_pacification_effect = { AMOUNT = harrying_minor_increase_value }
				}
				else_if = {
					limit = {
						character_is_valid_anglo_saxon_for_harrying_of_the_north_trigger = yes
						scope:recipient = { character_is_valid_norman_for_harrying_of_the_north_trigger = yes }
					}
					ep3_increase_resistance_effect = { AMOUNT = harrying_minor_increase_value }
				}
			}
		}

    scope:puppet_or_actor = {
        send_interface_toast = { type = event_toast_effect_good title = tca_judicial_arrest_success left_icon = scope:recipient right_icon = scope:tca_custodian }
    }
}

# 目标拒捕分支；沿用原生拒捕后果及战争处理，军事支持使用恢复后的 tca_governor。
tca_judicial_arrest_decline_native_effect = {
		#至亲可能获得仇敌关系进度。
		if = {
			limit = {
				scope:recipient = {
					is_close_family_of = scope:puppet_or_actor
				}
				#合法拘捕。
				trigger_if = {
					limit = {
						scope:puppet_or_actor = {
							has_imprisonment_reason = scope:recipient
						}
					}
					scope:recipient = {
						NOR = {
							has_trait = forgiving
							ai_honor >= low_positive_ai_value
						}
					}
				}
				#非法拘捕。
				trigger_else = {
					scope:recipient = {
						NOT = {
							has_trait = forgiving
						}
					}
				}
			}
			scope:recipient = {
				progress_towards_rival_effect = {
					CHARACTER = scope:puppet_or_actor
					REASON = rival_imprisoned_family
					OPINION = 0
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = {
					NOT = { has_title = title:e_minister_of_justice }
				}
			}
			if = {
				limit = {
					scope:recipient = {
						has_government = landless_adventurer_government
					}
				}
				scope:recipient = {
					if = {
						limit = { current_military_strength >= ep3_interactions_events_0500_troops_count_value }
						start_war = {
							cb = ep3_laamp_apprehend_adventurer_cb
							target = scope:puppet_or_actor
						}
					}
					else = {
						# 立即移除。
						remove_laamp_from_exiled_county_automatically_effect = {
							PROVINCE = scope:recipient.location
							LAAMP = scope:recipient
						}
					}
				}
			}
			else = {
				scope:puppet_or_actor = {
					imprison_HoF_consequences_effect = yes
					imprison_tyranny_effect = yes
				}
				scope:recipient = {
					#目标为伯爵或更高等级时，发动战争。
					if = {
						limit = {
							is_playable_character = yes
							primary_title.tier > tier_barony
						}
						if = {
							limit = {
								is_a_faction_member = yes
                                scope:tca_authority = scope:tca_governor
								liege = scope:tca_governor
								joined_faction = {
									faction_target = scope:tca_governor
									NOT = { faction_is_type = populist_faction }
								}
							}
							joined_faction = {
								save_scope_as = retaliating_faction # 供本地化使用。
								imprisonment_retribution_start_faction_war_if_valid = yes
							}
							hidden_effect = {
								if = {
									limit = { NOT = { exists = joined_faction.faction_war } }
									tca_judicial_war_for_imprison_effect = {
										LIEGE = scope:tca_governor
										REBEL = scope:recipient
									}
								}
							}
						}
						else = {
							tca_judicial_war_for_imprison_effect = {
								LIEGE = scope:tca_governor
								REBEL = scope:recipient
							}
						}
					}
					#目标为男爵或无地角色时，使其逃走。
					else = {
						if = { #防止其立即被重新雇用。
							limit = {
								is_councillor_of = scope:puppet_or_actor
								can_be_fired_from_council_trigger = { COURT_OWNER = scope:puppet_or_actor }
							}
							set_variable = {
								name = escaped_imprisonment_from
								value = scope:puppet_or_actor
								years = 20
							}
						}
						imprison_court_chaplain_consequences_effect = yes
						if = {
							limit = {
								is_landed_or_landless_administrative_or_theocratic = yes
							}
							depose_effect = { DEPOSER = scope:puppet_or_actor }
						}
						else_if  = {
							limit = {
								is_hostage = yes
								warden = scope:puppet_or_actor
							}
							save_scope_as = imprison_failure
							save_scope_as = hostage
							warden = { save_scope_as = warden }
							home_court = { save_scope_as = home_court }
							scope:warden = { trigger_event = bp2_hostage_system.0400 }
							scope:home_court = { trigger_event = bp2_hostage_system.0401 }
							return_hostage = yes
						}
						else_if = {
							limit = {
								is_landed_or_landless_administrative_or_theocratic = no
								trigger_if = {
									limit = {
										is_councillor = yes
									}
									can_be_fired_from_council_trigger = { COURT_OWNER = scope:recipient.liege }
								}
								trigger_if = {
									limit = {
										exists = scope:puppet_or_actor.faith.religious_head
									}
									scope:puppet_or_actor.faith.religious_head != scope:recipient
								}
							}
							select_and_move_to_pool_effect = yes
						}
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = attempted_imprisonment_opinion
						}
					}
				}

				if = {
					limit = {
						scope:puppet_or_actor = { has_realm_law_flag = vassal_refusal_is_treason }
					}
					scope:puppet_or_actor = {
						add_opinion = {
							target = scope:recipient
							modifier = treasonous_imprison_refusal
						}
					}
				}
			}

			#解除其与所有廷臣的监护关系；反馈由 char_interaction.0230 提供。
			scope:puppet_or_actor = {
				if = {
					limit = {
						any_courtier = {
							has_relation_guardian = scope:recipient
						}
					}
					every_courtier = {
						limit = {
							has_relation_guardian = scope:recipient
						}
						save_scope_as = ward
						scope:recipient = { remove_relation_ward = scope:ward }

						scope:ward = {
							if = {
								limit = {
									host != liege
								}
								return_to_court = yes
							}
							guardian_remove_opinion_effect = {
								GUARDIAN = scope:recipient
								WARD = this
								WARD_LIEGE = scope:puppet_or_actor
							}
						}

						add_to_list = ward_children
					}
				}
				if = {
					limit = {
						any_courtier = {
							has_relation_ward = scope:recipient
						}
					}
					random_courtier = {
						limit = {
							has_relation_ward = scope:recipient
						}
						save_scope_as = guardian
						scope:guardian = { remove_relation_ward = scope:recipient }

						scope:guardian = {
							if = {
								limit = {
									host != liege
								}
								return_to_court = yes
							}
							guardian_remove_opinion_effect = {
								GUARDIAN = this
								WARD = scope:recipient
								WARD_LIEGE = scope:puppet_or_actor
							}
						}

						add_to_list = guardian_list
					}

					hidden_effect = {
						scope:guardian = {
							send_interface_message = {
								type = event_childhood_neutral
								title = remove_guardian_interaction_notification
								left_icon = scope:recipient
								right_icon = scope:guardian
								remove_guardian_effect = {
									GUARDIAN = scope:guardian
									WARD = scope:recipient
									RETURN_WARD = yes
									HIDE_OPINION = no
								}
							}
						}
					}
				}
				if = {
					limit = {
						scope:recipient = {
							OR = {
								is_playable_character = no
								AND = {
									is_playable_character = yes
									primary_title.tier = tier_barony
								}
							}
						}
					}
					trigger_event = char_interaction.0230
				}

				if = {
					limit = {
						has_character_flag = flag_hostile_actions_disabled_delay
					}
					remove_character_flag = flag_hostile_actions_disabled_delay
				}
			}

			# 氏族政体下，此互动影响团结度。
			if = { # 无拘捕理由且目标有地时，团结度损失较大。
				limit = {
					NOT = {
						scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient }
					}
					scope:recipient = { is_landed_or_landless_administrative_or_theocratic  = yes }
				}
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:puppet_or_actor
					TARGET = scope:recipient
					VALUE = major_unity_loss
					DESC = clan_unity_imprison.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
			}
			else_if = { # 无拘捕理由且目标无地时，团结度损失稍小。
				limit = {
					NOT = {
						scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient }
					}
					scope:recipient = { is_landed_or_landless_administrative_or_theocratic  = no }
				}
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:puppet_or_actor
					TARGET = scope:recipient
					VALUE = medium_unity_loss
					DESC = clan_unity_imprison.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
			}
			else = { # 有正当理由时仅损失少量团结度：无论是否犯罪，对方仍是家人。
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:puppet_or_actor
					TARGET = scope:recipient
					VALUE = miniscule_unity_loss
					DESC = clan_unity_imprison.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
			}
		}

    scope:tca_authority = { save_scope_as = actor save_scope_as = puppet_or_actor }
    scope:recipient = {
        if = {
            limit = {
                is_at_war_with = scope:tca_governor
                scope:tca_authority = { tca_is_judicial_commissioner_trigger = yes }
            }
            set_variable = { name = tca_pending_judicial_title value = scope:tca_position }
            set_variable = { name = tca_pending_judicial_governor value = scope:tca_governor }
        }
    }
}

# 沿用原生拒捕宣战理由、派系盟友及集团行为，仅调整拘捕理由持有人检查。
tca_judicial_war_for_imprison_effect = {
	$LIEGE$ = { save_temporary_scope_as = liege_temp }
	$REBEL$ = {
		save_temporary_scope_as = rebel_temp
		save_scope_value_as = {
			name = war_for_imprisonment_flavour
			value = yes
		}
		start_war = {
			casus_belli = refused_liege_demand_war
			target = scope:liege_temp
		}
	}
	# 不满的封臣。
	if = {
		limit = {
			scope:tca_authority = { NOT = { has_imprisonment_reason = scope:rebel_temp } }
		}
		scope:liege_temp = {
			every_vassal = { #加入叛乱者的封臣。
				limit = {
					vassal_would_join_rebel_faction_base_trigger = yes
					OR = {
						#情人与朋友始终支持叛乱者，除非同时也是领主的情人或朋友。
						vassal_would_join_rebel_faction_relation_trigger = yes
						#领主实施暴政时，喜欢目标或重视荣誉的封臣会响应。
						AND = {
							# 特殊拘捕检查。
							NOT = {
								scope:tca_authority = { has_imprisonment_reason = scope:rebel_temp }
							}
							# 若无法加入反领主派系，则不应起兵。
							vassal_would_join_rebel_faction_tyranny_trigger = yes
						}
						#最后，极其厌恶领主的封臣也会加入，但不能同时厌恶目标。
						AND = {
							OR = {
								opinion = {
									target = scope:rebel_temp
									value >= -25
								}
								AND = {
									has_game_rule = hard_difficulty
									scope:liege_temp = { is_ai = no }
									opinion = {
										target = scope:rebel_temp
										value >= -50
									}
								}
								AND = {
									has_game_rule = very_hard_difficulty
									scope:liege_temp = { is_ai = no }
									opinion = {
										target = scope:rebel_temp
										value >= -75
									}
								}
							}
							OR = {
								opinion = {
									target = scope:liege_temp
									value <= -40
								}
								AND = {
									has_game_rule = hard_difficulty
									scope:liege_temp = { is_ai = no }
									opinion = {
										target = scope:liege_temp
										value <= -20
									}
								}
								AND = {
									has_game_rule = very_hard_difficulty
									scope:liege_temp = { is_ai = no }
									opinion = {
										target = scope:liege_temp
										value <= 0
									}
								}
							}
						}
					}
					# 天朝扩展：集团成员跟随领袖。
					trigger_if = {
						limit = { exists = house.house_confederation.leading_house.house_head }
						house.house_confederation.leading_house.house_head = this
					}
				}
				add_to_list = joining_vassals
			}
		}
	}
	# 即使拘捕合法，集团成员也会加入。
	bloc_member_join_war_effect = yes
	every_in_list = {
		list = joining_vassals
		vassal_join_rebel_war_effect = yes
	}
	every_player = {
		limit = { is_vassal_of = scope:liege_temp }
		scope:liege_temp = {
			save_scope_as = actor
			save_scope_as = puppet_or_actor
		}
		scope:rebel_temp = { save_scope_as = recipient }
		trigger_event = char_interaction.0232
	}
	scope:rebel_temp = {
		add_opinion = {
			target = scope:liege_temp
			modifier = attempted_arrest_opinion
		}
	}
}
