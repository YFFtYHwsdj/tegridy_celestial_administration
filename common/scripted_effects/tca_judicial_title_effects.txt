# Title scope. Create/bind, then maintain native realm succession and office identity.
tca_ensure_judicial_title_effect = {
    tca_register_historical_characters_effect = yes
    if = {
        limit = {
            tca_is_judicial_circuit_trigger = yes
            NOT = { has_variable = tca_judicial_disabled }
            NOT = { tca_has_judicial_title_trigger = yes }
        }
        save_scope_as = tca_circuit
        # Forget a missing object. A valid reverse-bound position is never replaced.
        remove_variable = song_judicial_title
        clear_saved_scope = tca_position
        var:tca_static_judicial_title ?= { save_scope_as = tca_position }
        if = {
            limit = { NOT = { exists = scope:tca_position } }
            holder = {
                create_dynamic_title = {
                    tier = duchy
                    name = tca_judicial_title_name
                    adj = tca_judicial_title_adj
                }
            }
            scope:new_title = { save_scope_as = tca_position }
        }
        scope:tca_position = {
            tca_configure_judicial_title_location_effect = yes
            set_no_automatic_claims = yes
            set_can_be_named_after_dynasty = no
            set_can_use_nomadic_naming = no
            # Fixed offices already declare delete_on_destroy = no in landed_titles.
            if = {
                limit = { NOT = { has_variable = tca_judicial_static_title } }
                set_delete_on_destroy = no
            }
            set_destroy_on_succession = no
            set_variable = { name = tca_judicial_title value = yes }
            set_variable = { name = song_judicial_circuit value = scope:tca_circuit }
            tca_sync_judicial_title_de_jure_effect = yes
            set_coa = scope:tca_circuit
            # No de-jure counties; initial appointment follows after binding.
            # Explicitly disable destruction on succession; this is a persistent office.
        }
        set_variable = { name = song_judicial_title value = scope:tca_position }
        if = {
            limit = { NOT = { has_variable = tca_judicial_registered } }
            add_to_global_variable_list = { name = tca_judicial_circuits target = this }
            set_variable = { name = tca_judicial_registered value = yes }
        }
    }
    # Refresh previously generated titles and follow subsequent circuit renames.
    if = {
        limit = {
            tca_is_judicial_circuit_trigger = yes
            NOT = { has_variable = tca_judicial_disabled }
            tca_has_judicial_title_trigger = yes
        }
        save_scope_as = tca_circuit
        var:song_judicial_title = { tca_refresh_judicial_position_effect = yes }
    }
}

# Judicial title scope, scope:tca_circuit = associated circuit.
# This is an engine location reference, not land ownership or an office building.
# Order matters: set_capital_county MUST precede set_landless_title.
tca_configure_judicial_title_location_effect = {
    if = {
        limit = { exists = scope:tca_circuit.title_capital_county }
        set_capital_county = scope:tca_circuit.title_capital_county
        set_landless_title = yes
    }
    else_if = {
        limit = { exists = scope:tca_circuit.holder.capital_county }
        set_capital_county = scope:tca_circuit.holder.capital_county
        set_landless_title = yes
    }
}

# Native de-jure setter requires a non-null title. Delete a temporary parent
# to detach the persistent office from real circuits. The engine retains the
# retired parent reference. This removes real circuit ancestry, not necessarily
# the parent pointer itself. Never delete the persistent judicial office.
tca_detach_judicial_title_de_jure_effect = {
    if = {
        limit = { exists = de_jure_liege NOT = { de_jure_liege = { has_variable = tca_judicial_retirement_parent } } }
        save_scope_as = tca_detach_position
        clear_saved_scope = tca_detach_actor
        var:song_judicial_circuit.holder.top_liege ?= {
            if = { limit = { is_alive = yes highest_held_title_tier >= tier_empire } save_scope_as = tca_detach_actor }
        }
        if = {
            limit = { NOT = { exists = scope:tca_detach_actor } }
            every_independent_ruler = {
                limit = { is_alive = yes highest_held_title_tier >= tier_empire NOT = { exists = scope:tca_detach_actor } }
                save_scope_as = tca_detach_actor
            }
        }
        if = {
            limit = { exists = scope:tca_detach_actor }
            scope:tca_detach_actor = { create_dynamic_title = { tier = kingdom name = tca_judicial_retirement_name } }
            scope:new_title = {
                save_scope_as = tca_detach_parent
                if = { limit = { exists = scope:tca_detach_position.title_capital_county } set_capital_county = scope:tca_detach_position.title_capital_county }
                else = { set_capital_county = scope:tca_detach_actor.capital_county }
                set_landless_title = yes
                set_variable = { name = tca_judicial_retirement_parent value = yes }
                set_delete_on_destroy = yes
                set_no_automatic_claims = yes
            }
            scope:tca_detach_actor = {
                create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_detach_grant }
                scope:tca_detach_parent = { change_title_holder = { holder = scope:tca_detach_actor change = scope:tca_detach_grant } }
                resolve_title_and_vassal_change = scope:tca_detach_grant
            }
            scope:tca_detach_position = { set_de_jure_liege_title = scope:tca_detach_parent }
            scope:tca_detach_parent = { set_delete_on_destroy = yes }
            scope:tca_detach_actor = { destroy_title = scope:tca_detach_parent }
        }
    }
}

# Title scope: only an active, filled institution belongs to the de-jure tree.
# Keep reverse binding data when detached, so re-establishment reuses this object.
tca_sync_judicial_title_de_jure_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = var:song_judicial_circuit
            tca_is_valid_judicial_holder_trigger = yes
            var:song_judicial_circuit = {
                tca_is_judicial_circuit_trigger = yes
                NOT = { has_variable = tca_judicial_disabled }
            }
        }
        set_de_jure_liege_title = var:song_judicial_circuit
    }
    else = { tca_detach_judicial_title_de_jure_effect = yes }
}

# Position title scope. Destruction retires custody but preserves office identity
# and both circuit pointers, but detaches real circuit ancestry. Also called
# by external on_title_destroyed events.
tca_retire_judicial_title_effect = {
    tca_retire_judicial_prisoners_effect = yes
    tca_detach_judicial_title_de_jure_effect = yes
    # Fixed offices already declare delete_on_destroy = no in landed_titles.
    if = {
        limit = { NOT = { has_variable = tca_judicial_static_title } }
        set_delete_on_destroy = no
    }
    set_destroy_on_succession = no
}

# Circuit title scope; scope:tca_lifecycle_actor must be a character.
# Keep the object and binding, withdrawing only the current appointment.
# The callback retires custody without recursively destroying this judicial office.
tca_destroy_judicial_title_effect = {
    if = {
        limit = { tca_has_judicial_title_trigger = yes }
        var:song_judicial_title = {
            save_scope_as = tca_position_to_destroy
            tca_retire_judicial_title_effect = yes
        }
        if = {
            limit = { exists = scope:tca_position_to_destroy.holder }
            scope:tca_position_to_destroy.holder = { destroy_title = scope:tca_position_to_destroy }
        }
        # Already vacant offices need no further destruction transaction.
    }
    else = {
        remove_variable = song_judicial_title
        remove_list_global_variable = { name = tca_judicial_circuits target = this }
        remove_variable = tca_judicial_registered
    }
}

# Explicit abolition keeps the yearly repair from immediately recreating the position.
# Call from circuit title scope, with a character in scope:tca_lifecycle_actor.
tca_abolish_judicial_title_effect = {
    set_variable = { name = tca_judicial_disabled value = yes }
    tca_destroy_judicial_title_effect = yes
}

tca_restore_judicial_title_effect = {
    remove_variable = tca_judicial_disabled
    tca_ensure_judicial_title_effect = yes
}

# Character scope: run after title transfers and government changes have settled.
tca_refresh_character_circuits_effect = {
    tca_ensure_court_system_law_effect = yes
    tca_refresh_official_name_flags_effect = yes
    save_scope_as = tca_lifecycle_actor
    every_held_title = {
        limit = { tier = tier_kingdom }
        if = {
            limit = { tca_is_judicial_circuit_trigger = yes }
            tca_ensure_judicial_title_effect = yes
        }
        else_if = {
            limit = { has_variable = song_judicial_title }
            tca_destroy_judicial_title_effect = yes
        }
    }
}

# Empty/global scope: bounded yearly repair, not a daily world scan.
tca_refresh_all_judicial_titles_effect = {
    every_independent_ruler = {
        tca_ensure_court_system_law_effect = yes
        tca_refresh_official_name_flags_effect = yes
        every_vassal_or_below = { tca_refresh_official_name_flags_effect = yes }
    }
    tca_initialize_static_judicial_bindings_effect = yes
    # Preserve inactive circuits in the registry so their offices remain reachable.
    # Pick one surviving ruler as the lifecycle context; this grants no position.
    every_independent_ruler = {
        limit = { NOT = { exists = scope:tca_lifecycle_actor } }
        save_scope_as = tca_lifecycle_actor
    }
    if = {
        limit = {
            exists = scope:tca_lifecycle_actor
            has_global_variable_list = tca_judicial_circuits
        }
        every_in_global_list = {
            variable = tca_judicial_circuits
            if = {
                limit = {
                    OR = {
                        NOT = { tca_is_judicial_circuit_trigger = yes }
                        has_variable = tca_judicial_disabled
                    }
                }
                tca_destroy_judicial_title_effect = yes
            }
        }
    }
    every_independent_ruler = {
        if = {
            limit = { government_has_flag = government_is_celestial }
            tca_refresh_character_circuits_effect = yes
        }
        every_vassal_or_below = {
            limit = {
                government_has_flag = government_is_celestial
                any_held_title = { tier = tier_kingdom }
            }
            tca_refresh_character_circuits_effect = yes
        }
    }
}

# Title scope. Restore identity LAST: native title/liege callbacks can rename
# titles or update CoAs. Applies equally to existing objects in older saves.
tca_restore_judicial_title_identity_effect = {
    if = {
        limit = {
            has_variable = tca_judicial_title
            exists = var:song_judicial_circuit
        }
        var:song_judicial_circuit = { save_scope_as = tca_circuit }
        tca_sync_judicial_title_de_jure_effect = yes
        tca_configure_judicial_title_location_effect = yes
        set_definitive_form = yes
        tca_set_judicial_title_name_effect = yes
        set_coa = scope:tca_circuit
        # Fixed offices already declare delete_on_destroy = no in landed_titles.
        if = {
            limit = { NOT = { has_variable = tca_judicial_static_title } }
            set_delete_on_destroy = no
        }
        set_destroy_on_succession = no
        # Old saves may retain the office-specific override. The holder's native
        # realm law is ensured by the appointment/maintenance caller before this.
        # Remove only our legacy law; family and other title laws stay untouched.
        if = {
            limit = { has_title_law = tca_judicial_appointment_succession_law }
            remove_title_law = tca_judicial_appointment_succession_law
        }
    }
}

# Character scope: coalesce gain/government/liege callbacks until the next day.
# Timed marker also recovers if a queued event cannot run; keep it during repair
# so appointments performed by that repair do not schedule themselves again.
tca_request_judicial_maintenance_effect = {
    if = {
        limit = { is_alive = yes NOT = { has_variable = tca_judicial_maintenance_pending } }
        set_variable = { name = tca_judicial_maintenance_pending value = yes days = 3 }
        trigger_event = { id = tca_judicial_titles.2 days = 1 }
    }
}

tca_run_judicial_maintenance_effect = {
    if = {
        limit = { is_alive = yes }
        set_variable = { name = tca_judicial_maintenance_pending value = yes days = 3 }
        tca_refresh_character_circuits_effect = yes
        tca_refresh_held_judicial_positions_effect = yes
        remove_variable = tca_judicial_maintenance_pending
    }
}
