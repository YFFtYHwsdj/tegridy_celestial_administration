# 原版共同提示效果的必要局部覆盖：白名单CB的胜利结算都从这里进入。
# 通用胜方回调在转地后，改隶回调只有old_liege而无战争/原因，无法精确归因同日主动改隶。
# 仅在本模组已登记战争内保存结算前快照；原版主体保持原样，未登记战争只走原版。
show_pow_release_message_effect = {
	if = {
		limit = {
			any_character_war = {
				primary_attacker = scope:attacker
				primary_defender = scope:defender
			}
		}
		every_character_war = {
			limit = {
				primary_attacker = scope:attacker
				primary_defender = scope:defender
			}
			save_scope_as = war
		}
		if = {
			limit = { prisoners_of_war_trigger = yes }
			custom_tooltip = release_pow_at_end_of_war
		}
	}
    tca_ws_prepare_resolution_effect = yes
}
