# 天朝军事差遣的季度铨选、迁转及地方军事内阁生命周期。
# 季度入口只扫描本政权的天朝官署与宫廷；按预先收集的出缺名单分配，不将本轮腾退旧职重新入队。
# 铨选与迁转委派文武共用入口；无候选时保留空缺，不生成人物。
# 地方内阁选席、模块保护及离任清理集中于本文件，玩家任务交接由独立任务效果文件维护。
# 各入口的作用域、参数及临时状态见相邻说明。

# 地方军事内阁联动；原生回调提供employee与liege，所有任命汇入同一个选席入口。
tca_military_court_position_on_received_effect = {
    tca_court_on_received_effect = yes
    scope:liege = { tca_military_sync_local_marshal_effect = yes }
}

# 主君作用域；同署可能同时雇用三种差遣，唯一军事内阁按都虞候、都监、监押排序。
# 只访问本署已有职位；本入口创建并清理tca_military_council_liege与tca_military_selected_marshal。
# 可选输入scope:tca_military_departing_officer由离任回调建立并清理，用于排除正在离任的旧官。
# 只有实际兼任者获得模块锁，既有原生保护不改写；无法合法替换外部受保护司马时不强制执行。
tca_military_sync_local_marshal_effect = {
    tca_court_sync_linked_council_effect = { COUNCIL = councillor_marshal SELECT = tca_military_select_local_marshal_effect CAPTURE = tca_military_capture_marshal_task_effect RESTORE = tca_military_restore_marshal_task_effect FALLBACK = tca_military_player_fallback_marshal_effect }
}

# 主君作用域；选席顺序不变，输出由共用同步入口消费。
tca_military_select_local_marshal_effect = {
    tca_military_select_marshal_from_position_effect = { POSITION = tca_corps_inspector_court_position }
    tca_military_select_marshal_from_position_effect = { POSITION = tca_military_inspector_court_position }
    tca_military_select_marshal_from_position_effect = { POSITION = tca_military_supervisor_court_position }
}

# 主君作用域；参数POSITION为静态职位ID，必需scope:tca_military_council_liege由选席入口提供。
# 仅在尚无高职候选时输出scope:tca_military_selected_marshal，交由调用者消费并清理。
tca_military_select_marshal_from_position_effect = {
    if = {
        limit = { NOT = { exists = scope:tca_military_selected_marshal } employs_court_position = $POSITION$ }
        court_position:$POSITION$ = {
            if = {
                limit = {
                    is_alive = yes
                    is_courtier_of = scope:tca_military_council_liege
                    can_be_marshal_trigger = { COURT_OWNER = scope:tca_military_council_liege }
                    trigger_if = { limit = { has_variable = block_fire_councillor } var:block_fire_councillor ?= scope:tca_military_council_liege }
                    trigger_if = { limit = { exists = scope:tca_military_departing_officer } NOT = { this = scope:tca_military_departing_officer } }
                }
                save_scope_as = tca_military_selected_marshal
            }
        }
    }
}

# 主君作用域；仅补真正空缺，不替换玩家已有司马，不生成候选。
# 私有临时scope：tca_military_departing_officer 由离任清理建立并在调用后清除；
# 排除该人以免撤职后立即返聘，或在先撤旧职后迁廷的升迁链中短暂重新入阁。
# 成功任命后消费主君的任务快照；本入口创建的候选与主君scope在本调用内清理。
# tca_test_force_player_governor只启用测试分支，不改变原生AI身份或证明真人界面通过。
tca_military_player_fallback_marshal_effect = {
    tca_court_player_fallback_council_effect = { COUNCIL = councillor_marshal HAS_POSITION = tca_court_has_marshal_commission_trigger ELIGIBILITY = can_be_marshal_trigger SKILL = martial RESTORE = tca_military_restore_marshal_task_effect }
}

# 回调作用域；只有模块标记与当前原生锁都指向同一主君时才删除锁。
# 输入scope:employee与scope:liege来自原生职位回调；失效后还可能收到vacated，故清理须幂等。
# 建立并清理离任者排除scope；只有仍属原主君且原生规则允许时才解雇，再统一选择剩余差遣或补位。
# 死亡人物不再访问变量；自然离廷/死亡由vacated覆盖，不能只依赖invalidated。
tca_military_court_position_on_departure_effect = {
    tca_court_linked_council_departure_effect = { COUNCIL = councillor_marshal CAPTURE = tca_military_capture_marshal_task_effect SYNC = tca_military_sync_local_marshal_effect }
}

# 保留生命周期入口供三种职位复用；清理规则只有一个维护位置。
tca_military_court_position_on_revoked_effect = { tca_military_court_position_on_departure_effect = yes }
tca_military_court_position_on_invalidated_effect = { tca_military_court_position_on_departure_effect = yes }
tca_military_court_position_on_vacated_effect = { tca_military_court_position_on_departure_effect = yes }

# 旧军事单职调用适配；正常季度直接使用共用目录。
# 参数POSITION、KEEP_CURRENT保留原调用契约；旧名单只在适配期间并入共用候选。
# 当前雇主与scope:target_liege须一致，输出scope:tca_chosen_officer来自共用最高分流程。
tca_military_fill_single_position_effect = {
    if = { limit = { tca_military_linked_council_available_trigger = { KEEP_CURRENT = $KEEP_CURRENT$ } }
    every_in_list = { list = tca_military_candidate_pool add_to_list = tca_court_candidates }
    set_variable = { name = tca_court_legacy_position value = flag:$POSITION$ }
    tca_court_visit_positions_effect = { VISITOR = tca_court_legacy_fill_effect }
    remove_variable = tca_court_legacy_position
    every_in_list = { list = tca_court_candidates remove_from_list = tca_court_candidates }
    }
}

# 人物作用域兼容入口；委派共用候选收集，不再按原职类拆分池。
tca_military_collect_officer_effect = { tca_court_collect_officer_effect = yes }

# 独立天朝最高领主作用域；保留原on_action调用名，所有铨选在共用框架执行。
tca_military_quarterly_fill_vacancies_effect = { tca_court_quarterly_fill_effect = yes }

# 官署领主作用域；供改制和主君死亡后的真实雇主重绑调用，只检查自己的三种差遣。
# 改制回归曾在两日观察窗口内见到原生未清理，因此此入口显式验证官署与人物资格。
# 非地方官署还会清理本模块任务快照；不据此推断原生所有失效检查的周期。
tca_military_validate_office_after_government_change_effect = {
    tca_military_revoke_invalid_office_position_effect = { POSITION = tca_military_supervisor_court_position VALIDITY = tca_military_supervisor_office_valid_trigger CHARACTER_VALIDITY = tca_military_supervisor_validity_trigger }
    tca_military_revoke_invalid_office_position_effect = { POSITION = tca_military_inspector_court_position VALIDITY = tca_military_inspector_office_valid_trigger CHARACTER_VALIDITY = tca_military_inspector_validity_trigger }
    tca_military_revoke_invalid_office_position_effect = { POSITION = tca_corps_inspector_court_position VALIDITY = tca_corps_inspector_office_valid_trigger CHARACTER_VALIDITY = tca_corps_inspector_validity_trigger }
    if = {
        limit = { tca_is_landed_celestial_governor_trigger = no }
        remove_variable = tca_military_marshal_task
        remove_variable = tca_military_marshal_target
    }
}

# 主君作用域；POSITION为职位ID，VALIDITY为官署准入，CHARACTER_VALIDITY为对应人物资格trigger。
# 正式撤职回调负责释放模块锁；保留已经存在的free_revoke_cost标记。
tca_military_revoke_invalid_office_position_effect = {
    if = {
        limit = { employs_court_position = $POSITION$ }
        court_position:$POSITION$ = { save_scope_as = tca_military_invalid_office_holder }
        if = {
            # 临时scope只在这次trigger求值内使用，不能从前一个effect带入。
            limit = {
                save_temporary_scope_as = liege
                scope:tca_military_invalid_office_holder = { save_temporary_scope_as = employee }
                OR = { $VALIDITY$ = no scope:tca_military_invalid_office_holder = { $CHARACTER_VALIDITY$ = no } }
            }
            if = {
                limit = { has_character_flag = free_revoke_cost }
                revoke_court_position = { recipient = scope:tca_military_invalid_office_holder court_position = $POSITION$ }
            }
            else = {
                add_character_flag = free_revoke_cost
                revoke_court_position = { recipient = scope:tca_military_invalid_office_holder court_position = $POSITION$ }
                remove_character_flag = free_revoke_cost
            }
        }
        clear_saved_scope = tca_military_invalid_office_holder
    }
}

# 武官作用域；参数OWNER为本次清理的原主君。归属匹配时移除模块标记；
# 只有原生block_fire_councillor的当前值也匹配OWNER，才一并删除原生保护变量。
# 共享于自然离任和主君死亡前的清理；不读取死亡武官，也不认领外部保护。
tca_military_release_council_lock_effect = {
    if = {
        limit = { is_alive = yes has_variable = tca_military_council_lock_owner var:tca_military_council_lock_owner ?= $OWNER$ }
        if = { limit = { has_variable = block_fire_councillor var:block_fire_councillor ?= $OWNER$ } remove_variable = block_fire_councillor }
        remove_variable = tca_military_council_lock_owner
    }
}
