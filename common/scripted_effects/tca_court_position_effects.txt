# 共用季度流程；人物池、空缺清单仅存续于本次调用。
# 目录访问器不另维护文武业务分支；目标倾向、准入和重要度由静态参数输入。
# 雇主作用域；一次冻结本轮所有空缺，保存数值而非后续重新扫描职位。
tca_court_snapshot_position_effect = {
    tca_court_metadata_contract_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ }
    save_scope_value_as = { name = tca_court_bias value = $BIAS$ }
    if = { limit = { $OFFICE$ = yes NOT = { employs_court_position = $POSITION$ }
        $POSITION$_council_available_trigger = yes }
        set_variable = { name = tca_court_vacant_$POSITION$ value = $BASE$ }
        if = { limit = { scope:tca_court_bias = 1 } change_variable = { name = tca_court_vacant_$POSITION$ add = tca_court_civil_office_weight } }
        else = { change_variable = { name = tca_court_vacant_$POSITION$ add = tca_military_office_strategic_weight } }
    }
}
# 雇主作用域，访问器只移除本模块临时标记。
tca_court_clear_snapshot_effect = {
    tca_court_metadata_contract_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ } remove_variable = tca_court_vacant_$POSITION$ }
# 人物作用域；共用候选只收集一次，任期未满者不加入。所有原生途径与硬资格在目标处仍会复验。
tca_court_collect_officer_effect = {
    if = { limit = {
        tca_court_person_eligible_trigger = yes
        NOT = { has_character_flag = tca_military_appointment_cooldown }
        NOT = { is_in_list = tca_court_candidates }
    } add_to_list = tca_court_candidates }
}
# 人物作用域；目录中命中真实旧差遣后，记录旧职位类型并精确撤职，不随机猜雇主。
tca_court_revoke_previous_effect = {
    if = { limit = { has_court_position = $POSITION$ }
        set_variable = { name = tca_court_old_position value = flag:$POSITION$ }
        scope:tca_court_previous_liege = {
            if = { limit = { has_character_flag = free_revoke_cost }
                revoke_court_position = { recipient = scope:tca_chosen_officer court_position = $POSITION$ }
            }
            else = {
                add_character_flag = free_revoke_cost
                revoke_court_position = { recipient = scope:tca_chosen_officer court_position = $POSITION$ }
                remove_character_flag = free_revoke_cost
            }
        }
    }
}
# 失败恢复在原宫廷和原职仍合法时执行；不抢占已被自然回调填补的旧职。
tca_court_restore_previous_effect = {
    if = { limit = { scope:tca_chosen_officer = { var:tca_court_old_position ?= flag:$POSITION$ } }
        scope:tca_court_previous_liege = {
            if = { limit = { is_alive = yes $OFFICE$ = yes NOT = { employs_court_position = $POSITION$ }
        $POSITION$_council_available_trigger = yes }
                appoint_court_position = { recipient = scope:tca_chosen_officer court_position = $POSITION$ }
            }
        }
        scope:tca_chosen_officer = { remove_character_flag = tca_military_appointment_cooldown }
    }
}
# 人物作用域；原生没有已确认的通用内阁类型读取接口，五席显式对应原生资格。
# VISITOR接收COUNCIL与ELIGIBILITY；配偶不可撤换、部寺在任者持正式头衔，均不在普通候选中。
tca_court_visit_unlinked_council_effect = {
    $VISITOR$ = { COUNCIL = councillor_chancellor ELIGIBILITY = can_be_chancellor_trigger }
    $VISITOR$ = { COUNCIL = councillor_steward ELIGIBILITY = can_be_steward_trigger }
    $VISITOR$ = { COUNCIL = councillor_marshal ELIGIBILITY = can_be_marshal_trigger }
    $VISITOR$ = { COUNCIL = councillor_spymaster ELIGIBILITY = can_be_spymaster_trigger }
    $VISITOR$ = { COUNCIL = councillor_court_chaplain ELIGIBILITY = can_be_court_chaplain_trigger }
}
# 人物作用域；调用前已确认是可流转普通内阁，必需scope:tca_court_previous_liege。
# 旧席只在本次同步铨选链保存；不保存AI任务，失败恢复后由原生AI重新选任务。
tca_court_snapshot_unlinked_council_slot_effect = {
    if = { limit = { has_council_position = $COUNCIL$ $ELIGIBILITY$ = { COURT_OWNER = scope:tca_court_previous_liege } }
        set_variable = { name = tca_court_old_council value = flag:$COUNCIL$ }
    }
}
# 人物作用域；调用者已复验原雇主与人物身份，仅恢复仍空缺、资格仍合法的旧席。
tca_court_restore_unlinked_council_slot_effect = {
    if = { limit = {
        var:tca_court_old_council ?= flag:$COUNCIL$
        scope:tca_court_previous_liege = { NOT = { exists = cp:$COUNCIL$ } }
        $ELIGIBILITY$ = { COURT_OWNER = scope:tca_court_previous_liege }
    } scope:tca_court_previous_liege = {
        assign_councillor_type = { type = $COUNCIL$ target = scope:tca_chosen_officer fire_on_actions = yes }
    } }
}
# 任命失败且人物已回原宫廷后调用；必需chosen_officer及previous_liege。
# 不抢占原生补位者，不在改制或玩家接管后恢复旧席；无旧席快照则无副作用。
tca_court_restore_unlinked_council_effect = {
    if = { limit = {
        scope:tca_court_previous_liege = { is_alive = yes is_ai = yes government_has_flag = government_is_celestial NOT = { has_character_flag = tca_test_force_player_governor } }
        scope:tca_chosen_officer = { has_variable = tca_court_old_council is_councillor = no is_courtier_of = scope:tca_court_previous_liege tca_court_person_eligible_trigger = yes }
    } scope:tca_chosen_officer = { tca_court_visit_unlinked_council_effect = { VISITOR = tca_court_restore_unlinked_council_slot_effect } } }
}
# 回调scope:employee为任职者、scope:liege为新雇主；成功授职才清理普通内阁旧席。
# 同署转换必须先清旧席再由调用者同步对应内阁；跨署旧席通常已由原生离廷处理。
# 短暂许可在回调内消费，失败授职由fill入口兜底清除，不预先撤掉同署普通内阁。
tca_court_on_received_effect = {
    if = { limit = { scope:employee = {
        has_character_flag = tca_court_unlinked_council_transfer
        is_councillor_of = scope:liege
        tca_court_ai_council_releasable_trigger = yes
    } } scope:liege = { fire_councillor = scope:employee } }
    scope:employee = { remove_character_flag = tca_court_unlinked_council_transfer }
    scope:employee = { add_character_flag = { flag = tca_military_appointment_cooldown days = 730 } }
}
tca_court_on_departure_effect = {
    scope:employee = {
        if = { limit = { is_alive = yes }
            remove_character_flag = tca_military_appointment_cooldown
            remove_variable = tca_court_organizing_office
        }
    }
}
# 雇主作用域，scope:target_liege须是当前雇主。候选存于tca_court_candidates。
# 参数来自目录；最高分名单保留所有同分者后随机，不截取前三或依赖列表排序同分顺序。
tca_court_fill_position_effect = {
    clear_saved_scope = tca_chosen_officer
    clear_saved_scope = tca_court_previous_liege
    save_scope_value_as = { name = tca_court_bias value = $BIAS$ }
    save_scope_value_as = { name = tca_court_admission value = $ADMISSION$ }
    save_scope_value_as = { name = tca_court_target_importance value = $BASE$ }
    if = { limit = { scope:tca_court_bias = 1 }
        save_scope_value_as = { name = tca_court_target_importance value = { value = $BASE$ add = tca_court_civil_office_weight } }
    }
    else = { save_scope_value_as = { name = tca_court_target_importance value = { value = $BASE$ add = tca_military_office_strategic_weight } } }
    if = { limit = {
        $OFFICE$ = yes NOT = { employs_court_position = $POSITION$ }
        $POSITION$_council_available_trigger = yes
        trigger_if = { limit = { scope:tca_court_bias = 2 } tca_military_linked_council_available_trigger = { KEEP_CURRENT = $KEEP_CURRENT$ } }
    }
        # 原生候选称职度仍使用人物的真实宫廷上下文；不人为迁廷后再评估或重算原始分。
        every_in_list = { list = tca_court_candidates
            if = { limit = {
                tca_court_target_person_trigger = { SKILL = $SKILL$ }
                $POSITION$_council_person_trigger = { OWNER = scope:target_liege }
                trigger_if = { limit = { has_variable = block_fire_councillor }
                    OR = {
                        var:block_fire_councillor = scope:target_liege
                        AND = { exists = liege_or_court_owner var:tca_military_council_lock_owner ?= liege_or_court_owner var:block_fire_councillor = liege_or_court_owner }
                    }
                }
                NOT = { has_character_flag = tca_military_appointment_cooldown }
                tca_court_current_importance < scope:tca_court_target_importance
                top_liege = scope:target_liege.top_liege
            }
                save_scope_value_as = { name = tca_court_this_score value = { value = aptitude:$POSITION$ multiply = 20 add = tca_court_personal_selection_value } }
                if = { limit = { NOT = { exists = scope:tca_chosen_officer } }
                    save_scope_as = tca_chosen_officer
                    save_scope_value_as = { name = tca_court_best_score value = scope:tca_court_this_score }
                    add_to_list = tca_court_best_candidates
                }
                else_if = { limit = { scope:tca_court_this_score > scope:tca_court_best_score }
                    every_in_list = { list = tca_court_best_candidates remove_from_list = tca_court_best_candidates }
                    save_scope_value_as = { name = tca_court_best_score value = scope:tca_court_this_score }
                    add_to_list = tca_court_best_candidates
                }
                else_if = { limit = { scope:tca_court_this_score = scope:tca_court_best_score } add_to_list = tca_court_best_candidates }
            }
        }
        clear_saved_scope = tca_chosen_officer
        random_in_list = { list = tca_court_best_candidates save_scope_as = tca_chosen_officer }
        every_in_list = { list = tca_court_best_candidates remove_from_list = tca_court_best_candidates }
        if = { limit = { exists = scope:tca_chosen_officer }
            scope:tca_chosen_officer = {
                remove_from_list = tca_court_candidates
                remove_variable = tca_court_old_position
                remove_variable = tca_court_old_council
                remove_character_flag = tca_court_unlinked_council_transfer
                liege_or_court_owner = { save_scope_as = tca_court_previous_liege }
                if = { limit = { tca_court_unlinked_council_candidate_trigger = yes }
                    tca_court_visit_unlinked_council_effect = { VISITOR = tca_court_snapshot_unlinked_council_slot_effect }
                    add_character_flag = tca_court_unlinked_council_transfer
                }
                tca_court_visit_lifecycle_positions_effect = { RESTORE = 0 }
                add_character_flag = tca_military_appointment_allowed
            }
            if = { limit = { NOT = { scope:tca_chosen_officer = { is_courtier_of = scope:target_liege } } }
                add_courtier = scope:tca_chosen_officer
            }
            appoint_court_position = { recipient = scope:tca_chosen_officer court_position = $POSITION$ }
            if = { limit = { scope:tca_chosen_officer = { has_court_position = $POSITION$ is_courtier_of = scope:target_liege } }
                if = { limit = { is_ai = no }
                    send_interface_message = { type = msg_court_position_gained title = tca_court_appointment_title desc = tca_court_appointment_desc left_icon = scope:tca_chosen_officer }
                }
                if = { limit = { scope:tca_chosen_officer = { has_variable = tca_court_old_position } exists = scope:tca_court_previous_liege NOT = { scope:tca_court_previous_liege = scope:target_liege } }
                    scope:tca_court_previous_liege = {
                        if = { limit = { is_alive = yes is_ai = no }
                            send_interface_message = { type = msg_court_position_removed title = tca_court_departure_title desc = tca_court_departure_desc left_icon = scope:tca_chosen_officer right_icon = scope:target_liege }
                        }
                    }
                }
            }
            else = {
                # 任命未落地时回原宫廷并尝试恢复，绝不发送成功公文。
                scope:tca_court_previous_liege ?= {
                    if = { limit = { is_alive = yes }
                        add_courtier = scope:tca_chosen_officer
                        tca_court_visit_lifecycle_positions_effect = { RESTORE = 1 }
                        tca_court_restore_unlinked_council_effect = yes
                    }
                }
                debug_log = "TCA_COURT appointment failed; original court restored where valid"
            }
            scope:tca_chosen_officer = { remove_character_flag = tca_military_appointment_allowed remove_character_flag = tca_court_unlinked_council_transfer remove_variable = tca_court_old_position remove_variable = tca_court_old_council }
        }
    }
}
# 雇主作用域；只处理冻结清单中的当前最高空缺。移除标记后，即使无合格人选也保证前进。
tca_court_dispatch_position_effect = {
    tca_court_metadata_contract_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ }
    if = { limit = { has_variable = tca_court_vacant_$POSITION$ var:tca_court_vacant_$POSITION$ = scope:tca_court_dispatch_importance NOT = { exists = scope:tca_court_dispatched } }
        save_scope_value_as = { name = tca_court_dispatched value = 1 }
        remove_variable = tca_court_vacant_$POSITION$
        tca_court_fill_position_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ }
    }
}
# 独立天朝最高领主作用域；只有此入口建立全朝清单，原季度与开局入口委派到这里。
tca_court_quarterly_fill_effect = {
    if = { limit = { NOT = { has_character_flag = tca_court_batch_active } }
        add_character_flag = tca_court_batch_active
        every_vassal_or_below = {
            limit = { government_has_flag = government_is_celestial }
            tca_court_visit_positions_effect = { VISITOR = tca_court_snapshot_position_effect }
            if = { limit = { tca_court_next_vacancy_value > 0 } add_to_list = tca_court_vacant_offices }
        }
        if = { limit = { list_size:tca_court_vacant_offices > 0 }
            every_courtier = { tca_court_collect_officer_effect = yes }
            every_vassal_or_below = {
                limit = { OR = { government_has_flag = government_is_celestial primary_title = { is_noble_family_title = yes } } }
                every_courtier = { tca_court_collect_officer_effect = yes }
            }
            # 每轮最多移除一个冻结职位，空缺数量有限；任何失败也消费该条，不产生重入循环。
            while = { limit = { list_size:tca_court_vacant_offices > 0 list_size:tca_court_candidates > 0 }
                ordered_in_list = { list = tca_court_vacant_offices order_by = tca_court_next_vacancy_value max = 1 check_range_bounds = no
                    save_scope_as = target_liege
                    save_scope_value_as = { name = tca_court_dispatch_importance value = tca_court_next_vacancy_value }
                    clear_saved_scope = tca_court_dispatched
                    tca_court_visit_positions_effect = { VISITOR = tca_court_dispatch_position_effect }
                    if = { limit = { tca_court_next_vacancy_value = 0 } remove_from_list = tca_court_vacant_offices }
                }
            }
        }
        # 无候选时也清除全部尚未消费的冻结变量，不让下季读取旧权重。
        every_vassal_or_below = { tca_court_visit_positions_effect = { VISITOR = tca_court_clear_snapshot_effect } }
        every_in_list = { list = tca_court_vacant_offices remove_from_list = tca_court_vacant_offices }
        every_in_list = { list = tca_court_candidates remove_from_list = tca_court_candidates }
        clear_saved_scope = tca_chosen_officer
            clear_saved_scope = tca_court_previous_liege
        remove_character_flag = tca_court_batch_active
    }
}

# 旧单职入口按目录匹配，不复制职位元数据。
tca_court_legacy_fill_effect = {
    tca_court_metadata_contract_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ }
    if = { limit = { var:tca_court_legacy_position = flag:$POSITION$ }
        tca_court_fill_position_effect = { POSITION = $POSITION$ BASE = $BASE$ BIAS = $BIAS$ ADMISSION = $ADMISSION$ SKILL = $SKILL$ OFFICE = $OFFICE$ KEEP_CURRENT = $KEEP_CURRENT$ }
    }
}
# 目录行的统一宏参数契约；原生拒绝未消费参数，故访问器转交此无副作用类型检查。
# 必须保持always=no，不作为资格或流程判断，不在运行时写状态。
tca_court_metadata_contract_effect = {
    if = { limit = { always = no has_court_position = $POSITION$ $SKILL$ >= 0 $OFFICE$ = yes $KEEP_CURRENT$ = yes }
        save_scope_value_as = { name = tca_court_unused_metadata value = { value = $BASE$ add = $BIAS$ add = $ADMISSION$ } }
    }
}

# 生命周期静态分发；只消费该操作所需的参数，不重入填缺目录。
tca_court_lifecycle_position_effect = {
    save_scope_value_as = { name = tca_court_restore_mode value = $RESTORE$ }
    if = { limit = { scope:tca_court_restore_mode = 0 } tca_court_revoke_previous_effect = { POSITION = $POSITION$ } }
    else = { tca_court_restore_previous_effect = { POSITION = $POSITION$ OFFICE = $OFFICE$ } }
}
