# 作用域：任意；无参数；输出唯一Situation；重复调用保留原对象、阶段与进度。
# P0仅供调试决议与原生探针调用，未接入正式开局或旧存档初始化。
tca_frontier_situation_start_effect = {
    if = { limit = { NOT = { exists = situation:tca_northern_frontier } }
        start_situation = { type = tca_northern_frontier }
    }
}

# 作用域：任意；无参数；结束本模块唯一Situation，无对象时不执行。
# P0无外挂计时器、县级修正或参与者私有状态，清理由原生Situation承担。
tca_frontier_situation_end_effect = {
    situation:tca_northern_frontier ?= { end_situation = yes }
}

# 作用域：任意；REGION为本局势子区键，CATALYST为已声明催化键；无saved scope输出。
# 只在所指子区触发原生催化，禁止在Situation层广播；不存在局势时不执行。
# 调用方应选择当前阶段可用的催化。测试和决议共用此入口，不直接设阶段或进度。
tca_frontier_situation_catalyst_effect = {
    situation:tca_northern_frontier ?= {
        situation_sub_region:$REGION$ = { trigger_sub_region_catalyst = $CATALYST$ }
    }
}
