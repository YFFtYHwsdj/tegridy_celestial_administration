# 人物作用域；零级中央来源按当前原生文武分类冻结唯一路线。
# 保存被借用前的路线，取消时精确还原；不碰已有活动身份或核定等级。
tca_merit_ensure_central_zero_route_effect = {
    # 当前原生分类是真值；取消时还原既有路线，不把借用的分类留给后续普通流程。
    if = {
        limit = { NOT = { has_character_flag = tca_merit_zero_route_owned } }
        if = { limit = { has_variable = tca_merit_route } set_variable = { name = tca_merit_zero_previous_route value = var:tca_merit_route } }
        add_character_flag = tca_merit_zero_route_owned
    }
    if = { limit = { is_civilian_candidate_trigger = yes is_military_candidate_trigger = no } set_variable = { name = tca_merit_route value = 1 } }
    else_if = { limit = { is_military_candidate_trigger = yes is_civilian_candidate_trigger = no } set_variable = { name = tca_merit_route value = 2 } }
}

# 人物作用域；取消时只撤回本事务的路线借用；成功或真实入品接管则解除所有权而不调用此撤回。
tca_merit_cancel_central_zero_source_effect = {
    if = {
        limit = { has_character_flag = tca_merit_zero_route_owned }
        if = { limit = { has_variable = tca_merit_zero_previous_route } set_variable = { name = tca_merit_route value = var:tca_merit_zero_previous_route } }
        else = { remove_variable = tca_merit_route }
        remove_variable = tca_merit_zero_previous_route
        remove_character_flag = tca_merit_zero_route_owned
    }
}

# 统一授品事务。人物作用域；TARGET为原生1..9，SOURCE仅中央真实未入品可为0，MODE为1普通/2中央，YEAR为自然年。
# prepare只建立持久核准队列，输出grant_result=0；不改双账、上限或成功年。
# begin同链输出grant_result：1已提交、0未提交；业务只读核定变量，不反读尚未刷新的原生等级。
# token永久单调保留；phase=1待提交，2已提交待审计，3审计失败隔离。隔离阻止所有后续事务。
tca_merit_prepare_grant_effect = {
    set_variable = { name = tca_merit_grant_result value = 0 }
    # 参数冻结为数值scope后比较；宏参数不能展开成以整数命名的trigger。
    save_scope_value_as = { name = tca_merit_prepare_source value = $SOURCE$ }
    save_scope_value_as = { name = tca_merit_prepare_target value = $TARGET$ }
    save_scope_value_as = { name = tca_merit_prepare_mode value = $MODE$ }
    if = {
        limit = {
            NOT = { has_variable = tca_merit_grant_phase }
            exists = top_liege
            OR = {
                AND = { scope:tca_merit_prepare_source >= 1 has_variable = tca_merit_route has_variable = tca_merit_approved_level }
                AND = { scope:tca_merit_prepare_source = 0 scope:tca_merit_prepare_mode = 2 OR = { scope:tca_merit_prepare_target = 8 scope:tca_merit_prepare_target = 9 } tca_merit_central_zero_source_trigger = yes tca_merit_central_target_value = $TARGET$ }
            }
        }
        if = { limit = { scope:tca_merit_prepare_source = 0 } tca_merit_ensure_central_zero_route_effect = yes }
        if = { limit = { NOT = { has_variable = tca_merit_grant_token } } set_variable = { name = tca_merit_grant_token value = 0 } }
        change_variable = { name = tca_merit_grant_token add = 1 }
        set_variable = { name = tca_merit_grant_source value = $SOURCE$ }
        set_variable = { name = tca_merit_grant_target value = $TARGET$ }
        set_variable = { name = tca_merit_grant_mode value = $MODE$ }
        set_variable = { name = tca_merit_grant_year value = $YEAR$ }
        set_variable = { name = tca_merit_grant_liege value = top_liege }
        set_variable = { name = tca_merit_grant_route value = var:tca_merit_route }
        set_variable = { name = tca_merit_grant_phase value = 1 }
    }
}

# 人物作用域；正式年度／中央入口使用此同链prepare→commit组合。
# 已有phase（包括保存后的phase1）不得被新begin偷用；续接队列只能带原token显式commit。
tca_merit_begin_grant_effect = {
    set_variable = { name = tca_merit_grant_result value = 0 }
    if = {
        limit = { NOT = { has_variable = tca_merit_grant_phase } }
        tca_merit_prepare_grant_effect = { SOURCE = $SOURCE$ TARGET = $TARGET$ MODE = $MODE$ YEAR = $YEAR$ }
        if = { limit = { var:tca_merit_grant_phase ?= 1 has_variable = tca_merit_grant_token } tca_merit_commit_grant_effect = { TOKEN = var:tca_merit_grant_token } }
    }
}

# 人物作用域；TOKEN必须为本事务号。核准与提交可分离测试，所有资格在落账前重核。
# 不检查实时Merit：普通年度以调用者冻结资望决定入池；中央不要求普通门槛和出身上限。
tca_merit_commit_grant_effect = {
    if = {
        limit = { var:tca_merit_grant_token ?= $TOKEN$ var:tca_merit_grant_phase ?= 1 }
        set_variable = { name = tca_merit_grant_valid value = 0 }
        if = {
            limit = {
                OR = {
                    AND = { var:tca_merit_grant_source >= 1 tca_merit_population_trigger = yes var:tca_merit_approved_level ?= var:tca_merit_grant_source }
                    AND = { var:tca_merit_grant_source = 0 var:tca_merit_grant_mode = 2 tca_merit_central_zero_source_commit_trigger = yes }
                }
                is_alive = yes
                exists = top_liege
                exists = var:tca_merit_grant_liege
                top_liege ?= var:tca_merit_grant_liege
                var:tca_merit_grant_year = current_year
                var:tca_merit_grant_target > var:tca_merit_grant_source
                var:tca_merit_grant_target <= 9
                var:tca_merit_route ?= var:tca_merit_grant_route
                OR = {
                    AND = { var:tca_merit_grant_route = 1 is_civilian_candidate_trigger = yes is_military_candidate_trigger = no }
                    AND = { var:tca_merit_grant_route = 2 is_military_candidate_trigger = yes is_civilian_candidate_trigger = no }
                }
            }
            if = {
                limit = {
                    var:tca_merit_grant_mode = 1
                    tca_merit_ordinary_hard_candidate_trigger = yes
                    var:tca_merit_grant_source < 7
                    var:tca_merit_grant_target <= tca_merit_theoretical_cap_value
                }
                set_variable = { name = tca_merit_grant_expected value = { value = var:tca_merit_grant_source add = 1 } }
                if = { limit = { var:tca_merit_grant_target = var:tca_merit_grant_expected } set_variable = { name = tca_merit_grant_valid value = 1 } }
                remove_variable = tca_merit_grant_expected
            }
            # 中央调用者仍须先判实际持职；落账边界再验，不能用过期任命快照授品。
            else_if = {
                limit = {
                    var:tca_merit_grant_mode = 2
                    OR = { var:tca_merit_grant_target = 8 var:tca_merit_grant_target = 9 }
                    tca_merit_central_target_value >= var:tca_merit_grant_target
                }
                set_variable = { name = tca_merit_grant_valid value = 1 }
            }
        }
        if = {
            limit = { var:tca_merit_grant_valid = 1 }
            add_character_flag = tca_merit_grant_suppress
            # 归零后重建来源经验，再刷新目标cap；只让解除保护后的异步上限刷新产生合法gain。
            set_merit_level = 0
            set_merit_level = var:tca_merit_grant_source
            set_variable = { name = tca_merit_approved_level value = var:tca_merit_grant_target }
            # 零级只在全部提交复验通过后转成正式身份，不制造中间九品/gain。
            if = {
                limit = { var:tca_merit_grant_source = 0 }
                add_character_flag = tca_merit_active
                remove_character_flag = tca_merit_entry_pending
                remove_character_flag = tca_merit_central_zero_prep_owned
                remove_character_flag = tca_merit_zero_route_owned
                remove_variable = tca_merit_zero_previous_route
            }
            tca_merit_refresh_cap_effect = yes
            set_merit_level = var:tca_merit_grant_target
            if = { limit = { var:tca_merit_grant_target = 2 } change_merit_no_experience = { value = merit_level_2_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 3 } change_merit_no_experience = { value = merit_level_3_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 4 } change_merit_no_experience = { value = merit_level_4_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 5 } change_merit_no_experience = { value = merit_level_5_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 6 } change_merit_no_experience = { value = merit_level_6_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 7 } change_merit_no_experience = { value = merit_level_7_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 8 } change_merit_no_experience = { value = merit_level_8_value subtract = merit } }
            else_if = { limit = { var:tca_merit_grant_target = 9 } change_merit_no_experience = { value = merit_level_9_value subtract = merit } }
            remove_character_flag = tca_merit_grant_suppress
            set_variable = { name = tca_merit_last_ordinary_year value = current_year }
            # 反馈/面板专用的最近一轮普通获准标记；中央授品不写，避免把中央迁秩当成普通落选或普通成功。
            if = { limit = { var:tca_merit_grant_mode = 1 } set_variable = { name = tca_merit_last_ordinary_success_year value = current_year } }
            set_variable = { name = tca_merit_grant_phase value = 2 }
            # 中央授品已提交；等待入品的补授所有权结束。
            if = { limit = { var:tca_merit_grant_mode = 2 } remove_character_flag = tca_merit_central_await_entry }
            set_variable = { name = tca_merit_grant_audit_attempts value = 0 }
            set_variable = { name = tca_merit_grant_result value = 1 }
            # saved value随事件队列保存，旧回调不能拿新事务号解锁；重载仍可续审。
            save_scope_value_as = { name = tca_merit_grant_token value = var:tca_merit_grant_token }
            trigger_event = { id = tca_merit_grant.1 days = 2 }
        }
        else = {
            remove_variable = tca_merit_grant_phase
            if = {
                limit = { var:tca_merit_grant_source = 0 }
                tca_merit_cancel_central_zero_prep_effect = yes
                tca_merit_cancel_central_zero_source_effect = yes
            }
        }
        remove_variable = tca_merit_grant_valid
    }
}

# 人物作用域；TOKEN为延迟事件冻结的事务号。未确认等级保留phase2，禁止再次落账。
# 确认后先冻结结果事件所有字段，再解锁和处理中央队列；按钮永远不拥有资源副作用。
tca_merit_audit_grant_effect = {
    if = {
        limit = { var:tca_merit_grant_token ?= $TOKEN$ OR = { var:tca_merit_grant_phase ?= 2 var:tca_merit_grant_phase ?= 3 } }
        if = {
            # 提交后真实退出不撤销既成双账；结束审计，避免返境后被旧事务永久阻塞。
            limit = { OR = { is_alive = no NOT = { has_character_flag = tca_merit_active } } }
            remove_variable = tca_merit_grant_phase
            tca_merit_process_central_pending_effect = yes
        }
        else_if = {
            limit = { var:tca_merit_grant_phase ?= 3 }
            # 隔离不自动恢复；旧排队事件至多到此，不续排、不通知、不处理中央待授。
        }
        else_if = {
            limit = { merit_level = var:tca_merit_grant_target }
            # 显式恢复后原生等级终于确认时，修复同一轮普通结果；不创建AI展示账，
            # 也不覆盖别年、别路线/来源或中央事务的记录。消息仍由原TOKEN防重。
            if = {
                limit = {
                    var:tca_merit_grant_mode = 1
                    has_variable = tca_merit_feedback_last_result
                    var:tca_merit_feedback_last_result = 2
                    var:tca_merit_feedback_last_year ?= var:tca_merit_grant_year
                    var:tca_merit_feedback_last_route ?= var:tca_merit_grant_route
                    var:tca_merit_feedback_last_source ?= var:tca_merit_grant_source
                }
                set_variable = { name = tca_merit_feedback_last_result value = 1 }
            }
            if = {
                limit = { is_ai = no OR = { NOT = { has_variable = tca_merit_grant_notified_token } NOT = { var:tca_merit_grant_notified_token ?= var:tca_merit_grant_token } } }
                tca_merit_prepare_grant_notice_effect = { TOKEN = var:tca_merit_grant_token }
                trigger_event = { id = tca_merit_grant.2 }
            }
            remove_variable = tca_merit_grant_phase
            tca_merit_process_central_pending_effect = yes
            # 事务期间被抑制的入品不会触发native_gain兜底；审计结束后再补一次。
            tca_merit_maybe_queue_central_catchup_effect = yes
        }
        else = {
            change_variable = { name = tca_merit_grant_audit_attempts add = 1 }
            if = {
                limit = { var:tca_merit_grant_audit_attempts < 5 }
                debug_log = "TCA_MERIT_GRANT AUDIT_WAIT"
                save_scope_value_as = { name = tca_merit_grant_token value = var:tca_merit_grant_token }
                trigger_event = { id = tca_merit_grant.1 days = 2 }
            }
            else = {
                # 有界失败不发成功通知、不重授或回滚；phase3隔离阻止下一笔及中央待授，停止事件循环。
                set_variable = { name = tca_merit_grant_audit_failed_token value = var:tca_merit_grant_token }
                # 面板最近一轮结果同步为“未形成普通结果”，但仍不发送落选消息。
                # 只改本轮普通成功记录，不覆盖同年中央迁秩或其它年度的旧结果。
                if = {
                    limit = {
                        tca_merit_feedback_human_recipient_trigger = yes
                        var:tca_merit_grant_mode = 1
                        has_variable = tca_merit_feedback_last_year
                        var:tca_merit_feedback_last_year ?= var:tca_merit_grant_year
                        var:tca_merit_feedback_last_result = 1
                    }
                    set_variable = { name = tca_merit_feedback_last_result value = 2 }
                }
                debug_log = "TCA_MERIT_GRANT AUDIT_FAILED"
                set_variable = { name = tca_merit_grant_phase value = 3 }
            }
        }
    }
}

# 人物作用域；仅准备一次结果数据，不派发UI。TOKEN必须匹配当前已提交事务。
# 正式审计只对玩家调用；原生数据用例可用AI检查幂等及冻结值，不能据此宣称玩家UI通过。
tca_merit_prepare_grant_notice_effect = {
    set_variable = { name = tca_merit_grant_notice_ready value = 0 }
    if = {
        limit = {
            var:tca_merit_grant_token ?= $TOKEN$
            var:tca_merit_grant_phase ?= 2
            OR = { NOT = { has_variable = tca_merit_grant_notified_token } NOT = { var:tca_merit_grant_notified_token ?= var:tca_merit_grant_token } }
        }
                set_variable = { name = tca_merit_grant_notified_token value = var:tca_merit_grant_token }
                save_scope_value_as = { name = tca_merit_notice_mode value = var:tca_merit_grant_mode }
                save_scope_value_as = { name = tca_merit_notice_route value = var:tca_merit_grant_route }
                save_scope_value_as = { name = tca_merit_notice_source value = 0 }
                if = { limit = { var:tca_merit_grant_source > 0 } save_scope_value_as = { name = tca_merit_notice_source value = { value = 10 subtract = var:tca_merit_grant_source } } }
                save_scope_value_as = { name = tca_merit_notice_target value = { value = 10 subtract = var:tca_merit_grant_target } }
                save_scope_value_as = { name = tca_merit_notice_cross value = 0 }
                if = { limit = { OR = { var:tca_merit_grant_source = 2 var:tca_merit_grant_source = 4 var:tca_merit_grant_source = 6 var:tca_merit_grant_mode = 2 } } save_scope_value_as = { name = tca_merit_notice_cross value = 1 } }
        set_variable = { name = tca_merit_grant_notice_ready value = 1 }
    }
}

# 人物作用域；明确恢复入口，不重授或回滚。仅重开当前隔离token的审计，仍先确认真实等级。
tca_merit_resume_grant_audit_effect = {
    if = {
        limit = { var:tca_merit_grant_token ?= $TOKEN$ var:tca_merit_grant_phase ?= 3 }
        set_variable = { name = tca_merit_grant_phase value = 2 }
        set_variable = { name = tca_merit_grant_audit_attempts value = 0 }
        tca_merit_audit_grant_effect = { TOKEN = $TOKEN$ }
    }
}
