# 官员身份沿用原生功绩与兵士权限；办公经费改由地方国库按契约比例拨付。
# 官职身份；scope:subject 为官员，仅正式同步效果临时设置更新标记时允许修改。
# 提刑、通判按官职互斥显示；只保留实际实现的两类官职。
tca_official_obligations = {
	display_mode = tree
	defaults_to_highest_valid_level = yes
	is_shown = {
		scope:subject = {
			tca_is_landless_official_trigger = yes
		}
	}
	# 脚本修改义务时，引擎也检查此条件。
	# 只有同步维护效果临时打开此门槛。
	can_be_changed = {
		scope:subject = { has_character_flag = tca_updating_official_contract }
	}
	obligation_levels = {
		tca_official_none = {
			position = { 0 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_standard_administration.dds"
			gui_tags = { civilian }
			default = yes

			ai_liege_desire = {
				value = 1
			}
			ai_subject_desire = 0

			score = 1
			color = { 0.7 0.7 0.7 1.0 }

			subject_modifier = {
			}

			flag = tca_official_obligation
		}
		tca_official_judicial = {
			position = { 1 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_standard_administration.dds"
			gui_tags = { civilian }
			parent = tca_official_none

			ai_liege_desire = {
				value = 2
			}
			ai_subject_desire = 0

			# 提刑使用20点任职功绩；与通判档位共用界面位置，显示条件互斥。
			is_shown = { scope:subject = { primary_title = { has_variable = tca_judicial_title } } }
			is_valid = {
				scope:subject = {
					tca_is_judicial_official_trigger = yes
				}
			}

			score = 2
			color = { 0.7 0.7 0.7 1.0 }

			enable_title_maa = no

			subject_modifier = {
				monthly_merit = 20
			}

			flag = tca_official_obligation
			flag = tca_official_judicial_obligation
			appointment_trait_flag = civilian_province
		}
		# 通判保持兵士限制，基础功绩为10；原生索引为2。
		tca_official_tongpan = {
			position = { 1 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_standard_administration.dds"
			gui_tags = { civilian }
			parent = tca_official_none
			ai_liege_desire = { value = 2 }
			ai_subject_desire = 0
			is_shown = { scope:subject = { primary_title = { has_variable = tca_tongpan_title } } }
			is_valid = {
				scope:subject = {
					tca_is_tongpan_official_trigger = yes
				}
			}
			score = 2
			color = { 0.7 0.7 0.7 1.0 }
			enable_title_maa = no
			subject_modifier = { monthly_merit = 10 }
			flag = tca_official_obligation
			flag = tca_official_tongpan_obligation
			appointment_trait_flag = civilian_province
		}
	}
}
