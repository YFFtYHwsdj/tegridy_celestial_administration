tca_official_obligations = {
	display_mode = tree
	defaults_to_highest_valid_level = yes
	is_shown = {
		scope:subject = {
			tca_is_landless_official_trigger = yes
		}
	}
	# The engine also checks this trigger for scripted obligation changes.

	can_be_changed = {
		scope:subject = { has_character_flag = tca_updating_official_contract }
	}
	obligation_levels = {
		tca_official_none = {
			position = { 0 0 }

			gui_tags = { civilian }
			default = yes

			ai_liege_desire = {
				value = 1
			}
			ai_subject_desire = 0

			score = 1
			color = { 0.7 0.7 0.7 1.0 }

			subject_modifier = {
			}

			flag = tca_official_obligation
		}
		tca_official_civil = {
			position = { 1 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_standard_administration.dds"
			gui_tags = { civilian }
			parent = tca_official_none

			ai_liege_desire = {
				value = 2
			}
			ai_subject_desire = 0

			is_valid = {
				scope:subject = {
					tca_is_civil_official_trigger = yes
				}
			}

			score = 2
			color = { 0.7 0.7 0.7 1.0 }

			enable_title_maa = no

			subject_modifier = {
				monthly_merit = 20
				monthly_treasury_from_ministry_budget_base = 1
			}

			flag = tca_official_obligation
			flag = tca_official_civil_obligation
			appointment_trait_flag = civilian_province
		}
		tca_official_military = {
			position = { 2 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_military_administration.dds"
			gui_tags = { military }
			parent = tca_official_none

			ai_liege_desire = {
				value = 3
			}
			ai_subject_desire = 0

			is_valid = {
				scope:subject = {
					tca_is_military_official_trigger = yes
				}
			}

			score = 3
			color = { 0.7 0.7 0.7 1.0 }

			enable_title_maa = yes

			subject_modifier = {
				monthly_merit = 20
				men_at_arms_recruitment_cost = -0.5

				monthly_treasury_from_military_budget_base = 2.5
				monthly_treasury_from_ministry_budget_base = 1
			}

			flag = tca_official_obligation
			flag = tca_official_military_obligation
			appointment_trait_flag = military_province
		}
	}
}
