# B阶段四类边境战争；使用A阶段计分参数，保留独立A探针不变。
# 互动承担费用；CB只承担战争目标、胜负效果和主战双方停战。
tca_frontier_county_raid_cb = {
    icon = county_conquest_cb
    group = tca_frontier_probe
    target_titles = all
    target_title_tier = county
    target_de_jure_regions_above = yes
    use_de_jure_wargoal_only = yes
    target_top_liege_if_outside_realm = no
    combine_into_one = yes
    mutually_exclusive_titles = { always = yes }
    allowed_for_character = { debug_only = yes has_character_flag = tca_frontier_border_war_enabled OR = { is_ai = no has_character_flag = tca_frontier_scripted_war_authorized } }
    allowed_against_character = { tca_frontier_border_war_parties_trigger = { ACTOR = scope:attacker DEFENDER = scope:defender } }
    valid_to_start = { scope:target = { tca_frontier_border_war_target_trigger = { ACTOR = scope:attacker DEFENDER = scope:defender } } }
    should_invalidate = { NOT = { any_in_list = { list = target_titles
        OR = { tca_frontier_border_war_county_trigger = { DEFENDER = scope:defender } any_in_de_jure_hierarchy = { tca_frontier_border_war_county_trigger = { DEFENDER = scope:defender } } }
    } } }
    on_primary_attacker_death = invalidate
    on_primary_defender_death = invalidate
    truce_days = 730
    cost = { prestige = 0 }
    attacker_ticking_warscore_delay = { months = 3 }
    defender_ticking_warscore_delay = { months = 6 }
    attacker_ticking_warscore = 0.5
    defender_ticking_warscore = 0.2
    max_attacker_score_from_battles = 20
    max_defender_score_from_battles = 40
    # 占领与战斗合计最高80分；俘虏领袖保留原生满分例外，仍允许自愿议和。
    max_attacker_score_from_occupation = 60
    max_defender_score_from_occupation = 40
    full_occupation_by_attacker_gives_victory = no
    full_occupation_by_defender_gives_victory = no
    # 不禁用俘虏计分：原版开关未阻止本CB主防守者被俘满分，按已确认设计保留原生行为。
    attacker_capital_gives_war_score = no
    defender_capital_gives_war_score = no
    attacker_wargoal_percentage = 1.0
    defender_wargoal_percentage = 1.0
    # 脚本start_war不会自动执行普通宣战处罚；原生effect仅在此调用一次。
    on_declaration = { on_declared_war = yes war = { tca_fs_start_effect = yes } }
    on_victory_desc = tca_frontier_raid_victory
    on_victory = {
        war = { tca_fs_finish_effect = { RESULT = 1 } }
        tca_frontier_border_war_victory_effect = { RAID = yes }
        scope:attacker = { add_truce_one_way = { name = tca_frontier_border_truce character = scope:defender years = 2 result = victory } }
    }
    on_white_peace_desc = tca_frontier_border_white_peace
    on_white_peace = { war = { tca_fs_finish_effect = { RESULT = 2 } } scope:attacker = { add_truce_one_way = { name = tca_frontier_border_truce character = scope:defender years = 2 result = white_peace } } }
    on_defeat_desc = tca_frontier_border_defeat
    on_defeat = { war = { tca_fs_finish_effect = { RESULT = 3 } } scope:attacker = { add_truce_one_way = { name = tca_frontier_border_truce character = scope:defender years = 2 result = defeat } } }
    on_invalidated_desc = tca_frontier_border_invalidated
    on_invalidated = { war = { tca_fs_finish_effect = { RESULT = 4 } } }
    war_name = tca_frontier_county_raid_war
    cb_name = tca_frontier_county_raid_cb
}
