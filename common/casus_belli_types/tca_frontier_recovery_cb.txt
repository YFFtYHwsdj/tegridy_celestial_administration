# D独立追讨CB；只能由失地记录同步授权启动，目标严格为该次实际转移的县。
# 不复用B征服结算，不发起C动员，不再次派生D。无额外威望费，胜利直接归还原主。
tca_frontier_recovery_cb = {
    icon = county_conquest_cb
    group = tca_frontier_recovery
    target_titles = all
    target_title_tier = county
    target_top_liege_if_outside_realm = no
    combine_into_one = yes
    use_de_jure_wargoal_only = yes
    mutually_exclusive_titles = { always = no }
    allowed_for_character = { has_variable = tca_fd_authorized }
    allowed_against_character = { scope:attacker.var:tca_fd_authorized ?= { var:tca_fd_enemy = scope:defender } }
    valid_to_start = {
        scope:attacker.var:tca_fd_authorized ?= { is_target_in_variable_list = { name = tca_fd_goals target = scope:target } }
    }
    should_invalidate = { NOT = { war = {
        exists = var:tca_fd_record
        OR = {
            tca_ws_recovery_transition_trigger = yes
            var:tca_fd_record ?= { OR = { has_variable = tca_ws_inheritance_grace tca_fd_land_valid_trigger = yes } }
        }
    } } }
    on_primary_attacker_death = inherit
    on_primary_defender_death = inherit
    # 原版_casus_belli.info提供的继承校验开关：继承已有战争不重查一次性的声明授权。
    # 原国、实际失地、敌我与身份仍由本战记录校验；新战争资格继续只接受临时声明授权。
    check_attacker_inheritance_validity = no
    check_defender_inheritance_validity = no
    attacker_allies_inherit = yes
    truce_days = 1825
    cost = { prestige = 0 }
    attacker_ticking_warscore_delay = { months = 3 }
    defender_ticking_warscore_delay = { months = 6 }
    attacker_ticking_warscore = 0.5
    defender_ticking_warscore = 0.2
    max_attacker_score_from_battles = 20
    max_defender_score_from_battles = 40
    max_attacker_score_from_occupation = 60
    max_defender_score_from_occupation = 40
    full_occupation_by_attacker_gives_victory = no
    full_occupation_by_defender_gives_victory = no
    attacker_capital_gives_war_score = no
    defender_capital_gives_war_score = no
    attacker_wargoal_percentage = 1.0
    defender_wargoal_percentage = 1.0
    on_declaration = { tca_fd_declaration_effect = yes }
    on_victory_desc = tca_frontier_recovery_victory
    on_victory = { hidden_effect = { tca_fd_victory_effect = yes scope:attacker = { add_truce_one_way = { character = scope:defender years = 5 name = tca_frontier_border_truce result = victory } } } }
    on_white_peace_desc = tca_frontier_recovery_failed
    on_white_peace = { hidden_effect = { tca_fd_failed_effect = yes scope:attacker = { add_truce_one_way = { character = scope:defender years = 5 name = tca_frontier_border_truce result = white_peace } } } }
    on_defeat_desc = tca_frontier_recovery_failed
    on_defeat = { hidden_effect = { tca_fd_failed_effect = yes scope:attacker = { add_truce_one_way = { character = scope:defender years = 5 name = tca_frontier_border_truce result = defeat } } } }
    on_invalidated_desc = tca_frontier_recovery_failed
    on_invalidated = { hidden_effect = { tca_fd_failed_effect = yes } }
    war_name = tca_frontier_recovery_war
    cb_name = tca_frontier_recovery_cb
}
