# No character root: observers and menus have no valid local player.

    scope = none
    saved_scopes = { tca_tongpan_hud_officer }

        exists = scope:tca_tongpan_hud_officer
        scope:tca_tongpan_hud_officer ?= { tca_can_open_tongpan_diarchy_trigger = yes }
    }

    }
