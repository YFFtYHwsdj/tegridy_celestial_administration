# 通判界面可见性桥接；调用 GUI 显式传入 saved scopes，本文件只判断显示，不打开或切换窗口。
# 不使用角色 root：观察者和菜单没有有效本地玩家。
tca_tongpan_hud = {
    scope = none
    saved_scopes = { tca_tongpan_hud_officer }
    is_shown = {
        exists = scope:tca_tongpan_hud_officer
        scope:tca_tongpan_hud_officer ?= { tca_can_open_tongpan_diarchy_trigger = yes }
    }
}

# 使命提示以 Mandate 上下文实例化，必须显式传入所有者。

    is_shown = {

    }
