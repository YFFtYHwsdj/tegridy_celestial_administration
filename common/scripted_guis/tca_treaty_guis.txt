    is_valid = { this = scope:tca_treaty_local_player  }
    effect = { if = { limit = { this = scope:tca_treaty_local_player  } tca_treaty_gui_clear_effect = yes } }
}
# 外交条目只传双方人物；从所看人物的正式索引重新查找，不依赖GUI提取story对象。
tca_treaty_gui_open_record = {
    scope = character
    saved_scopes = { tca_treaty_local_player tca_treaty_row_a tca_treaty_row_b tca_treaty_viewed }
    is_valid = {
        this = scope:tca_treaty_local_player
        exists = scope:tca_treaty_row_a
        exists = scope:tca_treaty_row_b
        exists = scope:tca_treaty_viewed
        scope:tca_treaty_viewed = {
            any_in_list = { variable = tca_treaties
                var:tca_treaty_a = scope:tca_treaty_row_a
                var:tca_treaty_b = scope:tca_treaty_row_b
            }
        }
    }
    effect = {
        if = { limit = {
            this = scope:tca_treaty_local_player
            exists = scope:tca_treaty_row_a
            exists = scope:tca_treaty_row_b
            exists = scope:tca_treaty_viewed
        }
            scope:tca_treaty_viewed = {
                every_in_list = { variable = tca_treaties
                    limit = {
                        var:tca_treaty_a = scope:tca_treaty_row_a
                        var:tca_treaty_b = scope:tca_treaty_row_b
                    }
                    save_scope_as = tca_treaty_record
                }
            }
            if = { limit = { exists = scope:tca_treaty_record }
                scope:tca_treaty_record = {
                    # 本国君主优先以其缔约身份选择对方；第三国查看者沿人物页方向查看。
                    if = { limit = { var:tca_treaty_a = root } var:tca_treaty_b = { save_scope_as = recipient } }
                    else_if = { limit = { var:tca_treaty_b = root } var:tca_treaty_a = { save_scope_as = recipient } }
                    else_if = { limit = { var:tca_treaty_a = scope:tca_treaty_viewed } var:tca_treaty_b = { save_scope_as = recipient } }
                    else = { var:tca_treaty_a = { save_scope_as = recipient } }
                }
                tca_treaty_gui_open_effect = yes
            }
        }
    }
}
