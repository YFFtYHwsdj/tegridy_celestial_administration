# GUI操作均以本地玩家为root；刷新检查资格，关闭始终允许且仅隐藏本人窗口。
# 无玩家的观察者／菜单仅检查显式作用域是否存在，避免对空root读取变量。
tca_confucian_statistics_visible_gui = {
    scope = none
    saved_scopes = { tca_stats_player }
    is_shown = {
        debug_only = yes
        exists = scope:tca_stats_player
        scope:tca_stats_player ?= { var:tca_stats_visible ?= 1 top_liege = { government_has_flag = government_is_celestial has_variable = tca_stats_updated } }
    }
}
tca_refresh_confucian_statistics_gui = {
    scope = character
    is_valid = { tca_can_view_confucian_statistics_trigger = yes }
    effect = { tca_force_confucian_statistics_effect = yes }
}
tca_close_confucian_statistics_gui = {
    scope = character
    effect = { tca_close_confucian_statistics_effect = yes }
}
