# 原契约页桥接：root为契约官员，actor为本地玩家；使用引擎已登记的原生actor作用域。
# 只打开普通互动；资格由共享权限检查，冷却由目标互动显示。
tca_open_office_budget_gui = {
 scope = character
 is_shown = { tca_is_landless_official_trigger = yes OR = { this = scope:actor tca_can_edit_admin_contract_trigger = { EDITOR = scope:actor } } }
 effect = {
  if = { limit = { this = scope:actor }
   open_interaction_window = { interaction = tca_request_office_budget_interaction actor = root recipient = root.liege redirect = no }
  }
  else_if = { limit = { tca_is_landless_official_trigger = yes tca_can_edit_admin_contract_trigger = { EDITOR = scope:actor } }
   open_interaction_window = { interaction = tca_adjust_office_budget_interaction actor = scope:actor recipient = root redirect = no }
  }
 }
}
