# 本地玩家窗口桥接；root和显式登记的tca_frontier_local_player必须是同一人物，业务由共享effect承担。
tca_frontier_window_open_gui = {
    scope = character
    saved_scopes = { tca_frontier_local_player }
    is_valid = { this = scope:tca_frontier_local_player }
    effect = { debug_log = "TCA_FRONTIER_GUI SCRIPTED_GUI_EXECUTED" if = { limit = { this = scope:tca_frontier_local_player } tca_frontier_window_open_effect = yes } }
}
