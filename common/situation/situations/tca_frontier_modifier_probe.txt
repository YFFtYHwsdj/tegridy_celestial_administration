# 仅由独立原生探针创建，不接入正常游戏开局或正式北疆阶段。
# 控制阶段无修正；两个实验阶段分别验证指定组与all的county_modifier。
tca_frontier_modifier_probe = {
    is_unique = yes
    icon = { reference = "gfx/interface/icons/situation_types/silk_road_situation.dds" }
    start_phase = tca_frontier_modifier_probe_control
    sub_regions = {
        tca_frontier_modifier_probe_region = {
            geographical_regions = { tca_frontier_region }
            start_phase = tca_frontier_modifier_probe_control
        }
    }
    participant_groups = {
        tca_frontier_modifier_probe_owner = {
            auto_add_rulers = yes
            auto_add_landless_rulers = no
            require_realm_in_sub_region = no
            is_character_valid = {
                has_character_flag = tca_frontier_modifier_probe_test_owner
                capital_county = { title_province = { geographical_region = tca_frontier_region } }
            }
        }
        # 同一人物实际迁都后自动改组；不由事件强制指定参与组。
        tca_frontier_modifier_probe_southern_owner = {
            auto_add_rulers = yes
            auto_add_landless_rulers = no
            require_realm_in_sub_region = no
            is_character_valid = {
                has_character_flag = tca_frontier_modifier_probe_test_owner
                capital_county = { title_province = { NOT = { geographical_region = tca_frontier_region } } }
            }
        }
    }
    phases = {
        tca_frontier_modifier_probe_control = { }
        tca_frontier_modifier_probe_group = {
            modifier_sets = {
                tca_frontier_modifier_probe_group_set = {
                    tca_frontier_modifier_probe_owner = {
                        county_modifier = { monthly_county_control_growth_factor = 7 }
                    }
                    tca_frontier_modifier_probe_southern_owner = {
                        county_modifier = { monthly_county_control_growth_factor = 13 }
                    }
                }
            }
        }
        tca_frontier_modifier_probe_all = {
            modifier_sets = {
                tca_frontier_modifier_probe_all_set = {
                    all = {
                        county_modifier = { monthly_county_control_growth_factor = 11 }
                    }
                }
            }
        }
    }
}
