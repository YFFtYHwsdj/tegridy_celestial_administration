# 天朝地方军事内阁动态称谓
# 当前character可为任职者或官署主君；读取本人或其军事内阁任职者的差遣输出称号。
# 无匹配时回退本语言默认称号；实际界面根作用域与空缺显示仍须人工验收。
tca_military_marshal_title = {
	type = character

	text = {
		trigger = {
			OR = {
				has_court_position = tca_military_supervisor_court_position
				cp:councillor_marshal ?= {
					has_court_position = tca_military_supervisor_court_position
				}
			}
		}
		localization_key = tca_military_supervisor_court_position
	}

	text = {
		trigger = {
			OR = {
				has_court_position = tca_military_inspector_court_position
				cp:councillor_marshal ?= {
					has_court_position = tca_military_inspector_court_position
				}
			}
		}
		localization_key = tca_military_inspector_court_position
	}

	text = {
		trigger = {
			OR = {
				has_court_position = tca_corps_inspector_court_position
				cp:councillor_marshal ?= {
					has_court_position = tca_corps_inspector_court_position
				}
			}
		}
		localization_key = tca_corps_inspector_court_position
	}

	text = {
		fallback = yes
		localization_key = tca_military_marshal_fallback_sima
	}
}

# 原版有期限保护显示日期，模块保护随差遣身份结束；仅在模块归属与原生锁一致且仍任对应文武内阁／差遣时改写。
# 本地化从TARGET_CHARACTER调用，当前人物就是被撤换的军事内阁任职者。
tca_military_council_fire_block_reason = {
    type = character
    text = {
        trigger = {
            has_variable = tca_military_council_lock_owner
            has_variable = block_fire_councillor
            var:block_fire_councillor ?= var:tca_military_council_lock_owner
            tca_court_matches_linked_council_trigger = yes
        }
        localization_key = tca_military_council_commission_blocks_firing
    }
    text = { fallback = yes localization_key = tca_military_council_native_blocks_firing }
}
