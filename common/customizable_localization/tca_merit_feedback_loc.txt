# 资格上限显示取 max(资格档位, 当前核定官品)；旧角色现官品高于新档位时按现官品解释，
# 避免出现“现官品四品、资格上限九品”的误读。资格只限制此后的普通升品，已得官品不追溯降低。
# 前置分支只在现官品高于资格档位时命中原官品；其后的资格档位映射不会吞掉更高的现官品。

    text = { trigger = { has_variable = tca_merit_approved_level var:tca_merit_approved_level = 8 tca_merit_theoretical_cap_value < 8 } localization_key = tca_merit_rank_2 }
    text = { trigger = { has_variable = tca_merit_approved_level var:tca_merit_approved_level = 7 tca_merit_theoretical_cap_value < 7 } localization_key = tca_merit_rank_3 }
    text = { trigger = { has_variable = tca_merit_approved_level var:tca_merit_approved_level = 6 tca_merit_theoretical_cap_value < 6 } localization_key = tca_merit_rank_4 }

    text = { trigger = { tca_merit_theoretical_cap_value = 8 } localization_key = tca_merit_rank_2 }
    text = { trigger = { tca_merit_theoretical_cap_value = 7 } localization_key = tca_merit_rank_3 }
    text = { trigger = { tca_merit_theoretical_cap_value = 6 } localization_key = tca_merit_rank_4 }
    text = { trigger = { tca_merit_theoretical_cap_value = 5 } localization_key = tca_merit_rank_5 }

    text = { trigger = { tca_merit_theoretical_cap_value = 3 } localization_key = tca_merit_rank_7 }
    text = { trigger = { tca_merit_theoretical_cap_value = 2 } localization_key = tca_merit_rank_8 }
    text = { trigger = { tca_merit_theoretical_cap_value = 1 } localization_key = tca_merit_rank_9 }
    text = { localization_key = tca_merit_rank_unknown }
