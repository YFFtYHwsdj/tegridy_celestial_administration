# 官库留储法四档。仅限天朝独立君主持有，常额为开局默认；其余档位须付改制代价。
# 三档只改保底倍率 L（0.75 / 1.00 / 1.50），不改 24 个月收入上限；倍率由数值层读取。
# 改制条件与冷却直接内联，不另建全局 trigger 文件；冷却变量三档共用。
# 效果层按年读取最高领主当前档位，故此处不须 on_after_pass 同步。
tca_treasury_reserve_strict_law = {
    law_group_type = tca_treasury_reserve_law_group
    index = 0
    flag = tca_treasury_reserve_strict
    can_have = {
        government_has_flag = government_is_celestial
        is_independent_ruler = yes
    }
    can_keep = {
        government_has_flag = government_is_celestial
        is_independent_ruler = yes
    }
    can_realm_have = {
        government_has_flag = government_is_celestial
        is_independent_ruler = yes
    }
    should_start_with = { always = no }
    can_pass = {
        is_imprisoned = no
        custom_tooltip = { text = tca_treasury_reserve_reform_cooldown NOT = { has_variable = tca_treasury_reserve_reform_cooldown } }
    }
    pass_cost = { prestige = 2000 influence = 1000 }
    on_pass = { hidden_effect = { set_variable = { name = tca_treasury_reserve_reform_cooldown years = 10 } } }
    ai_will_do = { value = 0 }
}
