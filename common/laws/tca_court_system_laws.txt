# 地方官名、中央官名和司法机构分为独立法律组；对象按默认／宋式或停用／启用成对声明。
# 官名使用即时刷新和次日补刷；司法机构在替换法律后维护，旧合并法律仅供存档迁移。
tca_local_names_default_law = {
    law_group_type = tca_local_names_law
    index = 0
    flag = tca_local_names_default
    can_have = { government_has_flag = government_is_celestial is_independent_ruler = yes tca_external_official_names_trigger = no }
    # 外部模组接管时保留休眠的选择。
    can_keep = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    can_realm_have = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    should_start_with = { NOT = { tca_should_start_with_song_system_trigger = yes } }
    can_pass = { tca_names_reform_can_pass_trigger = yes }
    pass_cost = { prestige = 200 }
    on_pass = { hidden_effect = {
        tca_refresh_court_system_realm_effect = yes
        trigger_event = { id = tca_court_system.1 days = 1 }
    } }
    ai_will_do = { value = 0 }
}

# 司法机构分组：启停共享十年改革冷却，维护读取替换后的法律以退休或恢复机构。
tca_judicial_institutions_disabled_law = {
    law_group_type = tca_judicial_institutions_law
    index = 0
    flag = tca_judicial_institutions_disabled
    can_have = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    can_keep = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    can_realm_have = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    should_start_with = { NOT = { tca_should_start_with_song_system_trigger = yes } }
    can_pass = { tca_judicial_reform_can_pass_trigger = yes }
    pass_cost = { prestige = 2000 influence = 1000 }
    on_pass = { hidden_effect = {
        set_variable = { name = tca_court_system_cooldown years = 10 }
    } }
    # on_pass 仍读取旧法律；只在法律替换后执行维护。
    on_after_pass = { hidden_effect = {
        tca_refresh_court_system_realm_effect = yes
        trigger_event = { id = tca_court_system.1 days = 1 }
    } }
    ai_will_do = { value = 0 }
}
