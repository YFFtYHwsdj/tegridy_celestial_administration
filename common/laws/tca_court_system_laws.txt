    law_group_type = tca_local_names_law
    index = 0
    flag = tca_local_names_default
    can_have = { government_has_flag = government_is_celestial is_independent_ruler = yes tca_external_official_names_trigger = no }
    # Preserve dormant choices when external mods take over.
    can_keep = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    can_realm_have = { government_has_flag = government_is_celestial is_independent_ruler = yes }
    should_start_with = { NOT = { tca_should_start_with_song_system_trigger = yes } }
    can_pass = { tca_names_reform_can_pass_trigger = yes }
    pass_cost = { prestige = 200 }
    on_pass = { hidden_effect = {
        tca_refresh_court_system_realm_effect = yes
        trigger_event = { id = tca_court_system.1 days = 1 }
    } }
    ai_will_do = { value = 0 }
}
