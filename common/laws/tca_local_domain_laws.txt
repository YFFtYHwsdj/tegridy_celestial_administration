    law_group_type = tca_local_domain_policy_law
    index = 0
    can_have = { tca_local_domain_sovereign_trigger = yes }
    can_keep = { tca_local_domain_sovereign_trigger = yes }
    can_realm_have = { tca_local_domain_sovereign_trigger = yes }
    should_start_with = { always = yes }
    can_pass = { tca_local_domain_reform_can_pass_trigger = yes }
    pass_cost = { prestige = 2000 influence = 1000 }
    on_pass = { hidden_effect = { set_variable = { name = tca_local_domain_reform_cooldown years = 10 } } }
    on_after_pass = { hidden_effect = { tca_sync_local_domain_realm_effect = yes } }
    ai_will_do = { value = 0 }
}
