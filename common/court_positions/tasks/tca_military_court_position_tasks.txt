tca_military_supervisor_drill_levies = {
    court_position_types = { tca_military_supervisor_court_position tca_military_inspector_court_position }
    cost = {
        round = no
        gold = { value = 0 if = { limit = { scope:liege = { has_treasury = no } } add = monthly_court_position_task_cost } }
        treasury = { value = 0 if = { limit = { scope:liege = { has_treasury = yes } } add = monthly_court_position_task_cost } }
    }
    scaling_employer_modifiers = {
        terrible = { levy_reinforcement_rate = 0.06 }
        poor = { levy_reinforcement_rate = 0.12 }
        average = { levy_reinforcement_rate = 0.18 }
        good = { levy_reinforcement_rate = 0.24 }
        excellent = { levy_reinforcement_rate = 0.3 }
    }
    # 员工作用域；原生启停维护本项状态，月度补标记兼容旧档。
    on_start = { set_variable = tca_court_paid_task_active set_variable = tca_court_replenishing_levies }
    on_end = { remove_variable = tca_court_paid_task_active remove_variable = tca_court_replenishing_levies }
    on_monthly = { set_variable = tca_court_paid_task_active set_variable = tca_court_replenishing_levies }
    ai_will_do = { value = tca_military_supervisor_drill_levies_ai_score }
}
