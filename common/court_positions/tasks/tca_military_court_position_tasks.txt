	cost = {

		}

	scaling_employer_modifiers = {
		terrible = {
			levy_reinforcement_rate = 0.05
		}

			levy_reinforcement_rate = 0.10
			army_toughness_mult = 0.01
		}
		average = {
			levy_reinforcement_rate = 0.15
			army_toughness_mult = 0.02
		}
		good = {
			levy_reinforcement_rate = 0.20
			army_toughness_mult = 0.02
		}
		excellent = {
			levy_reinforcement_rate = 0.25
			army_toughness_mult = 0.03
		}
	}

	# 原生任务生命周期维护在员工身上；月度补标记兼容旧档中已运行的任务。
	on_start = { set_variable = tca_court_paid_task_active }
	on_end = { remove_variable = tca_court_paid_task_active }
	on_monthly = { set_variable = tca_court_paid_task_active }

	ai_will_do = { value = tca_military_supervisor_drill_levies_ai_score }
}
