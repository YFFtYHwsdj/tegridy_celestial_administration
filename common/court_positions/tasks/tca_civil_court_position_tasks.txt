tca_court_encourage_farming = {
    court_position_types = { tca_household_registrar_court_position }
    cost = {
        round = no
        gold = {
            value = 0
            if = { limit = { scope:liege = { has_treasury = no } } add = monthly_court_position_task_cost }
        }
        treasury = {
            value = 0
            if = { limit = { scope:liege = { has_treasury = yes } } add = monthly_court_position_task_cost }
        }
    }
    scaling_employer_modifiers = {
        terrible = { development_growth_factor = 0.05 }
        poor = { development_growth_factor = 0.1 }
        average = { development_growth_factor = 0.15 }
        good = { development_growth_factor = 0.2 }
        excellent = { development_growth_factor = 0.25 }
    }
    # 原生任务生命周期维护在员工身上；月度补标记兼容旧档中已运行的任务。
    on_start = { set_variable = tca_court_paid_task_active }
    on_end = { remove_variable = tca_court_paid_task_active }
    on_monthly = { set_variable = tca_court_paid_task_active }

    ai_will_do = { value = tca_court_encourage_farming_ai_score }
}
