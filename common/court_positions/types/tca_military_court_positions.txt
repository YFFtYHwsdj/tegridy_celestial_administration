# 军事宫廷职位与地方军事内阁兼任入口
#
# 本文件维护天朝三大军事差遣宫廷职位定义：
# 1. 兵马监押（tca_military_supervisor_court_position）：仅伯爵级实土地方官署（知州、知府、知军）
# 2. 兵马都监（tca_military_inspector_court_position）：公爵级及以上实土地方官署，排除枢密院与高级军署
# 3. 都虞候（tca_corps_inspector_court_position）：枢密院与王国级及以上军制省、都护府
#
# 只接受系统授权除授；AI基础分为0，由季度铨选统一补缺。

tca_military_supervisor_court_position = {
	sort_order = 285
	max_available_positions = 1
	minimum_rank = county
	skill = martial

	court_position_asset = {
		trigger = {
			government_has_flag = government_is_celestial
		}
		animation = throne_room_two_handed_passive_1
		background = "gfx/interface/illustrations/event_scenes/tgp_courtyard_asia.dds"
	}

	opinion = {
		value = regular_court_position_opinion
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	# 称职度：军事为主、管理为辅，勇武作为次要加分项；任职能力门槛另由资格trigger检查。
	aptitude = {
		value = 1
		add = {
			value = martial
			multiply = 2.0
			max = 40
			desc = court_position_skill_martial
		}
		add = {
			value = stewardship
			multiply = 1.0
			max = 20
			desc = court_position_skill_stewardship
		}
		add = {
			value = prowess
			multiply = 0.5
			max = 10
			desc = court_position_skill_prowess
		}
		if = {
			limit = { has_trait = diligent }
			add = { value = 5 desc = court_position_diligent_trait }
		}
		if = {
			limit = { has_trait = brave }
			add = { value = 5 desc = court_position_brave_trait }
		}
		if = {
			limit = { has_trait = lazy }
			add = { value = -5 desc = court_position_lazy_trait }
		}
		if = {
			limit = { has_trait = craven }
			add = { value = -5 desc = court_position_craven_trait }
		}
		if = {
			limit = { has_trait = greedy }
			add = { value = -5 desc = court_position_greedy_trait }
		}
		add = court_position_aptitude_low_penalty_value
		add = general_aptitude_bonus
	}

	is_shown = {
		tca_military_supervisor_office_valid_trigger = yes
	}

	valid_position = {
		tca_military_supervisor_office_valid_trigger = yes
	}

	is_shown_character = {
		scope:employee = {
			is_courtier_of = scope:liege
			court_position_employee_shown_trigger = yes
		}
	}

	# 本署在籍待阙资格；只豁免合法军事内阁兼任，其他全职岗位继续排他。
	valid_character = {
		scope:employee = {
			tca_military_supervisor_validity_trigger = yes
		}
	}

	on_court_position_received = {
		tca_military_court_position_on_received_effect = yes
	}

	on_court_position_revoked = {
		tca_military_court_position_on_revoked_effect = yes
	}

	on_court_position_invalidated = {
		tca_military_court_position_on_invalidated_effect = yes
	}

	# 自然离廷与死亡触发vacated；失效可能随后再次触发，清理须幂等。
	on_court_position_vacated = {
		tca_military_court_position_on_vacated_effect = yes
	}

	salary = {
		gold = {
			value = 0
			if = {
				limit = {
					scope:liege = { has_treasury = no }
				}
				add = tca_court_salary_1
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					scope:liege = { has_treasury = yes }
				}
				add = tca_court_salary_1
			}
		}
		round = no
	}

	received_salary = {
		gold = { value = tca_court_salary_1 }
		round = no
	}

	revoke_cost = {
		influence = {
			value = tca_minor_court_position_influence_revoke_cost
			if = {
				limit = { free_court_position_revoke_trigger = { POS = tca_military_supervisor } }
				multiply = 0
			}
		}
	}

	# 常驻基础效果（优秀档标定：控制力增长+25%，征召补员+10%，防御优势+2）
	base_employer_modifier = {
	}

	# 任职人物修正：固定月度考绩、军事生活方式经验及声望，随职位解除而移除。
	modifier = {
		monthly_merit = 1.5
		monthly_martial_lifestyle_xp_gain_mult = 0.05
		monthly_prestige = 0.05
	}

	scaling_employer_modifiers = {
		terrible = {
			monthly_county_control_growth_factor = 0.05
			levy_reinforcement_rate = 0.02
		}
		poor = {
			monthly_county_control_growth_factor = 0.10
			levy_reinforcement_rate = 0.04
			defender_advantage = 1
		}
		average = {
			monthly_county_control_growth_factor = 0.15
			levy_reinforcement_rate = 0.06
			defender_advantage = 1
		}
		good = {
			monthly_county_control_growth_factor = 0.20
			levy_reinforcement_rate = 0.08
			defender_advantage = 2
		}
		excellent = {
			monthly_county_control_growth_factor = 0.25
			levy_reinforcement_rate = 0.10
			defender_advantage = 2
		}
	}

	ai_position_score = {
		value = 0
	}

	ai_candidate_score = {
		value = 100
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}
