namespace = tca_retirement_player_boot
tca_retirement_player_boot.1 = {
    type = character_event hidden = yes
    immediate = {
        if = { limit = { title:h_china = { NOT = { has_variable = tca_retirement_human_boot_started } } }
            title:h_china = { set_variable = { name = tca_retirement_human_boot_started value = 1 } }
            if = { limit = { debug_only = yes is_ai = no title:h_china.holder = root }
                debug_log = "TCA_RETIREMENT_HUMAN INITIAL_PLAYER_CONFIRMED"
                trigger_event = tca_retirement_session.1
            }
            else = { debug_log = "TCA_RETIREMENT_HUMAN PRECONDITION_FAIL_NOT_HUMAN" }
        }
    }
}
