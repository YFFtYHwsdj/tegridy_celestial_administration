on_councillors_swapped = {

	effect = {
		# swap ministry titles if the two councillors were ministers
		if = {
			limit = {
				scope:councillor_liege = {
					tgp_has_access_to_ministry_trigger = yes
				}
			}

			create_title_and_vassal_change = {
				type = appointment
				save_scope_as = minister_title_change
				add_claim_on_loss = no
			}

			# root: lose current title, gain new title
			primary_title = {
				change_title_holder = {
					holder = scope:councillor
					change = scope:minister_title_change
				}
			}
			scope:councillor.primary_title = {
				change_title_holder = {
					holder = root
					change = scope:minister_title_change
				}
			}

			resolve_title_and_vassal_change = scope:minister_title_change
		}
	}
}
