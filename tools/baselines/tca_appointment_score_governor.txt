appointment_score_governor = {
	if = {
		limit = { is_at_war_with = liege }
		add = {
			value = -9999
			desc = game_concept_revolt
		}
	}
	if = {
		limit = { has_trait = governor }
		add = {
			value = 5
			if = {
				limit = {
					has_trait_xp = { trait = governor value >= 25 }
				}
				add = 5
			}
			if = {
				limit = {
					has_trait_xp = { trait = governor value >= 50 }
				}
				add = 5
			}
			if = {
				limit = {
					has_trait_xp = { trait = governor value >= 75 }
				}
				add = 5
			}
			desc = governor_trait_desc_short
		}
	}
	# RITE BASED
	if = {
		limit = {
			exists = top_liege.primary_title.state_rite
		}
		top_liege.primary_title.state_rite = {
			save_temporary_scope_as = target_rite
		}
	}
	else = {
		top_liege.rite = {
			save_temporary_scope_as = target_rite
		}
	}
	# RITE HOSTILITY
	if = {
		limit = {
			scope:target_rite ?= {
				rite_hostility_level = {
					target = root.rite
					value = faith_astray_level
				}
			}
		} #Change to faith acceptance
		subtract = {
			value = 10
			desc = astray_faith_succession_desc
		}
	}
	if = {
		limit = {
			scope:target_rite ?= {
				rite_hostility_level = {
					target = root.rite
					value = faith_hostile_level
				}
			}
		} #Change to faith acceptance
		subtract = {
			value = 25
			desc = hostile_faith_succession_desc
		}
	}
	if = {
		limit = {
			scope:target_rite ?= {
				rite_hostility_level = {
					target = root.rite
					value >= faith_evil_level
				}
			}
		} #Change to faith acceptance
		subtract = {
			value = 50
			desc = evil_faith_sucession_desc
		}
	}
}
