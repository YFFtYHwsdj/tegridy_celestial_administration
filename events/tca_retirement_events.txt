# 超时清理使用原凭据，不能终止后来申请；纯加恩按原品结案。
tca_retirement.11 = {
    type = character_event hidden = yes
    immediate = {
        if = { limit = { is_alive = yes has_variable = tca_retirement_request_kind var:tca_retirement_request_serial ?= scope:tca_retirement_token }
            # 超时只按原品结案，不再按当前品级走自动加恩，防止过期高品请求降品后误授。
            if = { limit = { var:tca_retirement_request_kind = 4 }
                add_character_flag = tca_retirement_honor_processed
                tca_retirement_offer_transfer_effect = yes
            }
            set_variable = { name = tca_retirement_request_result value = -1 }
            tca_retirement_clear_request_effect = yes
            # 请求结案后再给传家邀请；共用资格禁止未结事务交家门。
            tca_retirement_offer_transfer_effect = yes
            if = { limit = { is_ai = no } trigger_event = tca_retirement.12 }
        }
    }
}
