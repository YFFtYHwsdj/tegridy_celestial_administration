namespace = tca_judicial_titles

# The hook schedules this through a one-day delay so title/government changes settle.
tca_judicial_titles.1 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { is_alive = yes }
            trigger_event = { id = tca_judicial_titles.2 days = 1 }
        }
    }
}

tca_judicial_titles.2 = {
    type = character_event
    hidden = yes
    immediate = { tca_refresh_character_circuits_effect = yes }
}

# Allow the native title/government/house callbacks to settle before reattaching.
tca_judicial_titles.3 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { is_alive = yes }
            tca_refresh_held_judicial_positions_effect = yes
        }
    }
}
