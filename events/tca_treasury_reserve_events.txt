# 官库留储法通知事件；由年度结算在缴款人／皇帝作用域排队，使用独立事件上下文渲染动态金额。
# 仅对 is_ai = no 的接收者发送；AI 不产生消息。冻结的 saved scope value 由触发效果携带。
namespace = tca_treasury_reserve

tca_treasury_reserve.1 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { is_alive = yes is_ai = no }
            send_interface_message = {
                type = tca_treasury_reserve_payer_notice
                left_icon = root
                tooltip = tca_treasury_reserve_payer_notice_tooltip
                desc = { desc = tca_treasury_reserve_payer_notice_desc }
            }
        }
    }
}

tca_treasury_reserve.2 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { is_alive = yes is_ai = no }
            if = {
                limit = { exists = scope:tca_tr_total scope:tca_tr_total > 0 }
                send_interface_message = {
                    type = tca_treasury_reserve_emperor_briefing
                    left_icon = root
                    tooltip = tca_treasury_reserve_emperor_briefing_tooltip
                    desc = { desc = tca_treasury_reserve_emperor_briefing_desc }
                }
            }
            else = {
                send_interface_message = {
                    type = tca_treasury_reserve_emperor_briefing
                    left_icon = root
                    tooltip = tca_treasury_reserve_emperor_briefing_tooltip
                    desc = { desc = tca_treasury_reserve_emperor_briefing_empty_desc }
                }
            }
        }
    }
}
