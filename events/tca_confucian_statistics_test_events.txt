                else = { debug_log = "TCA_TEST PASS tca_test_confucian_statistics" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_passed add = 1 } } }
            }
            else = { debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_confucian_statistics" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } } }
            remove_character_flag = tca_test_case_failed
        }
    }
