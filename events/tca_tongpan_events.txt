# 通判延迟维护事件；由 tca_request_tongpan_maintenance_effect 合并事务期间的重复请求。
namespace = tca_tongpan
# 最高领主角色的次日维护；排队时设置的 pending 在刷新期间保留，完成后清除以允许后续变化再排队。
tca_tongpan.1 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { has_variable = tca_tongpan_annual_only }
            tca_refresh_tongpan_realm_offices_effect = yes
            # 维护中的真实事务回调也可升级退出补漏，pending 保留到本事件结束。
            if = { limit = { NOT = { has_variable = tca_tongpan_annual_only } } tca_cleanup_invalid_tongpan_offices_effect = yes }
        }
        else = { tca_refresh_tongpan_realm_effect = yes }
        remove_variable = tca_tongpan_annual_only
        remove_variable = tca_tongpan_maintenance_pending
    }
}
