tca_retirement_vassal_test.1001 = {
    type = character_event hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = { limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            # 先验证仅持家门的结构；身份在同链加入，不跨日暴露半成品给正式头衔维护。
            scope:tca_rv_actor = { add_trait = tca_fengci }
            tca_test_retirement_vassal_snapshot_effect = { STAGE = stable_family }
            if = { limit = { scope:tca_rv_actor = { liege = root is_landless_ruler = yes tca_retirement_can_transfer_trigger = yes } scope:tca_rv_family.current_heir = scope:tca_rv_heir }
                debug_log = "TCA_TEST CHECK_PASS tca_test_retirement_vassal_limit 2"
            }
            else = { add_character_flag = tca_test_tca_retirement_vassal_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_retirement_vassal_limit 2" }
            set_variable = { name = tca_rv_direct_before value = var:tca_rv_direct }
            set_variable = { name = tca_rv_script_before value = vassal_count }
            # 使用正式入口建成唯一退休公国，仍核对同一对象、家门、领属及即时计数。
            # 原夹具另造未绑定公国，正式回调会合法增建宫观，无法作为唯一退休对象的阳性。
            scope:tca_rv_actor = {
                tca_retirement_finish_effect = { KIND = 1 APPROVE = no }
                var:tca_retirement_title = { save_scope_as = tca_rv_title }
            }
            tca_test_retirement_vassal_snapshot_effect = { STAGE = after_duchy }
            if = { limit = { var:tca_rv_direct = var:tca_rv_direct_before scope:tca_rv_actor = { liege = root is_landless_ruler = yes has_title = scope:tca_rv_title has_title = scope:tca_rv_family } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_retirement_vassal_limit 3"
            }
            else = { add_character_flag = tca_test_tca_retirement_vassal_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_retirement_vassal_limit 3" }
            trigger_event = { id = tca_retirement_vassal_test.1002 days = 3 }
        }
    }
}
tca_retirement_vassal_test.1002 = {
    type = character_event hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = { limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            tca_test_retirement_vassal_snapshot_effect = { STAGE = stable_duchy }
            scope:tca_rv_actor = {
                debug_log = "TCA_RETIREMENT_VASSAL stable_primary=[THIS.Char.GetPrimaryTitle.GetNameNoTier]"
                if = { limit = { has_variable = tca_retirement_title } debug_log = "TCA_RETIREMENT_VASSAL FORMAL_TITLE_CREATED" }
            }
            if = { limit = { scope:tca_rv_actor = { liege = root is_landless_ruler = yes primary_title = scope:tca_rv_title has_title = scope:tca_rv_family } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_retirement_vassal_limit 4"
            }
            else = { add_character_flag = tca_test_tca_retirement_vassal_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_retirement_vassal_limit 4" }
            scope:tca_rv_actor = {
                if = { limit = { tca_retirement_can_transfer_trigger = yes } debug_log = "TCA_RETIREMENT_VASSAL current_transfer=available" }
                else = { debug_log = "TCA_RETIREMENT_VASSAL current_transfer=blocked" }
                # 本例隔离领属与上限计数，显式转交真实家门；正式传家入口另有独立回归。
                # 不调用正式传家worker，也不声称验证了玩家切换或迁廷路径。
                create_title_and_vassal_change = { type = granted add_claim_on_loss = no save_scope_as = tca_rv_transfer }
                scope:tca_rv_family = { change_title_holder = { holder = scope:tca_rv_heir change = scope:tca_rv_transfer } }
                resolve_title_and_vassal_change = scope:tca_rv_transfer
                house = { if = { limit = { NOT = { house_head = scope:tca_rv_heir } } set_house_head = scope:tca_rv_heir } }
            }
            scope:tca_rv_heir = {
                if = { limit = { NOT = { liege = root } }
                    create_title_and_vassal_change = { type = swear_fealty add_claim_on_loss = no save_scope_as = tca_rv_heir_fealty }
                    change_liege = { liege = root change = scope:tca_rv_heir_fealty }
                    resolve_title_and_vassal_change = scope:tca_rv_heir_fealty
                }
            }
            tca_test_retirement_vassal_snapshot_effect = { STAGE = after_transfer }
            trigger_event = { id = tca_retirement_vassal_test.1003 days = 3 }
        }
    }
}
