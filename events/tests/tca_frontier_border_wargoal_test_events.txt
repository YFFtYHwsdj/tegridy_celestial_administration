tca_frontier_border_war_test.34 = {
    type = character_event hidden = yes
    immediate = {
        change_variable = { name = tca_frontier_border_war_test_polls add = 1 }
        if = { limit = {
            exists = scope:tca_frontier_border_war
            scope:tca_frontier_border_war = { attacker_war_score > 80 attacker_war_score < 100 }
        }
            if = { limit = { title:c_hangzhou.title_province = { is_occupied = yes } } debug_log = "TCA_BORDER_GOAL OUTSIDE_OCCUPIED" }
            if = { limit = { scope:recipient = { is_imprisoned_by = title:k_france.holder } } debug_log = "TCA_BORDER_GOAL NEUTRAL_PRISON_YES" }
            else = { debug_log = "TCA_BORDER_GOAL NEUTRAL_PRISON_NO" }
            title:c_lingzhou = { every_direct_de_jure_vassal_title = {
                limit = { tier = tier_barony title_province = { has_holding = yes OR = { is_occupied = no barony = { NOT = { barony_controller = scope:actor } } } } }
                debug_log = "TCA_BORDER_GOAL TARGET_NOT_CONTROLLED [THIS.Title.GetName]"
            } }
            # 总分超过占领与战斗合计上限80，进攻者无俘虏且未占领其他敌县，证明目标控制贡献分数。
            # 中立监禁只用于减少围城抓人干扰；原生AI可能释放囚犯，断言直接检查进攻者没有俘虏。
        if = { limit = { title:c_lingzhou = {
                title_province = { is_occupied = yes barony = { barony_controller = scope:actor } }
                NOT = { any_direct_de_jure_vassal_title = { tier = tier_barony title_province = { has_holding = yes OR = { is_occupied = no barony = { NOT = { barony_controller = scope:actor } } } } } }
            } title:c_hangzhou.title_province = { is_occupied = no } scope:actor = { NOT = { any_prisoner = { always = yes } } } } debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_duchy_wargoal 4" }
        else = { add_character_flag = tca_test_tca_frontier_border_war_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_duchy_wargoal 4" }
            set_variable = { name = tca_frontier_border_war_test_final_score value = scope:tca_frontier_border_war.attacker_war_score }
            if = { limit = { var:tca_frontier_border_war_test_final_score > var:tca_frontier_border_war_test_initial_score }
                debug_log = "TCA_BORDER_GOAL HELD score=[THIS.Char.MakeScope.Var('tca_frontier_border_war_test_final_score').GetValue]"
            }
        trigger_event = { id = tca_frontier_border_war_test.35 days = 2 }
        }
        else_if = { limit = { exists = scope:tca_frontier_border_war var:tca_frontier_border_war_test_polls < 20 }
        trigger_event = { id = tca_frontier_border_war_test.34 days = 15 }
        }
        else = {
        add_character_flag = tca_test_tca_frontier_border_war_test_case_failed
        debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_duchy_wargoal 4"
        trigger_event = { id = tca_frontier_border_war_test.35 days = 2 }
        }
    }
}
