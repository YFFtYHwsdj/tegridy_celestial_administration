# 用例 3：归正辖属跨辖区兼领阻断与书信执行流程（延迟用例）。

#   2) 移除越辖兼领后，归正辖属资格通过，派发书信 tca_admin_rectification.10（days = 1）；
#   3) 次日书信到达并由 AI 自动确认后，目标官员的直属上级成功变更为法理上级路主，头衔与法理不变。

                debug_only = yes
                is_alive = yes
                root = { has_title = title:h_china }
            }
            clear_saved_scope = tca_art_rectify_duke
            clear_saved_scope = tca_art_rectify_road_holder

            random_vassal = {
                limit = {
                    tca_is_admin_rectification_target_official_trigger = yes
                    primary_title.tier = tier_kingdom
                    any_vassal = {
                        tca_is_admin_rectification_target_official_trigger = yes

                    save_scope_as = tca_art_rectify_duke
                }
            }
            random_vassal_or_below = {
                limit = {
                    exists = scope:tca_art_rectify_duke
                    exists = scope:tca_art_rectify_road_holder
                    this != scope:tca_art_rectify_duke
                    this != scope:tca_art_rectify_road_holder
                    primary_title.tier = tier_county
                    is_landed = yes
                    NOT = { primary_title = { target_is_de_jure_liege_or_above = scope:tca_art_rectify_road_holder.primary_title } }
