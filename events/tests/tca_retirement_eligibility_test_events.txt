tca_retirement_eligibility.2001 = {
    type = character_event hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = { limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            # 用同宫廷对照人物实测外部健康偏移（对照基线health=5）：成熟世界中新建家门
            # 可能已聘到原版宫廷医师，其scaling_employer_court_modifiers会给宫廷内除宫主外
            # 的所有角色加health，档位随AI实际聘任情况变化。这里按运行时实测值补偿，
            # 不写死常量；全新局未聘医师时偏移为0，不添加任何补偿，与旧例等价。
            scope:tca_rh_control = { tca_test_restore_character_effect = yes }
            if = { limit = { scope:tca_rh_control = { is_alive = yes } }
                set_variable = { name = tca_rh_health_offset value = { value = scope:tca_rh_control.health subtract = 5 } }
            }
            else = { set_variable = { name = tca_rh_health_offset value = 0 } debug_log = "TCA_RH_HEALTH_OFFSET_FALLBACK" }
            # 输出实测偏移供复核；saved value只用于debug_log_scopes显示，随后清除。
            save_scope_value_as = { name = tca_rh_health_offset_log value = var:tca_rh_health_offset }
            debug_log = "TCA_RH_HEALTH_OFFSET" debug_log_scopes = yes
