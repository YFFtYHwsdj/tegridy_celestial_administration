tca_frontier_situation_test.1 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_frontier_situation_test.1 tca_test_frontier_situation"
        remove_character_flag = tca_test_tca_frontier_situation_test_case_failed
        if = { limit = { NOT = { exists = situation:tca_northern_frontier } }
            tca_frontier_situation_start_effect = yes
            if = { limit = {
                exists = situation:tca_northern_frontier
                situation:tca_northern_frontier = { any_situation_sub_region = { count = 1 } }
                situation:tca_northern_frontier.situation_sub_region:tca_frontier_all = {
                    sub_region_current_phase = tca_frontier_tension
                    any_situation_sub_region_county = { this = title:c_henan }
                    any_situation_sub_region_county = { this = title:c_lingzhou }
                }
            } debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_situation 1" }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_situation 1" }
            situation:tca_northern_frontier = { save_scope_as = tca_frontier_situation_test_original }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_all CATALYST = tca_frontier_to_amity }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_all CATALYST = tca_frontier_to_warfare }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_all CATALYST = tca_frontier_to_warfare }
            tca_frontier_situation_start_effect = yes
            if = { limit = {
                situation:tca_northern_frontier = { this = scope:tca_frontier_situation_test_original }
                situation:tca_northern_frontier.situation_sub_region:tca_frontier_all = {
                    phase_takeover_points:tca_frontier_amity = 25
                    phase_takeover_points:tca_frontier_warfare = 50
                }
            } debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_situation 2" }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_situation 2" }
            title:h_china = { set_variable = { name = tca_frontier_situation_checkpoint_phase value = 1 } }
            debug_log = "TCA_TEST WAIT tca_test_frontier_situation 45 days"
            trigger_event = { id = tca_frontier_situation_test.2 days = 45 }
        }
        else = {
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_frontier_situation"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
            tca_test_lane_end_effect = { GROUP = tca_frontier_situation_test }
        }
    }
}

tca_frontier_situation_test.10 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_frontier_situation_test.10 tca_test_frontier_unification"
        remove_character_flag = tca_test_tca_frontier_situation_test_case_failed
        title:h_china.holder = { save_scope_as = tca_frontier_unification_test_actor }
        title:c_henan.holder = { save_scope_as = tca_frontier_unification_test_vassal }
        if = {
            limit = {
                NOT = { exists = situation:tca_northern_frontier }
                scope:tca_frontier_unification_test_actor = { is_independent_ruler = yes is_at_war = no }
                scope:tca_frontier_unification_test_vassal = {
                    NOT = { this = scope:tca_frontier_unification_test_actor }
                    top_liege = scope:tca_frontier_unification_test_actor
                }
            }
            tca_frontier_situation_start_effect = yes
            if = { limit = { scope:tca_frontier_unification_test_actor = { NOT = { can_execute_decision = tca_frontier_unification_decision } } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_unification 1"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_unification 1" }
            scope:tca_frontier_unification_test_actor = {
                create_title_and_vassal_change = { type = conquest save_scope_as = tca_frontier_unification_test_change }
            }
            every_county_in_region = {
                region = tca_frontier_region
                limit = { NOT = { this = title:c_lingzhou } NOT = { this = title:c_henan } }
                change_title_holder = { holder = scope:tca_frontier_unification_test_actor change = scope:tca_frontier_unification_test_change }
            }
            resolve_title_and_vassal_change = scope:tca_frontier_unification_test_change
            if = { limit = { scope:tca_frontier_unification_test_actor = { NOT = { tca_frontier_unification_land_trigger = yes } } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_unification 2"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_unification 2" }
            scope:tca_frontier_unification_test_vassal = { create_title_and_vassal_change = { type = conquest save_scope_as = tca_frontier_unification_test_last } }
            title:c_lingzhou = { change_title_holder = { holder = scope:tca_frontier_unification_test_vassal change = scope:tca_frontier_unification_test_last } }
            resolve_title_and_vassal_change = scope:tca_frontier_unification_test_last
            if = { limit = { scope:tca_frontier_unification_test_actor = { can_execute_decision = tca_frontier_unification_decision } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_unification 3"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_unification 3" }
            if = { limit = { scope:tca_frontier_unification_test_vassal = { NOT = { can_execute_decision = tca_frontier_unification_decision } } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_unification 4"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_unification 4" }
            # 构造一场区外战争，确认和平要求不按战争目标筛选。
            title:e_byzantium.holder = { save_scope_as = tca_frontier_unification_test_enemy }
            scope:tca_frontier_unification_test_actor = {
                set_variable = { name = tca_frontier_probe_target value = scope:tca_frontier_unification_test_enemy }
                start_war = {
                    cb = tca_frontier_probe_county_cb
                    target = scope:tca_frontier_unification_test_enemy
                    target_title = scope:tca_frontier_unification_test_enemy.capital_county
                }
            }
            if = { limit = { scope:tca_frontier_unification_test_actor = { is_at_war = yes NOT = { can_execute_decision = tca_frontier_unification_decision } } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_unification 5"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_unification 5" }
            # 原生二级战争回调沿CB上下文排队；留出真实tick，不同帧结束战争。
            debug_log = "TCA_TEST WAIT tca_test_frontier_unification 2 days"
            trigger_event = { id = tca_frontier_situation_test.11 days = 2 }
        }
        else = {
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_frontier_unification"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
            tca_test_lane_end_effect = { GROUP = tca_frontier_situation_test }
        }
    }
