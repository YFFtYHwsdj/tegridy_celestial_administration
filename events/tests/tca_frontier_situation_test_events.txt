tca_frontier_situation_test.1 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_frontier_situation_test.1 tca_test_frontier_situation"
        remove_character_flag = tca_test_tca_frontier_situation_test_case_failed
        if = { limit = { NOT = { exists = situation:tca_northern_frontier } }
            tca_frontier_situation_start_effect = yes
            if = { limit = { exists = situation:tca_northern_frontier situation:tca_northern_frontier.situation_sub_region:tca_frontier_north = { sub_region_current_phase = tca_frontier_amity  } situation:tca_northern_frontier.situation_sub_region:tca_frontier_west = { sub_region_current_phase = tca_frontier_warfare  } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_situation 1"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_situation 1" }
            situation:tca_northern_frontier = { save_scope_as = tca_frontier_situation_test_original }
            # 重复创建须保留同一对象及已有进度；先推北境25点，西陲仍为零。
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_north CATALYST = tca_frontier_to_tension }
            tca_frontier_situation_start_effect = yes
            if = { limit = { situation:tca_northern_frontier = { this = scope:tca_frontier_situation_test_original } situation:tca_northern_frontier.situation_sub_region:tca_frontier_north = { sub_region_current_phase = tca_frontier_amity phase_takeover_points:tca_frontier_tension = 25 } situation:tca_northern_frontier.situation_sub_region:tca_frontier_west = { sub_region_current_phase = tca_frontier_warfare phase_takeover_points:tca_frontier_tension = 0 } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_situation 2"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_situation 2" }
            if = { limit = { situation:tca_northern_frontier.situation_sub_region:tca_frontier_north = { any_situation_sub_region_county = { this = title:c_henan } NOT = { any_situation_sub_region_county = { this = title:c_lingzhou } } } situation:tca_northern_frontier.situation_sub_region:tca_frontier_west = { any_situation_sub_region_county = { this = title:c_lingzhou } NOT = { any_situation_sub_region_county = { this = title:c_henan } } } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_situation 3"
            }
            else = { add_character_flag = tca_test_tca_frontier_situation_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_situation 3" }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_north CATALYST = tca_frontier_to_tension }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_north CATALYST = tca_frontier_to_tension }
            tca_frontier_situation_catalyst_effect = { REGION = tca_frontier_north CATALYST = tca_frontier_to_tension }
            trigger_event = { id = tca_frontier_situation_test.2 days = 2 }
        }
        else = {
            # 不结束玩家或其他探针已有的局势；本例仅拥有自己新建的对象。
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_frontier_situation"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
            tca_test_lane_end_effect = { GROUP = tca_frontier_situation_test }
        }
    }
}
