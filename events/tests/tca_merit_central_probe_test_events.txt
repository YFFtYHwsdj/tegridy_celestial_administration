tca_merit_central_probe.1 = {
    type = character_event
    hidden = yes
    immediate = {
        if = { limit = { has_character_flag = tca_test_delayed_case_active } debug_log = "TCA_TEST CASE_BUSY" }
        else = {
            debug_log = "TCA_TEST BEGIN tca_merit_central_probe.1 tca_test_merit_central_appointment"
            remove_character_flag = tca_test_case_failed
            if = {
                limit = {
                    debug_only = yes is_alive = yes title:h_china.holder = root
                    tgp_has_access_to_ministry_trigger = yes
                    exists = cp:minister_justice
                    cp:minister_justice = title:e_minister_of_justice.holder
                    situation:dynastic_cycle = {
                        situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
                        situation_top_has_catalyst = catalyst_hegemon_appointing_low_merit_councillor
                        situation_top_sub_region = { phase_takeover_points:situation_dynastic_cycle_phase_instability < 900 }
                    }
                }
                cp:minister_justice = { save_scope_as = tca_merit_central_original }
                # 正式锁品后全国可能暂时没有二品候选；本项必须自备合法候选，不能依赖旧缓存。
                # 只准备普通家门/考试/原生九级（一品）功绩资格并清冷却，不写候选分、不指定继承人。
                # 接口隔离标记使其不进入正式年度人口与授品，保留原生任命筛选。
                tca_test_create_merit_interface_character_effect = { CHARACTER = tca_merit_central_control GOVERNMENT = celestial_government }
                scope:tca_merit_central_control = {
                    tca_test_add_native_merit_probe_exams_effect = yes
                    add_trait = education_stewardship_1
                    change_merit = 24000
                    clear_appointment_timeout = yes
                }
                add_character_flag = tca_test_delayed_case_active
                debug_log = "TCA_TEST WAIT tca_test_merit_central_appointment 2 days"
                trigger_event = { id = tca_merit_central_probe.1011 days = 2 }
            }
            else = {
                debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_merit_central_appointment"
                if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
                tca_test_continue_delayed_effect = yes
            }
        }
    }
}

tca_merit_central_probe.1003 = {
    type = character_event hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = {
            limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            scope:tca_merit_central_high = {
                save_scope_value_as = { name = tca_merit_central_diag_high_merit value = merit }

                save_scope_value_as = { name = tca_merit_central_diag_high_stable value = var:tca_merit_central_probe_worker_stable }

            # 先令两个临时人物退出候选，再恢复原任，避免其 Merit 干扰原继承名单。
            scope:tca_merit_central_low = { remove_variable = tca_merit_central_probe_reporter tca_test_cleanup_merit_probe_character_effect = yes }
            scope:tca_merit_central_high = { remove_variable = tca_merit_central_probe_reporter tca_test_cleanup_merit_probe_character_effect = yes }
            # 真实获任后在同链失职；已排队的授品必须重新核对持有关系，不能只凭获任记录兑现。
            assign_councillor_type = { type = minister_justice target = scope:tca_merit_central_transient fire_on_actions = yes remove_existing_councillor = yes }
            # 原版AI也可任命低品大臣，原任不保证足二品；恢复须保留其原状并核对实际惩罚。
            # 只比较恢复前后同链点数，不以跨日总量猜测催化剂。
            situation:dynastic_cycle.situation_top_sub_region = { set_variable = { name = tca_merit_central_restore_expected value = phase_takeover_points:situation_dynastic_cycle_phase_instability } }
            scope:tca_merit_central_original = {
                if = {
                    limit = { NOT = { has_appointment_level_for_title = title:e_minister_of_justice } }
                    situation:dynastic_cycle.situation_top_sub_region = { change_variable = { name = tca_merit_central_restore_expected add = minor_situation_catalyst_gain } }
                }
            }
            # 资源只比较恢复前后同链，避免将自然跨月收入误作探针污染。
            scope:tca_merit_central_original = {
                save_scope_value_as = { name = tca_merit_central_restore_merit value = merit }
                save_scope_value_as = { name = tca_merit_central_restore_level value = merit_level }
            }
            # 恢复真实原任是探针复原，不新建正式中央授品；只临时排除此人本次入口。
            # 保留其已有核定/余额/pending；专用所有权标记避免删除原本已有的隔离标记。
            scope:tca_merit_central_original = {
                if = {
                    limit = { NOT = { has_character_flag = tca_merit_interface_probe } }
                    add_character_flag = tca_merit_interface_probe
                    add_character_flag = tca_merit_central_restore_owns_isolation
                }
            }
            if = { limit = { NOT = { cp:minister_justice = scope:tca_merit_central_original } } assign_councillor_type = { type = minister_justice target = scope:tca_merit_central_original fire_on_actions = yes remove_existing_councillor = yes } }
            scope:tca_merit_central_original = {
                if = {
                    limit = { has_character_flag = tca_merit_central_restore_owns_isolation }
                    remove_character_flag = tca_merit_interface_probe
                    remove_character_flag = tca_merit_central_restore_owns_isolation
                }
            }
            # 恢复后立即按最终议会/头衔重算继承并核对原首位；不再等两日让AI自然改选。
            scope:tca_merit_central_original = { scriptedtests_recalculate_succession = yes }
            if = { limit = { cp:minister_justice = scope:tca_merit_central_original title:e_minister_of_justice.holder = scope:tca_merit_central_original title:e_minister_of_justice.current_heir = scope:tca_merit_central_heir } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_central_appointment 9" }

            # 在正式回调完成后恢复事前剩余日数；不改获任资格、候选分或局势催化剂。
            # 快照取于 .1001，至本链确实经过四日；.1004 再等两日验证自然流逝。

                }
            }
            if = { limit = { situation:dynastic_cycle.situation_top_sub_region = { phase_takeover_points:situation_dynastic_cycle_phase_instability = var:tca_merit_central_restore_expected } } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_central_appointment 8" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_central_appointment 8" }
            if = { limit = { scope:tca_merit_central_original = { merit = scope:tca_merit_central_restore_merit merit_level = scope:tca_merit_central_restore_level NOT = { has_variable = tca_merit_central_probe_reporter } } } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_central_appointment 11" }
