tca_bribery_natural_test.3 = {
    type = character_event hidden = yes
    immediate = {
        tca_test_validate_environment_effect = yes
        if = { limit = { NOT = { has_character_flag = tca_test_suite_aborted } }
            change_variable = { name = tca_bribery_natural_days add = 1 }
            tca_test_bribery_natural_observe_effect = yes
            # 至少等待390天，超过完整12个月考虑窗口与最长5日答复；三档已齐才收口。
            # 真实全朝候选可能先选其他人，最多再等一轮年度窗口，不强制发送、不改概率。
            if = { limit = { OR = { var:tca_bribery_natural_days < 390 AND = {
                var:tca_bribery_natural_days < 780
                OR = { var:tca_bribery_natural_seen_1 = 0 var:tca_bribery_natural_seen_2 = 0 var:tca_bribery_natural_seen_3 = 0 }
            } } }
                trigger_event = { id = tca_bribery_natural_test.3 days = 1 }
            }
            else = { trigger_event = { id = tca_bribery_natural_test.4 days = 0 } }
        }
    }
}
