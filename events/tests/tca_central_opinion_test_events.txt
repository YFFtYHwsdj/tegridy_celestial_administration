                scope:tca_co_test_a = { has_appointment_level_for_title = title:e_minister_chancellor merit_level = scope:tca_co_test_b.merit_level }

                add_opinion = { target = scope:tca_co_test_a modifier = pleased_opinion opinion = { value = 0 subtract = "opinion(scope:tca_co_test_a)" } }

                scriptedtests_recalculate_succession = yes
            }
            add_opinion = { target = scope:tca_co_test_a modifier = pleased_opinion opinion = { value = 0 subtract = "opinion(scope:tca_co_test_a)" } }
            add_opinion = { target = scope:tca_co_test_b modifier = pleased_opinion opinion = { value = 0 subtract = "opinion(scope:tca_co_test_b)" } }
            trigger_event = { id = tca_central_opinion_test.3 days = 1 }

    }
}
