tca_treasury_secret_test.1 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_treasury_secret_test.1 tca_test_treasury_secret_risk"
        remove_character_flag = tca_test_tca_treasury_secret_test_case_failed
        random_vassal = { limit = { has_treasury = yes } save_scope_as = tca_treasury_test_victim }
        if = {
            limit = {
                exists = scope:tca_treasury_test_victim
                NOT = { any_secret = { secret_type = secret_siphoned_treasury secret_target = scope:tca_treasury_test_victim } }
            }
            save_scope_value_as = { name = tca_treasury_test_original_gold value = gold }
            save_scope_value_as = { name = tca_treasury_test_original_treasury value = treasury }
            save_scope_value_as = { name = tca_treasury_test_original_legitimacy value = legitimacy }
            scope:tca_treasury_test_victim = { save_scope_value_as = { name = tca_treasury_test_original_victim_treasury value = treasury } }
            start_scheme = { type = tca_treasury_test_hidden_scheme target_title = scope:tca_treasury_test_victim.primary_title save_scope_as = tca_treasury_test_scheme }
            save_scope_as = owner
            scope:tca_treasury_test_victim.primary_title = { save_scope_as = target }
            scope:tca_treasury_test_scheme = {
                set_variable = { name = embezzlement_stake value = 0 }
                set_variable = { name = embezzlement_stake_initial value = 10 }

            }
            trigger_event = { id = tca_treasury_secret_test.101 days = 1 }
        }
        else = {
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_treasury_secret_risk"
            tca_test_lane_end_effect = { GROUP = tca_treasury_secret_test }
        }
    }
}

        scope:tca_treasury_test_scheme = {
            set_variable = { name = embezzlement_stake value = 500 }
            tca_treasury_resolve_secret_effect = { TARGET = scope:tca_treasury_test_victim }
        }
        if = { limit = { any_secret = { secret_type = secret_siphoned_treasury secret_target = scope:tca_treasury_test_victim var:embezzlement_stake = 125 } } debug_log = "TCA_TEST CHECK_PASS tca_test_treasury_secret_risk 5" }
        else = { add_character_flag = tca_test_tca_treasury_secret_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_treasury_secret_risk 5" }
        # 零赃款收尾不能删除或重写已有秘密。
        scope:tca_treasury_test_scheme = {
            remove_variable = tca_treasury_secret_resolved
            set_variable = { name = embezzlement_stake value = 0 }
            tca_treasury_resolve_secret_effect = { TARGET = scope:tca_treasury_test_victim }
        }
        if = { limit = { any_secret = { secret_type = secret_siphoned_treasury secret_target = scope:tca_treasury_test_victim var:embezzlement_stake = 125 } } debug_log = "TCA_TEST CHECK_PASS tca_test_treasury_secret_risk 6" }
        else = { add_character_flag = tca_test_tca_treasury_secret_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_treasury_secret_risk 6" }
        scope:tca_treasury_test_scheme = { end_scheme = yes }

        # 精确归还结果事件可能改变的金币、国库与正统性，不把夹具收益带入后续用例。
        add_gold = { value = scope:tca_treasury_test_original_gold subtract = gold }
        add_treasury = { value = scope:tca_treasury_test_original_treasury subtract = treasury }
        add_legitimacy = { value = scope:tca_treasury_test_original_legitimacy subtract = legitimacy }
        scope:tca_treasury_test_victim = { add_treasury = { value = scope:tca_treasury_test_original_victim_treasury subtract = treasury } }
        # 前置已确认对此目标没有旧秘密，只清理本轮夹具，不接触其它受害者。
        every_secret = { limit = { secret_type = secret_siphoned_treasury secret_target = scope:tca_treasury_test_victim } remove_secret = yes }
        if = { limit = { has_character_flag = tca_test_tca_treasury_secret_test_case_failed } debug_log = "TCA_TEST FAIL tca_test_treasury_secret_risk" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_failed add = 1 } } }
        else = { debug_log = "TCA_TEST PASS tca_test_treasury_secret_risk" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_passed add = 1 } } }
        remove_character_flag = tca_test_tca_treasury_secret_test_case_failed
        tca_test_lane_end_effect = { GROUP = tca_treasury_secret_test }
    }
}
