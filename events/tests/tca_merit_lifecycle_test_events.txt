        remove_character_flag = tca_test_case_failed
        if = {
            limit = { debug_only = yes is_alive = yes title:h_china.holder = root }

            set_variable = { name = tca_ml_startup_central_count value = 0 }
            set_variable = { name = tca_ml_startup_central_wrong_count value = 0 }

            set_variable = { name = tca_ml_startup_central_side_effect_count value = 0 }
            # 本模块在任何新建夹具之前执行；不能用后续测试人物的U诊断推断真实开局。
            every_living_character = {
                limit = {
                    has_character_flag = tca_merit_active
                    NOT = { has_variable = tca_merit_route }
                    NOT = { tca_merit_interface_probe_trigger = yes }
                }
                root = { change_variable = { name = tca_ml_startup_unclassified_count add = 1 } }
                save_scope_as = tca_ml_unclassified_actor
                save_scope_value_as = { name = tca_ml_unclassified_age value = age }
                debug_log = "TCA_ML STARTUP_UNCLASSIFIED_ACTOR"
                debug_log_scopes = yes
            }
            every_living_character = {
                limit = { top_liege = root government_has_flag = government_is_celestial is_independent_ruler = no }
                if = { limit = { has_variable = tca_merit_initial_level } root = { change_variable = { name = tca_ml_startup_initial_captured_total add = 1 } } }
                if = { limit = { merit_level >= 3 } root = { change_variable = { name = tca_ml_startup_current_high_count add = 1 } } }
            }
            every_living_character = {
                limit = { has_variable = tca_merit_initial_level has_character_flag = tca_merit_active top_liege = root government_has_flag = government_is_celestial }
                if = { limit = { var:tca_merit_initial_level >= 3 } root = { change_variable = { name = tca_ml_startup_high_count add = 1 } } }
                if = { limit = { var:tca_merit_approved_level < var:tca_merit_initial_level } root = { change_variable = { name = tca_ml_startup_regression_count add = 1 } } }
            }
            # 用原生scope数值快照输出，不让本地化延迟求值读取已删除的即时变量。
            save_scope_value_as = { name = tca_ml_diag_high_count value = var:tca_ml_startup_high_count }
            save_scope_value_as = { name = tca_ml_diag_regression_count value = var:tca_ml_startup_regression_count }
            save_scope_value_as = { name = tca_ml_diag_initial_captured_total value = var:tca_ml_startup_initial_captured_total }
            save_scope_value_as = { name = tca_ml_diag_current_high_count value = var:tca_ml_startup_current_high_count }
            save_scope_value_as = { name = tca_ml_diag_unclassified_count value = var:tca_ml_startup_unclassified_count }

            debug_log_scopes = yes

            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 1" }
            if = { limit = { var:tca_ml_startup_high_count > 0 var:tca_ml_startup_regression_count = 0 } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_native_startup 2" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 2" }
            if = { limit = { var:tca_ml_startup_unclassified_count = 0 } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_native_startup 3" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 3" }
            # 独立按九职头衔列举预期，不以正式目标值计算断言；缺少开局核定时应失败。
            every_living_character = {
                limit = { top_liege = root government_has_flag = government_is_celestial is_independent_ruler = no }
                if = {
                    limit = { OR = { has_title = title:e_minister_chancellor has_title = title:e_minister_censor has_title = title:e_minister_grand_marshal } }
                    root = { change_variable = { name = tca_ml_startup_central_count add = 1 } }
                    if = { limit = { OR = { NOT = { var:tca_merit_approved_level ?= 9 } NOT = { merit_level = 9 } NOT = { has_variable = tca_merit_initial_balance } } } root = { change_variable = { name = tca_ml_startup_central_wrong_count add = 1 } } }
                }
                else_if = {
                    limit = { OR = { has_title = title:e_minister_of_personnel has_title = title:e_minister_of_revenue has_title = title:e_minister_of_rites has_title = title:e_minister_of_war has_title = title:e_minister_of_justice has_title = title:e_minister_of_works } }
                    root = { change_variable = { name = tca_ml_startup_central_count add = 1 } }
                    if = { limit = { OR = { NOT = { var:tca_merit_approved_level >= 8 } merit_level < 8 NOT = { has_variable = tca_merit_initial_balance } } } root = { change_variable = { name = tca_ml_startup_central_wrong_count add = 1 } } }
                }
                else_if = {
                    limit = { has_variable = tca_merit_initial_level NOT = { var:tca_merit_approved_level ?= var:tca_merit_initial_level } }
                    root = { change_variable = { name = tca_ml_startup_noncentral_changed_count add = 1 } }
                }
            }
            if = { limit = { var:tca_ml_startup_central_count = 9 var:tca_ml_startup_central_wrong_count = 0 } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_native_startup 4" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 4" }
            if = { limit = { var:tca_ml_startup_noncentral_changed_count = 0 } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_native_startup 5" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 5" }
            # 本项在开局同日检查；合法的跨日保护尚未释放，释放结果由延迟边界用例检查。
            every_living_character = {
                limit = { top_liege = root has_variable = tca_merit_initial_balance }
                if = {
                    limit = { OR = { NOT = { merit = var:tca_merit_initial_balance } has_variable = tca_merit_grant_token has_variable = tca_merit_last_ordinary_year has_variable = tca_merit_last_ordinary_success_year has_character_flag = tca_merit_central_pending has_character_flag = tca_merit_startup_audit_failed } }
                    root = { change_variable = { name = tca_ml_startup_central_side_effect_count add = 1 } }
                }
            }
            if = { limit = { var:tca_ml_startup_central_side_effect_count = 0 } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_native_startup 6" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_native_startup 6" }
            remove_variable = tca_ml_startup_high_count
            remove_variable = tca_ml_startup_regression_count
            remove_variable = tca_ml_startup_initial_captured_total
            remove_variable = tca_ml_startup_current_high_count
            remove_variable = tca_ml_startup_unclassified_count
            remove_variable = tca_ml_startup_central_count
            remove_variable = tca_ml_startup_central_wrong_count
            remove_variable = tca_ml_startup_noncentral_changed_count
            remove_variable = tca_ml_startup_central_side_effect_count
            if = { limit = { has_character_flag = tca_test_case_failed } debug_log = "TCA_TEST FAIL tca_test_merit_native_startup" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_failed add = 1 } } }

        }
        else = { debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_merit_native_startup" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } } }
        remove_character_flag = tca_test_case_failed
    }
}
