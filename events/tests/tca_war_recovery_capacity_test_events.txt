tca_war_recovery_capacity_test.10 = {
    type = character_event hidden = yes
    immediate = {
        scope:tca_war_recovery_capacity_test_victim = { create_story = { type = tca_frontier_support_record save_scope_as = tca_fs_record } }
        scope:tca_fs_record = {
            save_scope_as = tca_war_recovery_capacity_test_record
            set_variable = { name = tca_ws_schema value = 2 }
            set_variable = { name = tca_fs_closed value = 1 }
            set_variable = { name = tca_fs_defender value = scope:tca_war_recovery_capacity_test_victim }
            set_variable = { name = tca_fs_attacker value = scope:tca_war_recovery_capacity_test_enemy }
            set_variable = { name = tca_fd_country value = scope:tca_war_recovery_capacity_test_leader.primary_title }
            set_variable = { name = tca_fd_attacker_country value = scope:tca_war_recovery_capacity_test_enemy.primary_title }
        }
        scope:tca_war_recovery_capacity_test_victim = { create_story = { type = tca_frontier_support_row save_scope_as = tca_war_recovery_capacity_test_row } }
        scope:tca_war_recovery_capacity_test_row = {
            set_variable = { name = tca_fs_parent value = scope:tca_fs_record }
            set_variable = { name = tca_fs_person value = scope:tca_war_recovery_capacity_test_leader }
            set_variable = { name = tca_fs_kind value = 2 }
            set_variable = { name = tca_fs_depth value = 1 }
            set_variable = { name = tca_fs_accepted value = yes }
            set_variable = { name = tca_fd_eligible value = yes }
        }
        scope:tca_fs_record = { add_to_variable_list = { name = tca_fs_rows target = scope:tca_war_recovery_capacity_test_row } }
        set_variable = { name = tca_war_recovery_capacity_test_count value = 0 }
        scope:tca_war_recovery_capacity_test_enemy = { every_sub_realm_county = {
            if = { limit = { root = { var:tca_war_recovery_capacity_test_count < var:tca_war_recovery_capacity_test_size } }
                save_scope_as = tca_war_recovery_capacity_test_county
                root = { change_variable = { name = tca_war_recovery_capacity_test_count add = 1 } }
                scope:tca_war_recovery_capacity_test_victim = { create_story = { type = tca_frontier_recovery_loss save_scope_as = tca_war_recovery_capacity_test_loss } }
                scope:tca_war_recovery_capacity_test_loss = {
                    set_variable = { name = tca_fd_parent value = scope:tca_fs_record }
                    set_variable = { name = tca_fd_title value = scope:tca_war_recovery_capacity_test_county }
                    set_variable = { name = tca_fd_holder value = scope:tca_war_recovery_capacity_test_leader }
                    set_variable = { name = tca_fd_government value = scope:tca_war_recovery_capacity_test_leader.government_type }
                    set_variable = { name = tca_fd_transfer_kind value = 0 }
                }
                scope:tca_fs_record = {
                    add_to_variable_list = { name = tca_fd_losses target = scope:tca_war_recovery_capacity_test_loss }
                    add_to_variable_list = { name = tca_fd_goals target = scope:tca_war_recovery_capacity_test_county }
                }
            }
            else_if = { limit = { scope:tca_fs_record = { NOT = { has_variable = tca_war_recovery_capacity_test_excluded } } }
                scope:tca_fs_record = { set_variable = { name = tca_war_recovery_capacity_test_excluded value = prev } }
                holder = { scope:tca_fs_record = { set_variable = { name = tca_war_recovery_capacity_test_excluded_holder value = prev } } }
            }
        } }
        debug_log = "TCA_CAPACITY BEFORE_BEGIN"
        tca_war_recovery_capacity_test_diagnose_effect = yes
        scope:tca_fs_record = { tca_fd_begin_effect = yes }
        debug_log = "TCA_CAPACITY AFTER_BEGIN"
        tca_war_recovery_capacity_test_diagnose_effect = yes
        if = { limit = {
            var:tca_war_recovery_capacity_test_count = var:tca_war_recovery_capacity_test_size
            exists = scope:tca_fs_record.var:tca_war_recovery_capacity_test_excluded
            scope:tca_fs_record = { tca_fd_goal_count_value = root.var:tca_war_recovery_capacity_test_size }
            OR = {
                AND = { var:tca_war_recovery_capacity_test_size <= 128 scope:tca_fs_record = { var:tca_fd_state ?= 1 var:tca_fd_pending ?= scope:tca_war_recovery_capacity_test_row } }
                AND = { var:tca_war_recovery_capacity_test_size = 129 scope:tca_fs_record = { var:tca_fd_state ?= 4 var:tca_fd_result ?= 0 NOT = { has_variable = tca_fd_active } NOT = { has_variable = tca_fd_war } } }
            }
        }
            if = { limit = { var:tca_war_recovery_capacity_test_size = 12 } debug_log = "TCA_TEST CHECK_PASS tca_test_war_recovery_capacity 1" }
            if = { limit = { var:tca_war_recovery_capacity_test_size = 128 } debug_log = "TCA_TEST CHECK_PASS tca_test_war_recovery_capacity 4" }
            if = { limit = { var:tca_war_recovery_capacity_test_size = 129 } debug_log = "TCA_TEST CHECK_PASS tca_test_war_recovery_capacity 7" }
        }
        else = {
            add_character_flag = tca_test_tca_war_recovery_capacity_test_case_failed
            if = { limit = { var:tca_war_recovery_capacity_test_size = 12 } debug_log = "TCA_TEST CHECK_FAIL tca_test_war_recovery_capacity 1" }
            if = { limit = { var:tca_war_recovery_capacity_test_size = 128 } debug_log = "TCA_TEST CHECK_FAIL tca_test_war_recovery_capacity 4" }
            if = { limit = { var:tca_war_recovery_capacity_test_size = 129 } debug_log = "TCA_TEST CHECK_FAIL tca_test_war_recovery_capacity 7" }
        }
        if = { limit = { var:tca_war_recovery_capacity_test_size <= 128 }
            scope:tca_fs_record = {
                var:tca_fd_pending ?= { save_scope_as = tca_fd_choice_row }
                tca_fd_accept_effect = yes
            }
            debug_log = "TCA_CAPACITY AFTER_ACCEPT"
            tca_war_recovery_capacity_test_diagnose_effect = yes
            # 跨日等待真实started派发后才检查和议和，不留陈旧的开战回调。
            trigger_event = { id = tca_war_recovery_capacity_test.20 days = 1 }
        }
        else = {
            if = { limit = {
                scope:tca_fs_record = { tca_fd_goal_count_value = 129 NOT = { any_in_list = { variable = tca_fd_goals NOT = { holder.top_liege = scope:tca_war_recovery_capacity_test_enemy } } } }
                scope:tca_war_recovery_capacity_test_leader = { NOT = { has_variable = tca_fd_authorized } NOT = { any_character_war = { using_cb = tca_frontier_recovery_cb primary_attacker = scope:tca_war_recovery_capacity_test_leader } } }
                scope:tca_fs_record.var:tca_war_recovery_capacity_test_excluded = { holder = scope:tca_fs_record.var:tca_war_recovery_capacity_test_excluded_holder holder.top_liege = scope:tca_war_recovery_capacity_test_enemy }
            } debug_log = "TCA_TEST CHECK_PASS tca_test_war_recovery_capacity 8" }
            else = { add_character_flag = tca_test_tca_war_recovery_capacity_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_war_recovery_capacity 8" }

        }
    }
