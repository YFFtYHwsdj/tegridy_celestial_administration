				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed

				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed
					debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_validity 3"
				}

				# 4. 验证无地人物（is_landed = no）官署准入全面阻断
				scope:tca_test_emperor = {
					random_courtier = {
						limit = {
							is_landed = no
						}
						save_scope_as = tca_test_landless_courtier
					}
				}
				if = {
					limit = {
						exists = scope:tca_test_landless_courtier

							tca_corps_inspector_office_valid_trigger = no
							tca_military_inspector_office_valid_trigger = no
							tca_military_supervisor_office_valid_trigger = no

							limit = { exists = scope:tca_test_regular_duchy_vassal }

								tca_corps_inspector_office_valid_trigger = no
