	type = character_event
	hidden = yes
	immediate = {
		if = { limit = { has_character_flag = tca_test_lane_tca_military_court_positions_test_active } debug_log = "TCA_TEST CASE_BUSY" }

			debug_log = "TCA_TEST BEGIN tca_military_court_positions_test.1 tca_test_military_court_positions_validity"
			remove_character_flag = tca_test_tca_military_court_positions_test_case_failed
			if = {
				limit = {
					game_start_date = 1066.9.15
					title:h_china = {
						exists = holder
						holder = {
							is_alive = yes
							is_independent_ruler = yes
							government_has_flag = government_is_celestial
						}
					}
				}
				title:h_china.holder = {
					save_scope_as = tca_test_emperor
				}

				# 1. 验证天朝领主自身不能作为廷臣担任军职（is_ruler = no 阻断）

					limit = {
						scope:tca_test_emperor = {
							tca_military_supervisor_validity_trigger = no
							tca_military_inspector_validity_trigger = no
							tca_corps_inspector_validity_trigger = no
						}
					}
					debug_log = "TCA_TEST CHECK_PASS tca_test_military_court_positions_validity 1"
				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed

				}

				if = {
					limit = {
						scope:tca_test_emperor = {
							tca_military_officer_base_eligibility_trigger = no
						}

					debug_log = "TCA_TEST CHECK_PASS tca_test_military_court_positions_validity 2"
				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed
					debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_validity 2"

				# 3. 验证未获系统授权 flag 时，任何人均无法被手动任免（第四阶段闭环）

				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed
					debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_validity 3"
				}

				# 4. 验证无地人物（is_landed = no）官署准入全面阻断
				scope:tca_test_emperor = {
					random_courtier = {

							is_landed = no

						save_scope_as = tca_test_landless_courtier
					}
				}

					limit = {

							tca_corps_inspector_office_valid_trigger = no
							tca_military_inspector_office_valid_trigger = no
							tca_military_supervisor_office_valid_trigger = no

							limit = { exists = scope:tca_test_regular_duchy_vassal }

								tca_corps_inspector_office_valid_trigger = no
								tca_military_inspector_office_valid_trigger = yes
								tca_military_supervisor_office_valid_trigger = yes
							}
						}
					}
					debug_log = "TCA_TEST CHECK_PASS tca_test_military_court_positions_validity 5"
				}

					add_character_flag = tca_test_tca_military_court_positions_test_case_failed

					title:h_china = {
						exists = holder
						holder = {
							is_alive = yes
							is_independent_ruler = yes
						}
					}
				}
				title:h_china.holder = {
					save_scope_as = tca_test_emperor
				}

				# 1. 验证任期磨勘冷却期（730天）对在职调动收集的拦截
				scope:tca_test_emperor = {
					random_courtier = {
						limit = {
							tca_military_officer_base_eligibility_trigger = yes
						}
						save_scope_as = tca_test_cooldown_candidate
					}
				}
				if = {
					limit = {
						exists = scope:tca_test_cooldown_candidate
					}
					scope:tca_test_cooldown_candidate = {
						add_character_flag = {
							flag = tca_military_appointment_cooldown
							days = 730
						}
					}
					# 验证处于冷却期时，无法通过收集筛选（NOT has_character_flag = tca_military_appointment_cooldown 为 false）
					if = {
						limit = {
							scope:tca_test_cooldown_candidate = {
								has_character_flag = tca_military_appointment_cooldown
							}
						}
						debug_log = "TCA_TEST CHECK_PASS tca_test_military_court_positions_cooldown_and_empty_pool 1"
					}
					else = {
						add_character_flag = tca_test_tca_military_court_positions_test_case_failed
						debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_cooldown_and_empty_pool 1"
					}
					# 恢复清理
					scope:tca_test_cooldown_candidate = {
						remove_character_flag = tca_military_appointment_cooldown
					}
				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed
					debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_cooldown_and_empty_pool 1"
				}

				# 2. 验证待阙池耗尽（宁缺毋滥）安全熔断：空池下执行除授效果绝不任命且安全退出
				scope:tca_test_emperor = {
					random_vassal = {
						limit = {
							highest_held_title_tier = tier_duchy
							government_has_flag = government_is_celestial
							NOT = { employs_court_position = tca_military_inspector_court_position }
						}
						save_scope_as = tca_test_empty_target_liege
					}
				}
				# 确保候选池为空
				every_in_list = {
					list = tca_military_candidate_pool
					remove_from_list = tca_military_candidate_pool
				}
				if = {
					limit = {
						exists = scope:tca_test_empty_target_liege
					}
					scope:tca_test_empty_target_liege = {
						save_scope_as = target_liege
						tca_military_fill_single_position_effect = {
							POSITION = tca_military_inspector_court_position
						}
					}
				}

				# 断言：空池下坚决未聘任任何官员，保持出缺，宁缺毋滥
				if = {
					limit = {
						trigger_if = {
							limit = { exists = scope:tca_test_empty_target_liege }
							scope:tca_test_empty_target_liege = {
								NOT = { employs_court_position = tca_military_inspector_court_position }
							}
						}
						NOT = { any_in_list = { list = tca_military_candidate_pool count >= 1 } }
					}
					debug_log = "TCA_TEST CHECK_PASS tca_test_military_court_positions_cooldown_and_empty_pool 2"
				}
				else = {
					add_character_flag = tca_test_tca_military_court_positions_test_case_failed
					debug_log = "TCA_TEST CHECK_FAIL tca_test_military_court_positions_cooldown_and_empty_pool 2"
				}

				# 结果汇报
				if = {
					limit = { has_character_flag = tca_test_tca_military_court_positions_test_case_failed }
					debug_log = "TCA_TEST FAIL tca_test_military_court_positions_cooldown_and_empty_pool"
					if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_failed add = 1 } }
				}
				else = {
					debug_log = "TCA_TEST PASS tca_test_military_court_positions_cooldown_and_empty_pool"
					if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_passed add = 1 } }
				}
			}
			else = {
				debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_military_court_positions_cooldown_and_empty_pool"
				if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
			}
			remove_character_flag = tca_test_tca_military_court_positions_test_case_failed
		}
	}
