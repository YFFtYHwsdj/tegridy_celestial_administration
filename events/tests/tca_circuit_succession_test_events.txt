            clear_saved_scope = tca_cs_holder
            random_in_global_list = {
                variable = tca_circuit_titles
                limit = {
                    tca_circuit_is_tracked_road_trigger = yes
                    has_variable = tca_circuit_kind
                    var:tca_circuit_kind = 1
                    tca_circuit_has_celestial_vassal_holder_trigger = yes
                    NOT = { has_variable = tca_circuit_pending_appointment }
                    holder = {
                        OR = {
                            has_realm_law = celestial_appointment_succession_law
                            AND = {
                                NOT = { vassal_contract_has_flag = celestial_military_appointment }
                                NOT = { vassal_contract_has_flag = locked_to_partition }
                            }
                        }
                    }
                }
                save_scope_as = tca_cs_road
            }
            if = {
                limit = { exists = scope:tca_cs_road }
                scope:tca_cs_road = { holder = { save_scope_as = tca_cs_holder } }
                # P4 默认关闭；用例显式开启补法开关，验证“补充等效”路径。
                set_global_variable = { name = tca_circuit_civil_law_ensure_enabled value = yes }
                # 前置已保证两条合法分支之一：已具备任命继承法时 ensure 守卫空操作（调用前后状态不变，
                # 作为有效断言核对）；缺少且契约允许时 ensure 补挂。两者第一次调用后都应为“已具备”。
                scope:tca_cs_road = { tca_circuit_ensure_civil_appointment_law_effect = yes }
                if = {
                    limit = { scope:tca_cs_holder = { has_realm_law = celestial_appointment_succession_law } }
                    debug_log = "TCA_TEST CHECK_PASS tca_test_circuit_civil_law_ensure 1"
                }
                else = {
                    add_character_flag = tca_test_tca_circuit_succession_test_case_failed
                    debug_log = "TCA_TEST CHECK_FAIL tca_test_circuit_civil_law_ensure 1"
                }
                # 第二次调用仍应稳定（已具备时为空操作，不重复处理）。
                scope:tca_cs_road = { tca_circuit_ensure_civil_appointment_law_effect = yes }
                if = {
                    limit = { scope:tca_cs_holder = { has_realm_law = celestial_appointment_succession_law } }
                    debug_log = "TCA_TEST CHECK_PASS tca_test_circuit_civil_law_ensure 2"
                }
                else = {
                    add_character_flag = tca_test_tca_circuit_succession_test_case_failed
                    debug_log = "TCA_TEST CHECK_FAIL tca_test_circuit_civil_law_ensure 2"
                }
                if = {
                    limit = { scope:tca_cs_road = { holder = scope:tca_cs_holder } }
                    debug_log = "TCA_TEST CHECK_PASS tca_test_circuit_civil_law_ensure 3"
                }
                else = {
                    add_character_flag = tca_test_tca_circuit_succession_test_case_failed
                    debug_log = "TCA_TEST CHECK_FAIL tca_test_circuit_civil_law_ensure 3"
                }
                # 恢复现场：关闭补法开关。
                remove_global_variable = tca_circuit_civil_law_ensure_enabled
            }
            else = {
                add_character_flag = tca_cs_precondition_failed
                debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_circuit_civil_law_ensure"
            }
        }
        else = {
            add_character_flag = tca_cs_precondition_failed
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_circuit_civil_law_ensure"
        }
        if = { limit = { has_character_flag = tca_cs_precondition_failed } remove_character_flag = tca_cs_precondition_failed if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } } }
        else_if = {
            limit = { has_character_flag = tca_test_tca_circuit_succession_test_case_failed }
            debug_log = "TCA_TEST FAIL tca_test_circuit_civil_law_ensure"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_failed add = 1 } }
        }
        else = {
            debug_log = "TCA_TEST PASS tca_test_circuit_civil_law_ensure"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_passed add = 1 } }
        }
        remove_character_flag = tca_test_tca_circuit_succession_test_case_failed
        tca_test_lane_end_effect = { GROUP = tca_circuit_succession_test }
    }
}
