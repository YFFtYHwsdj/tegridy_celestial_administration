tca_county_return_test.203 = {
    type = character_event hidden = yes
    immediate = {
        if = { limit = { scope:tca_vcrt_target = { holder = scope:tca_vcrt_actor }
            scope:tca_vcrt_actor = { NOT = { has_variable = tca_county_return_title } }
            scope:tca_vcrt_reserved = { NOT = { has_variable = tca_admin_rectification_reserved_successor } }
        } debug_log = "TCA_TEST CHECK_PASS tca_test_county_return_invalidation 1" }
        else = { add_character_flag = tca_test_tca_county_return_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_county_return_invalidation 1" }
        scope:tca_vcrt_actor = {
            set_realm_capital = scope:tca_vcrt_home
            tca_county_return_start_effect = { TITLE = scope:tca_vcrt_target }
            # 旧事件不能执行或清理新单；旧占位已被释放后，新单应能启动。
            tca_county_return_execute_effect = { SERIAL = scope:tca_vcrt_old_token ATTEMPT = 0 }
        }
        if = { limit = { scope:tca_vcrt_actor = { has_variable = tca_county_return_title var:tca_county_return_serial > scope:tca_vcrt_old_token }
            scope:tca_vcrt_target = { holder = scope:tca_vcrt_actor }
        } debug_log = "TCA_TEST CHECK_PASS tca_test_county_return_invalidation 2" }
        else = { add_character_flag = tca_test_tca_county_return_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_county_return_invalidation 2" }
        scope:tca_vcrt_actor = {
            var:tca_county_return_successor = {
                save_scope_as = tca_vcrt_other_reservation
                # 模拟占位被其他流程取得，原单不得删掉他单标记；同时强制触发一次补选。
                set_variable = { name = tca_admin_rectification_reserved_successor value = scope:tca_vcrt_home days = 30 }
                set_variable = { name = tca_county_return_reserved_by value = root days = 30 }
                set_variable = { name = tca_county_return_reserved_serial value = 999 days = 30 }
            }
        }
        debug_log = "TCA_TEST WAIT tca_test_county_return_invalidation 3 days replacement"
        trigger_event = { id = tca_county_return_test.204 days = 3 }
    }
}
