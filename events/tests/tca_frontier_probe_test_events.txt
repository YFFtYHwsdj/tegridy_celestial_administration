tca_frontier_probe_test.1 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_frontier_probe_test.1 tca_test_frontier_protected_county"
        remove_character_flag = tca_test_tca_frontier_probe_test_case_failed
        clear_saved_scope = tca_frontier_probe_test_actor
        clear_saved_scope = tca_frontier_probe_test_defender
        random_vassal_or_below = {
            limit = { is_alive = yes is_landed = yes highest_held_title_tier = tier_county is_at_war = no government_has_flag = government_is_celestial vassal_contract_has_flag = celestial_province_military NOT = { has_variable = tca_frontier_probe_target } }
            save_scope_as = tca_frontier_probe_test_actor
        }
        title:k_xia.holder ?= {
            random_vassal_or_below = {
                limit = { is_alive = yes is_landed = yes highest_held_title_tier = tier_county is_at_war = no exists = liege liege = { is_at_war = no can_vassals_be_attacked = no } top_liege = { can_vassals_be_attacked = no } }
                save_scope_as = tca_frontier_probe_test_defender
                liege = { save_scope_as = tca_frontier_probe_test_upper }
                capital_county = { save_scope_as = tca_frontier_probe_test_target }
            }
        }
        if = { limit = { exists = scope:tca_frontier_probe_test_actor exists = scope:tca_frontier_probe_test_defender exists = scope:tca_frontier_probe_test_target }
            scope:tca_frontier_probe_test_actor = { tca_test_protect_character_effect = yes set_variable = { name = tca_frontier_probe_target value = scope:tca_frontier_probe_test_defender } save_scope_as = actor }
            scope:tca_frontier_probe_test_defender = { tca_test_protect_character_effect = yes save_scope_as = recipient }
            scope:tca_frontier_probe_test_upper = { tca_test_protect_character_effect = yes }
            if = { limit = { scope:tca_frontier_probe_test_defender.top_liege = { can_vassals_be_attacked = no has_realm_law = top_liege_redirected_to_border_wars_law } }
                debug_log = "TCA_TEST CHECK_PASS tca_test_frontier_protected_county 1"
            }
            else = { add_character_flag = tca_test_tca_frontier_probe_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_frontier_protected_county 1" }
            run_interaction = { interaction = tca_frontier_probe_war_interaction actor = scope:actor recipient = scope:recipient redirect = no send_threshold = accept }
            trigger_event = { id = tca_frontier_probe_test.2 days = 1 }
        }
        else = {
            debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_frontier_protected_county"
            if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } }
            tca_test_lane_end_effect = { GROUP = tca_frontier_probe_test }
        }
    }
}
