tca_merit_reset_probe.3 = {
    type = character_event hidden = yes
    immediate = {
        debug_log = "TCA_TEST BEGIN tca_merit_reset_probe.3 tca_test_merit_reset_matrix"
        remove_character_flag = tca_test_case_failed
        if = {
            limit = { debug_only = yes title:h_china.holder = root }
            clear_variable_list = tca_merit_reset_matrix_persistent
            set_variable = { name = tca_merit_reset_index value = 1 }
            while = {
                limit = { var:tca_merit_reset_index <= 10 }
                tca_test_create_merit_interface_character_effect = { CHARACTER = tca_merit_reset_matrix_actor GOVERNMENT = celestial_government }
                scope:tca_merit_reset_matrix_actor = {
                    add_to_temporary_list = tca_merit_reset_matrix
                    root = { add_to_variable_list = { name = tca_merit_reset_matrix_persistent target = prev } }
                    tca_test_add_native_merit_probe_exams_effect = yes
                    add_character_modifier = tca_merit_probe_approved_base_modifier
                    set_variable = { name = tca_merit_reset_target value = root.var:tca_merit_reset_index }
                    change_merit = 1000000.125
                    change_merit_no_experience = -7000.5
                    if = { limit = { var:tca_merit_reset_target = 10 } set_variable = { name = tca_merit_reset_target value = 2 } change_merit_experience = -1000500.125 }
                    # 独立列明原版门槛；下一门槛用于从外部行为证明没有残余经验。
                    switch = {
                        trigger = var:tca_merit_reset_target
                        1 = { set_variable = { name = tca_merit_reset_base value = 100 } set_variable = { name = tca_merit_reset_next value = 1000 } }
                        2 = { set_variable = { name = tca_merit_reset_base value = 1000 } set_variable = { name = tca_merit_reset_next value = 2000 } }
                        3 = { set_variable = { name = tca_merit_reset_base value = 2000 } set_variable = { name = tca_merit_reset_next value = 3500 } }
                        4 = { set_variable = { name = tca_merit_reset_base value = 3500 } set_variable = { name = tca_merit_reset_next value = 5500 } }
                        5 = { set_variable = { name = tca_merit_reset_base value = 5500 } set_variable = { name = tca_merit_reset_next value = 8500 } }
                        6 = { set_variable = { name = tca_merit_reset_base value = 8500 } set_variable = { name = tca_merit_reset_next value = 12000 } }
                        7 = { set_variable = { name = tca_merit_reset_base value = 12000 } set_variable = { name = tca_merit_reset_next value = 17000 } }
                        8 = { set_variable = { name = tca_merit_reset_base value = 17000 } set_variable = { name = tca_merit_reset_next value = 23000 } }
                        9 = { set_variable = { name = tca_merit_reset_base value = 23000 } }
                    }
                }
                change_variable = { name = tca_merit_reset_index add = 1 }
            }
            if = { limit = { any_in_list = { list = tca_merit_reset_matrix count = 10 merit = 992999.625 } } debug_log = "TCA_TEST CHECK_PASS tca_test_merit_reset_matrix 1" }
            else = { add_character_flag = tca_test_case_failed debug_log = "TCA_TEST CHECK_FAIL tca_test_merit_reset_matrix 1" }
            add_character_flag = tca_test_delayed_case_active
            trigger_event = { id = tca_merit_reset_probe.3001 days = 2 }
        }
        else = { debug_log = "TCA_TEST PRECONDITION_FAIL tca_test_merit_reset_matrix" if = { limit = { has_character_flag = tca_test_suite_running } change_variable = { name = tca_test_suite_skipped add = 1 } } tca_test_continue_delayed_effect = yes }
    }
}
