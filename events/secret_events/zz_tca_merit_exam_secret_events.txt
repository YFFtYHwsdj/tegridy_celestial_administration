# 原版礼部审议舞弊事件覆盖；保留原事件与选项，只在撤销考试资格后刷新cap。
namespace = secrets

secrets.4001 = {
	id_override_priority = 1
	type = character_event
	title = secrets.4001.t
	desc = secrets.4001.desc
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:minister
		animation = disgust
	}
	lower_right_portrait = {
		character = scope:secret_target
		trigger = { scope:secret_target != scope:minister }
	}
	theme = secret
	
	trigger = {
		OR = {
			has_character_modifier = tgp_passed_palace_exam_modifier
			has_character_modifier = tgp_passed_metropolitan_exam_modifier
		}
	}

	immediate = {
		title:e_minister_of_rites.holder ?= {
			if = {
				limit = { this != root }
				save_scope_as = minister
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:minister }
			}
			scope:secret_target = {
				save_scope_as = minister
			}
		}
	}

	# Try to bribe one last time
	option = {
		name = secrets.4001.a
		scope:secret_target = {
			send_interface_message = {
				type = event_msg_merit_effect_good
				title = exam_cheater_bribe_t
				left_icon = root
				right_icon = title:e_minister_of_rites
				root = {
					pay_short_term_gold = {
						target = scope:minister
						gold = monumental_gold_value
					}
					add_character_modifier = {
						modifier = tgp_known_exam_cheater_modifier
						years = 5
					}
				}
			}
		}
		ai_chance = {
			base = 0
		}
	}
	# Give back to the government
	option = {
		name = secrets.4001.b
		scope:secret_target = {
			send_interface_message = {
				type = event_msg_merit_effect_good
				title = exam_cheater_reimburse_t
				left_icon = root
				right_icon = title:e_minister_of_rites
				root = {
					pay_gold_to_treasury = {
						target = scope:minister
						value = major_gold_value
					}
					pay_gold_to_treasury = {
						target = root
						value = major_gold_value
					}
					add_character_modifier = {
						modifier = tgp_known_exam_cheater_modifier
						years = 5
					}
				}
				situation:dynastic_cycle ?= {
					if = {
						limit = {
							situation_top_has_catalyst = catalyst_minister_caught_exam_cheater
						}
						trigger_situation_catalyst = {
							catalyst = catalyst_minister_caught_exam_cheater
							character = root
						}
					}
				}
			}
		}
		ai_chance = {
			base = 0
		}
	}
	# Face consequences
	option = {
		name = secrets.4001.c
		scope:secret_target = {
			send_interface_message = {
				type = event_msg_merit_effect_good
				title = exam_cheater_guilty_t
				left_icon = root
				right_icon = title:e_minister_of_rites
				root = {
					if = {
						limit = {
							has_character_modifier = tgp_passed_palace_exam_modifier
						}
						remove_character_flag = passed_palace_exam
						remove_character_modifier = tgp_passed_palace_exam_modifier
						tca_merit_after_reward_effect = yes
					}
					else = {
						remove_character_flag = passed_metropolitan_exam
						remove_character_modifier = tgp_passed_metropolitan_exam_modifier
						tca_merit_after_reward_effect = yes
					}
					force_step_down_landed_titles = yes
				}
				situation:dynastic_cycle ?= {
					if = {
						limit = {
							situation_top_has_catalyst = catalyst_minister_caught_exam_cheater
						}
						trigger_situation_catalyst = {
							catalyst = catalyst_minister_caught_exam_cheater
							character = root
						}
					}
				}
			}
		}
		ai_chance = {
			base = 10
		}
	}
}
