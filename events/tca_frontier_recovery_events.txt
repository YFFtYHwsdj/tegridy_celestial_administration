# 原生书信：收到即接受或拒绝，无延期选项。排队次日投递不构成选择期限。
namespace = tca_frontier_recovery

tca_frontier_recovery.1 = {
    type = letter_event
    sender = scope:tca_fd_sender
    opening = tca_frontier_recovery.opening
    desc = tca_frontier_recovery.1.desc
    trigger = {
        exists = scope:tca_fs_record
        exists = scope:tca_fd_choice_row
        scope:tca_fs_record = { var:tca_fd_state = 1 var:tca_fd_pending = scope:tca_fd_choice_row tca_fd_land_valid_trigger = yes }
        scope:tca_fd_choice_row ?= { tca_fd_leader_valid_trigger = yes }
    }
    on_trigger_fail = {
        scope:tca_fs_record ?= { if = { limit = { var:tca_fd_state = 1 var:tca_fd_pending = scope:tca_fd_choice_row } tca_fd_end_effect = { RESULT = 0 } } }
    }
    option = {
        name = tca_frontier_recovery.1.a
        custom_tooltip = tca_frontier_recovery.1.a.tt
        hidden_effect = { scope:tca_fs_record = { tca_fd_accept_effect = yes } }
        ai_chance = { base = 0 modifier = { add = tca_fd_accept_weight_value } }
    }
    option = {
        name = tca_frontier_recovery.1.b
        timeout_option = yes
        hidden_effect = { scope:tca_fs_record = { tca_fd_refuse_effect = yes } }
        ai_chance = { base = 50 }
    }
}

        hidden_effect = { scope:tca_fd_invitation = { tca_fd_join_refuse_effect = yes } }
        ai_chance = { base = 50 }
    }
}
