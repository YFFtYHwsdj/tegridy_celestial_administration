# Engine tests; run in a disposable, paused 1066 China new game.
# Each test prepares its own laws; no reliance on file execution order.

# Repeated effect/expect parts follow vanilla tests/event_target_lists_tests.txt.
# skip_effects deliberately bypasses enactment costs, on_pass and event queues.

tca_test_court_local_only = {
    name = "tca_test_court_local_only"
    assert = {
        game_start_date = 1066.9.15
        title:h_china = {
            exists = holder
            holder = {
                is_alive = yes
                is_independent_ruler = yes
                government_has_flag = government_is_celestial
                tca_external_official_names_trigger = no
            }
        }

    }
    effect = {
        title:h_china.holder = {
            save_scope_as = tca_test_sovereign
            if = { limit = { has_realm_law = tca_court_system_song_law } remove_realm_law = tca_court_system_song_law }
            if = { limit = { has_realm_law = tca_court_system_tang_law } remove_realm_law = tca_court_system_tang_law }
            set_variable = { name = tca_court_system_migrated_v2 value = yes }
            if = { limit = { NOT = { has_realm_law = tca_local_names_default_law } } add_realm_law_skip_effects = tca_local_names_default_law }
            if = { limit = { NOT = { has_realm_law = tca_central_names_default_law } } add_realm_law_skip_effects = tca_central_names_default_law }
            if = { limit = { NOT = { has_realm_law = tca_judicial_institutions_disabled_law } } add_realm_law_skip_effects = tca_judicial_institutions_disabled_law }
            tca_refresh_official_name_flags_effect = yes
        }
    }
    effect = {
        scope:tca_test_sovereign = {
            if = { limit = { NOT = { has_realm_law = tca_local_names_song_law } } add_realm_law_skip_effects = tca_local_names_song_law }
            tca_ensure_court_system_law_effect = yes
        }
    }
    expect = {
        scope:tca_test_sovereign = {
            tca_uses_song_local_names_trigger = yes
            tca_uses_song_central_names_trigger = no
            has_realm_law = tca_central_names_default_law
            tca_has_judicial_institutions_trigger = no
        }
    }
}
