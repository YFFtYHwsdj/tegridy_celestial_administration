# Disposable paused 1066 China test realm only. Each case rebuilds its own state.
# Realm-wide assertions use a nonempty snapshot of ALL eligible civilian circuits.
# on_pass cases deliberately do not call repair after real add_realm_law.
# Other cases isolate settled-state behavior with skip_effects + formal maintenance.
# These immediate tests do not establish UI costs, asynchronous replies or reload behavior.

tca_test_judicial_law_enabled_all_circuits = {
    name = "tca_test_judicial_law_enabled_all_circuits"
    assert = { game_start_date = 1066.9.15
        title:h_china = {
            exists = holder
            holder = {
                is_alive = yes
                is_imprisoned = no
                is_independent_ruler = yes
                government_has_flag = government_is_celestial
                tca_external_official_names_trigger = no
                any_vassal_or_below = {
                    count >= 2
                    is_alive = yes
                    is_imprisoned = no
                    government_has_flag = government_is_celestial
                    highest_held_title_tier = tier_kingdom
                    NOT = { tgp_is_any_minister = yes }
                    NOT = { vassal_contract_has_flag = celestial_province_military }
                    NOT = { vassal_contract_has_flag = celestial_province_protectorate }
                    any_held_title = {
                        tier = tier_kingdom
                        is_landless_type_title = no
                        is_noble_family_title = no
                        OR = { exists = title_capital_county exists = holder.capital_county }
                    }
                }
            }
        }

    }
    effect = { title:h_china.holder = { tca_test_prepare_judicial_law_realm_effect = yes } }
    expect = {
        scope:tca_test_law_sovereign = {
            has_realm_law = tca_judicial_institutions_enabled_law
            any_in_list = { variable = tca_test_law_circuits count >= 2 }
            NOT = { any_in_list = {
                variable = tca_test_law_circuits
                NOT = {
                    holder = var:tca_test_law_original_governor
                    trigger_if = {
                        limit = { holder = { is_imprisoned = no NOT = { tgp_is_any_minister = yes } } }
                        holder = { tca_is_judicial_governor_trigger = yes tca_uses_judicial_arrest_trigger = yes }
                    }
                    tca_is_judicial_circuit_trigger = yes
                    tca_has_judicial_title_trigger = yes
                    var:song_judicial_title = {
                        this = prev.var:tca_test_law_original_position
                        is_landless_type_title = yes
                        tier = tier_duchy
                        tca_is_valid_judicial_holder_trigger = yes
                        holder = {
                            liege = prev.var:song_judicial_circuit.holder
                            has_realm_law = celestial_appointment_succession_law
                            tca_is_judicial_commissioner_trigger = yes
                            tca_uses_judicial_arrest_trigger = yes
                        }
                    }
                }
            } }
        }
    }
}
